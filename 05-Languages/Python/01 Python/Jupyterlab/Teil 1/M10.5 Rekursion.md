Rekursion ist eine elegante Programmiertechnik, bei der eine Funktion sich selbst aufruft, um komplexe Probleme in kleinere, ähnliche Teilprobleme zu zerlegen.

## 5.1 Grundprinzip der Rekursion

Jede rekursive Funktion benötigt zwei wichtige Komponenten: einen Basisfall und einen rekursiven Fall.

### Die Anatomie einer rekursiven Funktion

```python
def einfache_rekursion(n):
    """
    Grundstruktur einer rekursiven Funktion.
    
    Args:
        n: Parameter, der sich bei jedem Aufruf ändert
    """
    # 1. BASISFALL - Stoppt die Rekursion
    if n <= 0:
        print(f"Basisfall erreicht mit n={n}")
        return "Fertig!"
    
    # 2. REKURSIVER FALL - Funktion ruft sich selbst auf
    print(f"Rekursiver Aufruf mit n={n}")
    return einfache_rekursion(n - 1)  # Problem verkleinern

# Test der Rekursion
print("=== Einfache Rekursion ===")
result = einfache_rekursion(5)
print(f"Endergebnis: {result}")
```

### Klassisches Beispiel: Fakultät

```python
def fakultät(n):
    """
    Berechnet die Fakultät einer Zahl rekursiv.
    n! = n × (n-1) × (n-2) × ... × 1
    """
    # Basisfall
    if n <= 1:
        print(f"Basisfall: {n}! = 1")
        return 1
    
    # Rekursiver Fall
    print(f"Berechne {n}! = {n} × {n-1}!")
    ergebnis = n * fakultät(n - 1)
    print(f"Zurück: {n}! = {ergebnis}")
    return ergebnis

print("=== Fakultät berechnen ===")
print(f"5! = {fakultät(5)}")
print()

# Ohne Debug-Ausgaben
def fakultät_sauber(n):
    """Fakultät ohne Debug-Ausgaben."""
    if n <= 1:
        return 1
    return n * fakultät_sauber(n - 1)

print("Verschiedene Fakultäten:")
for i in range(6):
    print(f"{i}! = {fakultät_sauber(i)}")
```

def fibonacci_rekursiv(n):
    """
    Berechnet die n-te Fibonacci-Zahl rekursiv.
    F(0) = 0, F(1) = 1, F(n) = F(n-1) + F(n-2)
    """
    # Basisfälle
    if n <= 0:
        return 0
    elif n == 1:
        return 1
    
    # Rekursiver Fall
    return fibonacci_rekursiv(n - 1) + fibonacci_rekursiv(n - 2)

def fibonacci_iterativ(n):
    """Fibonacci iterativ - viel effizienter!"""
    if n <= 0:
        return 0
    elif n == 1:
        return 1
    
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

def summe_rekursiv(liste):
    """Berechnet die Summe einer Liste rekursiv."""
    # Basisfall: leere Liste
    if not liste:
        return 0
    
    # Rekursiver Fall: erstes Element + Summe des Rests
    return liste[0] + summe_rekursiv(liste[1:])

def länge_rekursiv(liste):
    """Berechnet die Länge einer Liste rekursiv."""
    if not liste:
        return 0
    return 1 + länge_rekursiv(liste[1:])

def maximum_rekursiv(liste):
    """Findet das Maximum einer Liste rekursiv."""
    if not liste:
        return None
    if len(liste) == 1:
        return liste[0]
    
    rest_max = maximum_rekursiv(liste[1:])
    return liste[0] if liste[0] > rest_max else rest_max

def liste_umkehren_rekursiv(liste):
    """Kehrt eine Liste rekursiv um."""
    if len(liste) <= 1:
        return liste
    return liste_umkehren_rekursiv(liste[1:]) + [liste[0]]

    """
    Findet alle Werte für einen Schlüssel in verschachtelten Dictionaries/Listen.

    gefundene_werte = []

import sys

def tiefe_rekursion(n):
    """Testet die Rekursionstiefe."""
    if n <= 0:
        return 0

    count = 0

# Beispiele wo Rekursion BESSER ist
def potenz_rekursiv(basis, exponent):
    """Potenz mit cleverer Rekursion (Divide & Conquer)."""
    if exponent == 0:
        return 1
    elif exponent == 1:
        return basis
    elif exponent % 2 == 0:
        # Gerade: (basis^(exp/2))^2
        half_power = potenz_rekursiv(basis, exponent // 2)
        return half_power * half_power
    else:
        # Ungerade: basis * basis^(exp-1)
        return basis * potenz_rekursiv(basis, exponent - 1)

def potenz_iterativ(basis, exponent):
    """Potenz iterativ."""
    ergebnis = 1
    for _ in range(exponent):
        ergebnis *= basis
    return ergebnis

# Vergleich bei großen Exponenten

iter_result = potenz_iterativ(basis, exponent)

print(f"{basis}^{exponent}:")
print(f"  Rekursiv: {rek_result}")
print(f"  Iterativ: {iter_result}")
print(f"  Built-in: {builtin_result}")
print(f"  Alle gleich: {rek_result == iter_result == builtin_result}")

def debug_rekursion(original_func):
    """Decorator für Rekursions-Debugging."""
    call_depth = 0
    
    def wrapper(*args, **kwargs):
        nonlocal call_depth
        
        # Eingang
        indent = "  " * call_depth
        print(f"{indent}→ {original_func.__name__}{args}")
        
        call_depth += 1
        result = original_func(*args, **kwargs)
        call_depth -= 1
        
        # Ausgang
        print(f"{indent}← {result}")
        return result
    
    return wrapper

def memoize(func):
    """Decorator für Memoization."""
    cache = {}
    
    def wrapper(*args):
        if args in cache:
            return cache[args]
        result = func(*args)
        cache[args] = result
        return result
    
    # Cache-Info für Debugging
    wrapper.cache = cache
    wrapper.cache_info = lambda: {"hits": len([v for v in cache.values()]), "size": len(cache)}
    
    return wrapper

@memoize

def factorial_tail_recursive(n, accumulator=1):
    """Tail-recursive Fakultät - theoretisch optimierbar."""
    if n <= 1:
        return accumulator
    return factorial_tail_recursive(n - 1, n * accumulator)

def fibonacci_tail_recursive(n, a=0, b=1):
    """Tail-recursive Fibonacci."""
    if n == 0:
        return a
    if n == 1:
        return b
    return fibonacci_tail_recursive(n - 1, b, a + b)

def summe_tail_recursive(liste, accumulator=0):
    """Tail-recursive Listensumme."""
    if not liste:
        return accumulator
    return summe_tail_recursive(liste[1:], accumulator + liste[0])

class TreeNode:
    """Einfacher Baum-Knoten für rekursive Operationen."""
    
    def __init__(self, value, children=None):
        self.value = value
        self.children = children or []
    
    def add_child(self, child_node):
        self.children.append(child_node)
    
    def __repr__(self):
        return f"TreeNode({self.value})"

def tree_height(node):
    """Berechnet die Höhe eines Baums rekursiv."""
    if not node.children:
        return 1
    return 1 + max(tree_height(child) for child in node.children)

def tree_size(node):
    """Zählt alle Knoten im Baum rekursiv."""
    return 1 + sum(tree_size(child) for child in node.children)

def find_in_tree(node, target_value):
    """Sucht einen Wert im Baum rekursiv."""
    if node.value == target_value:
        return node
    
    for child in node.children:
        result = find_in_tree(child, target_value)
        if result:
            return result
    
    return None

def tree_values(node):
    """Sammelt alle Werte aus dem Baum rekursiv."""
    values = [node.value]
    for child in node.children:
        values.extend(tree_values(child))
    return values

def print_tree(node, depth=0):
    """Zeigt den Baum hierarchisch an."""
    print("  " * depth + str(node.value))
    for child in node.children:
        print_tree(child, depth + 1)

## Zusammenfassung

Rekursion ist eine mächtige Technik, die bei richtiger Anwendung zu eleganten und verständlichen Lösungen führt:

**Wann Rekursion verwenden:**

- ✅ Problem ist natürlich rekursiv (Bäume, Fraktale)
- ✅ Divide & Conquer Ansatz passt
- ✅ Datenstruktur ist hierarchisch
- ✅ Mathematische Definition ist rekursiv

**Wann Iteration bevorzugen:**

- ✅ Einfache Schleifen ausreichen
- ✅ Performance kritisch
- ✅ Große Datenmengen
- ✅ Stack Overflow-Gefahr

**Wichtige Konzepte:**

- 🎯 **Basisfall**: Stoppt die Rekursion
- 🔄 **Rekursiver Fall**: Problem verkleinern
- 📈 **Memoization**: Effizienz durch Caching
- 🐍 **Python-Limit**: Standard 1000 Aufrufe

**Nächste Schritte:**

- Scope und Namensräume für besseres Verständnis von Variablen-Sichtbarkeit
- Closures als Verbindung zu funktionaler Programmierung
- Dekoratoren als praktische Anwendung aller gelernten Konzepte

Rekursion erweitert Ihr Programmier-Toolkit um eine elegante Methode für komplexe Probleme! 🎯
