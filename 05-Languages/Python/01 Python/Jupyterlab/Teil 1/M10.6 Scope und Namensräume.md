Scope (Gültigkeitsbereich) und Namensräume bestimmen, wo und wie Variablen in Python verfügbar sind. Das Verstehen dieser Konzepte ist entscheidend für sauberen, fehlerfreien Code.

## 6.1 LEGB-Regel verstehen

Python folgt der LEGB-Regel für die Variablenauflösung: Local → Enclosing → Global → Built-in.

### Die vier Scope-Ebenen

```python
# Built-in Scope: Eingebaute Funktionen und Konstanten
print("Built-in Funktionen:", dir(__builtins__)[:10])  # Erste 10 anzeigen
print("Built-in Typ:", type(len))  # len ist im Built-in Scope
```

```python
# Global Scope: Modul-Ebene
globale_variable = "Ich bin global"
print(f"Globale Variable: {globale_variable}")
```

```python
def demonstriere_legb():
    """Demonstriert die LEGB-Regel."""
    
    # Enclosing Scope (wird später bei verschachtelten Funktionen gezeigt)
    enclosing_var = "Ich bin enclosing"
    
    def innere_funktion():
        # nonlocal enclosing_var
        # Local Scope: Innerhalb der Funktion
        lokale_variable = "Ich bin lokal"
        print("=== Zugriff aus innerer_funktion ===")
        print(f"Local: {lokale_variable}")
        print(f"Enclosing: {enclosing_var}")  
        print(f"Global: {globale_variable}")
        print(f"Built-in: {len([1,2,3])}")  # len ist built-in
    
    innere_funktion()
    
    print("\n=== Zugriff aus äußerer Funktion ===")
    print(f"Enclosing: {enclosing_var}")
    print(f"Global: {globale_variable}")
    # print(f"Local: {lokale_variable}")  # ❌ NameError - nicht verfügbar!

demonstriere_legb()
```

```python
def scope_visualisierung():
    """Zeigt die verschiedenen Scopes visuell."""
    
    # Global scope lookup
    global_var = "Global verfügbar"
    
    def äußere_funktion(param):
        # Enclosing scope
        enclosing_var = f"Enclosing mit {param}"
        
        def innere_funktion():
            # Local scope
            local_var = "Nur hier verfügbar"
            
            # Alle Scopes abfragen
            scopes = {
                "Local": locals(),
                "Global": {k: v for k, v in globals().items() 
                          if not k.startswith('__') and k in ['global_var', 'globale_variable']},
                "Built-in": {"len": len, "print": print, "str": str}  # Beispiele
            }
            
            return scopes
        
        return innere_funktion
    
    # Scope-Inspektion
    inner = äußere_funktion("Test")
    scopes = inner()
    
    print("=== Scope-Visualisierung ===")
    for scope_name, variables in scopes.items():
        print(f"\n{scope_name} Scope:")
        for name, value in list(variables.items())[:5]:  # Erste 5
            print(f"  {name}: {value}")
```

### Built-in Scope erkunden

```python
def erkunde_builtins():
    """Erkundet den Built-in Namespace."""
    
    # Alle Built-in Namen anzeigen
    builtin_names = dir(__builtins__)
    
    print("=== Built-in Scope ===")
    print(f"Anzahl Built-in Namen: {len(builtin_names)}")
    
    # Kategorien von Built-ins
    kategorien = {
        "Typen": [],
        "Funktionen": [],
        "Exceptions": [],
        "Konstanten": []
    }
    
    for name in builtin_names:
        if not name.startswith('_'):  # Private ignorieren
            obj = getattr(__builtins__, name)
            
            if isinstance(obj, type):
                if issubclass(obj, Exception):
                    kategorien["Exceptions"].append(name)
                else:
                    kategorien["Typen"].append(name)
            elif callable(obj):
                kategorien["Funktionen"].append(name)
            else:
                kategorien["Konstanten"].append(name)
    
    for kategorie, items in kategorien.items():
        print(f"\n{kategorie} ({len(items)}):")
        print(f"  {', '.join(items[:10])}")  # Erste 10
        if len(items) > 10:
            print(f"  ... und {len(items)-10} weitere")

erkunde_builtins()
```

## 6.2 global und nonlocal Schlüsselwörter

Mit `global` und `nonlocal` können Sie Variablen in äußeren Scopes modifizieren.

### Das global Schlüsselwort

```python
# Globale Variable
counter = 0
app_config = {"debug": False, "version": "1.0"}

def increment_counter():
    """Erhöht den globalen Counter."""
    global counter
    counter += 1
    print(f"Counter erhöht auf: {counter}")

def reset_counter():
    """Setzt den Counter zurück."""
    global counter
    counter = 0
    print("Counter zurückgesetzt")

def update_config(key, value):
    """Aktualisiert die globale Konfiguration."""
    global app_config
    app_config[key] = value
    print(f"Konfiguration aktualisiert: {app_config}")

def problematisch_ohne_global():
    """Zeigt was passiert OHNE global."""
    # Das erstellt eine NEUE lokale Variable!
    counter = 999
    print(f"Lokaler Counter: {counter}")

print("=== Global Keyword ===")
print(f"Initial Counter: {counter}")

increment_counter()
increment_counter()
increment_counter()

problematisch_ohne_global()
print(f"Globaler Counter nach problematisch_ohne_global: {counter}")

reset_counter()
update_config("debug", True)
```

### Das nonlocal Schlüsselwort

```python
def erstelle_zähler(start=0):
    """Erstellt einen Zähler mit nonlocal."""
    count = start
    
    def increment():
        nonlocal count
        count += 1
        return count
    
    def decrement():
        nonlocal count
        count -= 1
        return count
    
    def get_value():
        return count  # Kein nonlocal nötig - nur lesen
    
    def reset():
        nonlocal count
        count = start
        return count
    
    # Funktionen-Dictionary zurückgeben
    return {
        "inc": increment,
        "dec": decrement,
        "get": get_value,
        "reset": reset
    }

def demonstriere_nonlocal():
    """Zeigt nonlocal in Aktion."""
    
    # Zwei unabhängige Zähler erstellen
    zähler1 = erstelle_zähler(10)
    zähler2 = erstelle_zähler(100)
    
    print("=== Nonlocal Demonstration ===")
    print(f"Zähler 1 Start: {zähler1['get']()}")
    print(f"Zähler 2 Start: {zähler2['get']()}")
    
    # Zähler 1 manipulieren
    print(f"Zähler 1 +3: {zähler1['inc']()}, {zähler1['inc']()}, {zähler1['inc']()}")
    print(f"Zähler 1 -1: {zähler1['dec']()}")
    
    # Zähler 2 manipulieren
    print(f"Zähler 2 +5: {zähler2['inc']()}")
    
    print(f"Zähler 1 final: {zähler1['get']()}")
    print(f"Zähler 2 final: {zähler2['get']()}")
    
    # Reset
    print(f"Zähler 1 nach Reset: {zähler1['reset']()}")

demonstriere_nonlocal()
```

### Verschachtelte Scopes mit nonlocal

```python
def erstelle_bankkonto(anfangssaldo=0):
    """Simuliert ein Bankkonto mit verschachtelten Scopes."""
    saldo = anfangssaldo
    transaktionen = []
    
    def einzahlen(betrag):
        nonlocal saldo
        if betrag > 0:
            saldo += betrag
            transaktionen.append(f"Einzahlung: +{betrag}€")
            return True
        return False
    
    def abheben(betrag):
        nonlocal saldo
        if betrag > 0 and betrag <= saldo:
            saldo -= betrag
            transaktionen.append(f"Abhebung: -{betrag}€")
            return True
        return False
    
    def kontostand():
        return saldo
    
    def verlauf():
        return transaktionen.copy()
    
    def zinsen_hinzufügen(zinssatz):
        nonlocal saldo
        zinsen = saldo * (zinssatz / 100)
        saldo += zinsen
        transaktionen.append(f"Zinsen: +{zinsen:.2f}€ ({zinssatz}%)")
        return zinsen
    
    # Konto-Interface
    return {
        "einzahlen": einzahlen,
        "abheben": abheben,
        "kontostand": kontostand,
        "verlauf": verlauf,
        "zinsen": zinsen_hinzufügen
    }

# Bankkonto testen
print("\n=== Bankkonto Simulation ===")
konto = erstelle_bankkonto(1000)

print(f"Anfangssaldo: {konto['kontostand']()}€")
print(f"Einzahlung 500€: {'Erfolgreich' if konto['einzahlen'](500) else 'Fehlgeschlagen'}")
print(f"Abhebung 200€: {'Erfolgreich' if konto['abheben'](200) else 'Fehlgeschlagen'}")
print(f"Abhebung 2000€: {'Erfolgreich' if konto['abheben'](2000) else 'Fehlgeschlagen'}")

zinsen = konto['zinsen'](2.5)
print(f"Zinsen (2.5%): {zinsen:.2f}€")

print(f"Endsaldo: {konto['kontostand']()}€")
print("Transaktionsverlauf:")
for transaktion in konto['verlauf']():
    print(f"  {transaktion}")
```

## 6.3 Closures

Closures sind Funktionen, die Zugriff auf Variablen aus ihrem umschließenden Scope behalten, auch nachdem dieser Scope verlassen wurde.

### Grundlagen von Closures

```python
def erstelle_gruß_funktion(gruß):
    """Erstellt eine spezialisierte Gruß-Funktion."""
    
    def grüßen(name):
        # Diese Funktion "erinnert" sich an 'gruß'
        return f"{gruß}, {name}!"
    
    return grüßen

# Verschiedene Gruß-Funktionen erstellen
hallo_sagen = erstelle_gruß_funktion("Hallo")
guten_tag_sagen = erstelle_gruß_funktion("Guten Tag")
servus_sagen = erstelle_gruß_funktion("Servus")

print("=== Closure Grundlagen ===")
print(hallo_sagen("Anna"))
print(guten_tag_sagen("Bob"))
print(servus_sagen("Charlie"))

# Closure-Eigenschaften untersuchen
print(f"\nFunktionsname: {hallo_sagen.__name__}")

    for i, cell in enumerate(hallo_sagen.__closure__):
```

```python
def erstelle_validator(min_wert, max_wert, datentyp=int):
    """Erstellt eine Validierungsfunktion mit gespeicherten Regeln."""
    
    def validiere(wert):
        # Closure "erinnert" sich an min_wert, max_wert, datentyp
        if not isinstance(wert, datentyp):
            return {"gültig": False, "fehler": f"Muss {datentyp.__name__} sein"}
        
        if wert < min_wert:
            return {"gültig": False, "fehler": f"Muss mindestens {min_wert} sein"}
        
        if wert > max_wert:
            return {"gültig": False, "fehler": f"Darf maximal {max_wert} sein"}
        
        return {"gültig": True, "wert": wert}
    
    # Zusätzliche Info für Debugging
    validiere.regeln = {"min": min_wert, "max": max_wert, "typ": datentyp}
    return validiere

def erstelle_formatter(präfix="", suffix="", transformation=str.upper):
    """Erstellt eine Formatierungsfunktion."""
    
    def formatiere(text):
        transformiert = transformation(text)
        return f"{präfix}{transformiert}{suffix}"
    
    return formatiere
```

### State-Machine mit Closures

```python
def erstelle_state_machine(initial_state):
    """Erstellt eine einfache State Machine mit Closures."""
    
    current_state = initial_state
    transitions = {}
    history = [initial_state]
    
    def add_transition(from_state, to_state, action):
        """Fügt eine Zustandsübergabe hinzu."""
        nonlocal transitions
        if from_state not in transitions:
            transitions[from_state] = {}
        transitions[from_state][action] = to_state
    
    def trigger(action):
        """Löst eine Aktion aus und ändert den Zustand."""
        nonlocal current_state, history
        
        if current_state in transitions and action in transitions[current_state]:
            old_state = current_state
            current_state = transitions[current_state][action]
            history.append(current_state)
            return {
                "erfolg": True,
                "von": old_state,
                "zu": current_state,
                "aktion": action
            }
        else:
            return {
                "erfolg": False,
                "fehler": f"Keine Transition von '{current_state}' mit '{action}'"
            }
    
    def get_state():
        """Gibt den aktuellen Zustand zurück."""
        return current_state
    
    def get_history():
        """Gibt die Zustandshistorie zurück."""
        return history.copy()
    
    def get_transitions():
        """Gibt alle möglichen Transitionen zurück."""
        return transitions.copy()
    
    return {
        "add_transition": add_transition,
        "trigger": trigger,
        "state": get_state,
        "history": get_history,
        "transitions": get_transitions
    }

# Ampel-State-Machine
print("\n=== State Machine: Ampel ===")
ampel = erstelle_state_machine("rot")

# Transitionen definieren
ampel["add_transition"]("rot", "grün", "schalten")
ampel["add_transition"]("grün", "gelb", "schalten")
ampel["add_transition"]("gelb", "rot", "schalten")

print(f"Startzustand: {ampel['state']()}")

# Ampel-Zyklus
for i in range(3):
    result = ampel["trigger"]("schalten")
    if result["erfolg"]:
        print(f"Schritt {i+1}: {result['von']} → {result['zu']}")
    else:
        print(f"Fehler: {result['fehler']}")

print(f"Geschichte: {' → '.join(ampel['history']())}")
```

### Closure Factory Pattern

```python
def erstelle_mathematik_funktionen():
    """Factory für mathematische Funktionen mit Closures."""
    
    def erstelle_potenz_funktion(exponent):
        return lambda x: x ** exponent
    
    def erstelle_polynom(koeffizienten):
        """Erstellt eine Polynomfunktion: a₀ + a₁x + a₂x² + ..."""
        def polynom(x):
            result = 0
            for i, koeff in enumerate(koeffizienten):
                result += koeff * (x ** i)
            return result
        
        polynom.koeffizienten = koeffizienten
        polynom.grad = len(koeffizienten) - 1
        return polynom
    
    def erstelle_intervall_funktion(func, min_x, max_x):
        """Wrapper für Funktionen mit definierten Intervallen."""
        def bounded_func(x):
            if not (min_x <= x <= max_x):
                raise ValueError(f"x={x} außerhalb des Intervalls [{min_x}, {max_x}]")
            return func(x)
        
        bounded_func.intervall = (min_x, max_x)
        bounded_func.original = func
        return bounded_func
    
    return {
        "potenz": erstelle_potenz_funktion,
        "polynom": erstelle_polynom,
        "intervall": erstelle_intervall_funktion
    }

# Mathematische Funktionen testen
print("\n=== Mathematische Closures ===")
math_factory = erstelle_mathematik_funktionen()

# Potenz-Funktionen
quadrat = math_factory["potenz"](2)
kubik = math_factory["potenz"](3)

print(f"5² = {quadrat(5)}")
print(f"3³ = {kubik(3)}")

# Polynom: 2 + 3x + x² (2 + 3*x + 1*x²)
polynom = math_factory["polynom"]([2, 3, 1])
print(f"Polynom bei x=2: {polynom(2)}")  # 2 + 3*2 + 1*4 = 12
print(f"Polynom-Grad: {polynom.grad}")

# Beschränkte Funktion
import math
sinus_eingeschränkt = math_factory["intervall"](math.sin, -math.pi, math.pi)

print(f"sin(π/2) = {sinus_eingeschränkt(math.pi/2):.3f}")

try:
    print(sinus_eingeschränkt(2*math.pi))  # Außerhalb des Intervalls
except ValueError as e:
    print(f"Fehler: {e}")
```
