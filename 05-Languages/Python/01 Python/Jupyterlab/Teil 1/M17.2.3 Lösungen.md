def dokumentiere_rueckgabe_funktionen():
    
    def finde_buch_nach_titel(buecher_liste, gesuchter_titel):
        """
        Sucht ein Buch nach seinem Titel in der Bücherliste.
        
        Returns:
            dict: Gefundenes Buch-Dictionary oder None wenn nicht gefunden
        """
        for buch in buecher_liste:
            if buch["titel"] == gesuchter_titel:
                return buch
        return None
    
    def zaehle_verfuegbare_buecher(buecher_liste):
        """
        Zählt alle verfügbaren Bücher in der Liste.
        
        Returns:
            int: Anzahl der verfügbaren Bücher
        """
        anzahl = 0
        for buch in buecher_liste:
            if buch["status"] == "verfügbar":
                anzahl += 1
        return anzahl
    
    def erstelle_autor_liste(buecher_liste):
        """
        Erstellt eine Liste aller eindeutigen Autoren.
        
        Returns:
            list: Liste mit allen verschiedenen Autorennamen
        """
        autoren = []
        for buch in buecher_liste:
            if buch["autor"] not in autoren:
                autoren.append(buch["autor"])
        return autoren
    
    # Test mit Beispieldaten

# Zelle 2.1: Ausführliche Parameter-Dokumentation
def dokumentiere_parameter_funktionen():
    
    def suche_buecher(buecher_liste, suchfeld, suchbegriff):
        """
        Sucht Bücher nach einem bestimmten Feld und Begriff.
        
        Args:
            buecher_liste (list): Liste aller Bücher als Dictionaries
            suchfeld (str): Das Feld zum Suchen (titel, autor, jahr)
            suchbegriff (str): Der gesuchte Begriff
            
        Returns:
            list: Liste der gefundenen Bücher
        """
        gefundene_buecher = []
        for buch in buecher_liste:
            if suchfeld in buch and str(buch[suchfeld]) == str(suchbegriff):
                gefundene_buecher.append(buch)
        return gefundene_buecher
    
    def berechne_durchschnittsjahr(buecher_liste, nur_verfuegbare):
        """
        Berechnet das durchschnittliche Erscheinungsjahr der Bücher.
        
        Args:
            buecher_liste (list): Liste aller Bücher
            nur_verfuegbare (bool): True = nur verfügbare Bücher berücksichtigen
            
        Returns:
            float: Durchschnittliches Erscheinungsjahr
        """
        relevante_buecher = []
        for buch in buecher_liste:
            if not nur_verfuegbare or buch["status"] == "verfügbar":
                relevante_buecher.append(buch)
        
        if not relevante_buecher:
            return 0
        
        jahre_summe = sum(buch["jahr"] for buch in relevante_buecher)
        return jahre_summe / len(relevante_buecher)
    
    # Test der Funktionen
    test_daten = [
        {"titel": "Altes Buch", "autor": "Autor 1", "jahr": 2010, "status": "verfügbar"},
        {"titel": "Neues Buch", "autor": "Autor 2", "jahr": 2020, "status": "ausgeliehen"},
        {"titel": "Aktuelles Buch", "autor": "Autor 1", "jahr": 2023, "status": "verfügbar"}
    ]
    
    suchergebnis = suche_buecher(test_daten, "autor", "Autor 1")
    durchschnitt_alle = berechne_durchschnittsjahr(test_daten, False)
    durchschnitt_verfuegbar = berechne_durchschnittsjahr(test_daten, True)
    
    print("\n📋 SCHRITT 2.1: Parameter vollständig dokumentiert")
    print("=" * 52)
    print("✅ Args-Abschnitt listet alle Parameter auf")
    print("✅ Datentypen in Klammern angegeben (list, str, bool)")
    print("✅ Parameterbeschreibungen erklären Zweck und Format")
    print("✅ Spezielle Werte dokumentiert (True/False Bedeutung)")
    return f"Suche: {len(suchergebnis)} Treffer, Ø alle: {durchschnitt_alle:.1f}, Ø verfügbar: {durchschnitt_verfuegbar:.1f}"

# Test der Funktion
print(dokumentiere_parameter_funktionen())

def dokumentiere_datentyp_funktionen():
    
    def erstelle_buch_dictionary(titel, autor, jahr, ist_verfuegbar):
        """
        Erstellt ein vollständiges Buch-Dictionary aus Einzelwerten.
        
        Args:
            titel (str): Titel des Buches
            autor (str): Name des Autors  
            jahr (int): Erscheinungsjahr als Zahl
            ist_verfuegbar (bool): True wenn verfügbar, False wenn ausgeliehen
            
        Returns:
            dict: Vollständiges Buch-Dictionary mit allen Feldern

            "titel": titel,
            "autor": autor,
            "jahr": jahr,
            "status": "verfügbar" if ist_verfuegbar else "ausgeliehen"
        }
    
    def filtere_buecher_nach_zeitraum(buecher_liste, von_jahr, bis_jahr):
        """
        Filtert Bücher nach einem bestimmten Erscheinungszeitraum.
        
        Args:
            buecher_liste (list): Liste von Buch-Dictionaries
            von_jahr (int): Frühestes Jahr (inklusive)

            list: Alle Bücher die im angegebenen Zeitraum erschienen sind

        for buch in buecher_liste:
            if von_jahr <= buch["jahr"] <= bis_jahr:
                gefilterte_buecher.append(buch)

# Zelle 3.1: Komplexe Sortier-Logik kommentieren
def kommentiere_sortier_funktion():
    
    def sortiere_buecher_nach_beliebtheit(buecher_liste):
        """Sortiert Bücher nach einem Beliebtheitsscore absteigend."""
        
        def berechne_beliebtheitsscore(buch):
            # Aktualität: Neuere Bücher bekommen mehr Punkte (max 25 für 2025)
            aktualitaets_punkte = max(0, buch["jahr"] - 2000)
            
            # Lesbarkeit: Kürzere Titel sind oft eingängiger (max 50 Punkte)
            titel_faktor = 50 - min(50, len(buch["titel"]))
            
            # Verfügbarkeit: Verfügbare Bücher werden bevorzugt (20 Bonus-Punkte)
            verfuegbarkeits_bonus = 20 if buch["status"] == "verfügbar" else 0
            
            # Gesamtscore aus allen drei Faktoren
            gesamt_score = aktualitaets_punkte + titel_faktor + verfuegbarkeits_bonus
            
            return gesamt_score
        
        # sorted() sortiert die Liste mit key-Funktion für Bewertung
        # reverse=True sorgt für absteigende Reihenfolge (höchster Score zuerst)
        sortierte_buecher = sorted(buecher_liste, key=berechne_beliebtheitsscore, reverse=True)
        
        return sortierte_buecher
    
    # Test der Sortierung
    test_buecher = [
        {"titel": "Sehr langer Buchtitel der schwer zu lesen ist", "jahr": 2020, "status": "verfügbar"},
        {"titel": "Kurz", "jahr": 2023, "status": "verfügbar"},
        {"titel": "Alt", "jahr": 1990, "status": "ausgeliehen"}
    ]
    
    sortiert = sortiere_buecher_nach_beliebtheit(test_buecher)
    erstes_buch = sortiert[0]["titel"]
    
    print("\n🔄 SCHRITT 3.1: Algorithmus-Logik kommentiert")
    print("=" * 48)
    print("✅ Jeder Berechnungsschritt ist erklärt")
    print("✅ Bewertungsfaktoren sind begründet")
    print("✅ Funktionsparameter sind erläutert (key, reverse)")
    print("✅ Komplexe Operationen haben Inline-Kommentare")
    return f"Sortierung: Beliebtestes Buch ist '{erstes_buch}'"

# Test der Funktion
print(kommentiere_sortier_funktion())

# Zelle 3.2: Komplexe Datenanalyse kommentieren
def kommentiere_statistik_funktion():
    
    def erstelle_bibliothek_bericht(buecher_liste):
        """Erstellt einen detaillierten Bericht über die Bibliothek."""
        
        # Zähler-Variablen für verschiedene Statistiken initialisieren
        gesamt_buecher = len(buecher_liste)
        verfuegbare_buecher = 0
        autoren_set = set()  # set() vermeidet automatisch Duplikate
        jahr_summe = 0
        
        # Jedes Buch einmal durchgehen und alle Statistiken sammeln
        # Effizienter als mehrere separate Schleifen
        for buch in buecher_liste:
            # Verfügbare Bücher zählen
            if buch["status"] == "verfügbar":
                verfuegbare_buecher += 1
            
            # Autor zum Set hinzufügen (Duplikate werden automatisch ignoriert)
            autoren_set.add(buch["autor"])
            
            # Jahre für Durchschnittsberechnung sammeln
            jahr_summe += buch["jahr"]
        
        # Durchschnittsjahr berechnen, Division durch Null vermeiden
        durchschnittsjahr = jahr_summe / gesamt_buecher if gesamt_buecher > 0 else 0
        
        # Verfügbarkeit als Prozentsatz berechnen
        verfuegbarkeits_prozent = (verfuegbare_buecher / gesamt_buecher * 100) if gesamt_buecher > 0 else 0
        
        # Alle Ergebnisse in einem Dictionary strukturiert zurückgeben
        bericht = {
            "gesamt": gesamt_buecher,
            "verfuegbar": verfuegbare_buecher,
            "verschiedene_autoren": len(autoren_set),  # Anzahl eindeutiger Autoren
            "durchschnittsjahr": round(durchschnittsjahr),  # Auf ganze Jahre runden
            "verfuegbarkeit_prozent": round(verfuegbarkeits_prozent, 1)  # Eine Nachkommastelle
        }
        
        return bericht
    
    # Test der Berichterstellung
    beispiel_bibliothek = [
        {"titel": "Buch A", "autor": "Autor 1", "jahr": 2020, "status": "verfügbar"},
        {"titel": "Buch B", "autor": "Autor 2", "jahr": 2021, "status": "ausgeliehen"},
        {"titel": "Buch C", "autor": "Autor 1", "jahr": 2022, "status": "verfügbar"}
    ]
    
    bericht = erstelle_bibliothek_bericht(beispiel_bibliothek)
    
    print("\n📊 SCHRITT 3.2: Datenverarbeitung erklärt")
    print("=" * 46)
    print("✅ Variablen-Initialisierung kommentiert")
    print("✅ Schleifen-Logik Schritt für Schritt erklärt")
    print("✅ Berechnungen mit Begründung versehen")
    print("✅ Spezialfälle dokumentiert (Division durch Null)")
    print("✅ Datenstrukturen erklärt (set() für Eindeutigkeit)")
    return f"Bericht: {bericht['gesamt']} Bücher, {bericht['verschiedene_autoren']} Autoren, {bericht['verfuegbarkeit_prozent']}% verfügbar"

# Test der Funktion
print(kommentiere_statistik_funktion())
