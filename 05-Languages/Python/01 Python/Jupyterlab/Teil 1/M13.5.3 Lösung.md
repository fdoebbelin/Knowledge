def bereinige_email(email):
    """Bereinigt und validiert E-Mail-Adressen"""
    # 1. Leerzeichen entfernen
    bereinigte_email = email.strip()
    
    # 2. Zu Kleinbuchstaben
    bereinigte_email = bereinigte_email.lower()
    
    # 3. E-Mail-Pattern für Validierung
    email_pattern = r'^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$'
    ist_gueltig = bool(re.match(email_pattern, bereinigte_email))
    
    return bereinigte_email, ist_gueltig

def bereinige_telefonnummer(telefon):
    """Bereinigt deutsche Telefonnummern"""
    # 1. Alle Nicht-Ziffern entfernen (außer + am Anfang)
    if telefon.startswith('+'):
        # Plus am Anfang beibehalten
        bereinigt = '+' + re.sub(r'[^\d]', '', telefon[1:])
    else:
        bereinigt = re.sub(r'[^\d]', '', telefon)
    
    # 2. Deutsche Vorwahl normalisieren (0 → +49)
    if bereinigt.startswith('0'):
        bereinigt = '+49' + bereinigt[1:]
    elif not bereinigt.startswith('+'):
        # Falls keine Vorwahl, deutsche Vorwahl hinzufügen
        bereinigt = '+49' + bereinigt
    
    # 3. Leerzeichen zur besseren Lesbarkeit einfügen
    if bereinigt.startswith('+49'):
        # Deutsche Nummer: +49 XXX XXXXXXX
        if len(bereinigt) >= 6:
            bereinigt = bereinigt[:3] + ' ' + bereinigt[3:6] + ' ' + bereinigt[6:]
    
    return bereinigt

def bereinige_text_komplett(text, optionen=None):
    """Vollständige Text-Bereinigung mit Optionen"""
    # Standard-Optionen definieren
    if optionen is None:
        optionen = {"leerzeichen": True, "gross_klein": "klein"}
    
    ergebnis = text
    
    # Leerzeichen bereinigen
    if optionen.get("leerzeichen", False):
        # Führende/nachfolgende Leerzeichen entfernen
        ergebnis = ergebnis.strip()
        # Mehrfache Leerzeichen zu einem reduzieren
        ergebnis = re.sub(r'\s+', ' ', ergebnis)
    
    # Groß-/Kleinschreibung
    gross_klein = optionen.get("gross_klein")
    if gross_klein == "klein":
        ergebnis = ergebnis.lower()
    elif gross_klein == "gross":
        ergebnis = ergebnis.upper()
    elif gross_klein == "title":
        ergebnis = ergebnis.title()
    
    # Sonderzeichen entfernen
    if optionen.get("sonderzeichen", False):
        ergebnis = re.sub(r'[^A-Za-z0-9\s]', '', ergebnis)
    
    # Zahlen entfernen
    if optionen.get("zahlen", False):
        ergebnis = re.sub(r'\d', '', ergebnis)
    
    return ergebnis

def parse_key_value_config(config_text):
    """Parst einfache Key=Value Konfigurationen"""
    config = {}
    
    for zeile in config_text.split('\n'):
        # Leerzeichen entfernen und leere Zeilen/Kommentare überspringen
        zeile = zeile.strip()
        if not zeile or zeile.startswith('#'):
            continue
        
        # Key=Value Paare finden
        if '=' in zeile:
            key, value = zeile.split('=', 1)  # Nur beim ersten = trennen
            key = key.strip()
            value = value.strip()
            
            config[key] = value
    
    return config

def parse_section_config(config_text):
    """Parst Konfiguration mit Sektionen [section]"""
    config = {"_global": {}}
    aktuelle_sektion = "_global"
    
    for zeile in config_text.split('\n'):
        zeile = zeile.strip()
        if not zeile or zeile.startswith('#'):
            continue
        
        # Sektion erkannt [section_name]
        if zeile.startswith('[') and zeile.endswith(']'):
            aktuelle_sektion = zeile[1:-1]
            config[aktuelle_sektion] = {}
        
        # Key=Value in aktueller Sektion
        elif '=' in zeile:
            key, value = zeile.split('=', 1)
            key = key.strip()
            value = value.strip()
            
            # Typ-Konvertierung
            if value.lower() == 'true':
                value = True
            elif value.lower() == 'false':
                value = False
            elif value.isdigit():
                value = int(value)
            
            config[aktuelle_sektion][key] = value
        
        # Listen-Format: key: item1, item2, item3
        elif ':' in zeile:
            key, values = zeile.split(':', 1)
            key = key.strip()
            # Komma-getrennte Werte
            items = [item.strip() for item in values.split(',')]
            config[aktuelle_sektion][key] = items
    
    return config

def extrahiere_csv_daten(csv_text, delimiter=','):
    """Extrahiert strukturierte Daten aus CSV-Text"""
    zeilen = csv_text.strip().split('\n')
    if not zeilen:
        return []
    
    # Header-Zeile verarbeiten
    header = [spalte.strip() for spalte in zeilen[0].split(delimiter)]
    
    # Daten-Zeilen verarbeiten
    daten = []
    for zeile in zeilen[1:]:
        if not zeile.strip():
            continue
        
        werte = [wert.strip() for wert in zeile.split(delimiter)]
        
        # Dictionary für diese Zeile erstellen
        datensatz = {}
        for i, wert in enumerate(werte):
            if i < len(header):
                spalte = header[i]
                
                # Typ-Konvertierung versuchen
                if wert.isdigit():
                    wert = int(wert)
                elif wert.replace('.', '').isdigit():
                    wert = float(wert)
                
                datensatz[spalte] = wert
        
        daten.append(datensatz)
    
    return daten

def analysiere_text_statistiken(text):
    """Umfassende Text-Statistiken"""
    # Grundlegende Zählungen
    zeichen_gesamt = len(text)
    zeichen_ohne_leerzeichen = len(text.replace(' ', ''))
    
    # Wörter analysieren
    woerter = text.split()
    anzahl_woerter = len(woerter)
    
    # Sätze zählen (vereinfacht: durch .!? getrennt)
    saetze = re.split(r'[.!?]+', text)
    anzahl_saetze = len([s for s in saetze if s.strip()])
    
    # Absätze zählen
    absaetze = text.split('\n\n')
    anzahl_absaetze = len([a for a in absaetze if a.strip()])
    
    # Wortlängen-Statistiken
    wort_laengen = [len(wort.strip('.,!?')) for wort in woerter]
    
    durchschnittliche_wortlaenge = sum(wort_laengen) / len(wort_laengen) if wort_laengen else 0
    laengstes_wort = max(woerter, key=lambda w: len(w.strip('.,!?'))) if woerter else ""
    kuerzestes_wort = min(woerter, key=lambda w: len(w.strip('.,!?'))) if woerter else ""
    
    # Readability-Score (vereinfacht)
    avg_words_per_sentence = anzahl_woerter / anzahl_saetze if anzahl_saetze > 0 else 0
    readability_score = 206.835 - (1.015 * avg_words_per_sentence) - (84.6 * durchschnittliche_wortlaenge)
    
    return {
        'zeichen_gesamt': zeichen_gesamt,
        'zeichen_ohne_leerzeichen': zeichen_ohne_leerzeichen,
        'anzahl_woerter': anzahl_woerter,
        'anzahl_saetze': anzahl_saetze,
        'anzahl_absaetze': anzahl_absaetze,
        'durchschnittliche_wortlaenge': round(durchschnittliche_wortlaenge, 2),
        'laengstes_wort': laengstes_wort.strip('.,!?'),
        'kuerzestes_wort': kuerzestes_wort.strip('.,!?'),
        'woerter_pro_satz': round(avg_words_per_sentence, 2),
        'readability_score': round(readability_score, 1)
    }

def finde_haeufige_woerter(text, top_n=5, min_laenge=3):
    """Findet die häufigsten Wörter im Text"""
    # Text bereinigen und in Wörter aufteilen
    text_bereinigt = re.sub(r'[^\w\s]', '', text.lower())
    woerter = text_bereinigt.split()
    
    # Kurze Wörter und Stoppwörter filtern
    stoppwoerter = {'der', 'die', 'das', 'und', 'oder', 'aber', 'ist', 'sind', 'ein', 'eine', 'mit', 'von'}
    gefilterte_woerter = [
        wort for wort in woerter 
        if len(wort) >= min_laenge and wort not in stoppwoerter
    ]
    
    # Häufigkeiten zählen
    wort_haeufigkeiten = {}
    for wort in gefilterte_woerter:
        wort_haeufigkeiten[wort] = wort_haeufigkeiten.get(wort, 0) + 1
    
    # Top N Wörter sortieren
    top_woerter = sorted(wort_haeufigkeiten.items(), key=lambda x: x[1], reverse=True)[:top_n]
    
    return top_woerter

def extrahiere_entities(text):
    """Extrahiert potentielle Entitäten (E-Mails, URLs, Zahlen)"""
    entities = {
        'emails': [],
        'urls': [],
        'telefonnummern': [],
        'zahlen': [],
        'grossgeschriebene_woerter': []
    }
    
    # E-Mail-Adressen finden
    email_pattern = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b'
    entities['emails'] = re.findall(email_pattern, text)
    
    # URLs finden (vereinfacht)
    url_pattern = r'https?://[A-Za-z0-9.-]+\.[A-Za-z]{2,}[/\w]*'
    entities['urls'] = re.findall(url_pattern, text)
    
    # Telefonnummern finden (deutsche Muster)
    telefon_pattern = r'(\+49[\s-]?|0)(\d{2,5})[\s-]?(\d{6,8})'
    entities['telefonnummern'] = re.findall(telefon_pattern, text)
    
    # Zahlen finden
    zahlen_pattern = r'\b\d+[.,]?\d*\b'
    entities['zahlen'] = re.findall(zahlen_pattern, text)
    
    # Großgeschriebene Wörter (potentielle Namen)
    gross_pattern = r'\b[A-Z][a-z]+'
    entities['grossgeschriebene_woerter'] = re.findall(gross_pattern, text)
    
    return entities

class TextProcessor:
    """Klasse für verkettbare Text-Verarbeitung"""
    
    def __init__(self, text):
        self.text = text
        self.history = [text]  # Protokolliert alle Änderungen
    
    def bereinigen(self, entferne_sonderzeichen=True, normalisiere_leerzeichen=True):
        """Bereinigt den Text"""
        text = self.text
        
        if normalisiere_leerzeichen:
            text = re.sub(r'\s+', ' ', text.strip())
        
        if entferne_sonderzeichen:
            text = re.sub(r'[^\w\s]', '', text)
        
        self.text = text
        self.history.append(text)
        return self
    
    def normalisieren(self, case='lower'):
        """Normalisiert Groß-/Kleinschreibung"""
        if case == 'lower':
            self.text = self.text.lower()
        elif case == 'upper':
            self.text = self.text.upper()
        elif case == 'title':
            self.text = self.text.title()
        
        self.history.append(self.text)
        return self
    
    def filtern(self, min_wortlaenge=1, entferne_stoppwoerter=False):
        """Filtert Wörter nach Kriterien"""
        woerter = self.text.split()
        
        # Nach Wortlänge filtern
        woerter = [wort for wort in woerter if len(wort) >= min_wortlaenge]
        
        # Stoppwörter entfernen
        if entferne_stoppwoerter:
            stoppwoerter = {'der', 'die', 'das', 'und', 'oder', 'aber', 'ist', 'sind'}
            woerter = [wort for wort in woerter if wort not in stoppwoerter]
        
        self.text = ' '.join(woerter)
        self.history.append(self.text)
        return self
    
    def transformieren(self, transformation):
        """Wendet eine custom Transformation an"""
        if callable(transformation):
            self.text = transformation(self.text)
        
        self.history.append(self.text)
        return self
    
    def ergebnis(self):
        """Gibt das finale Ergebnis zurück"""
        return self.text
    
    def zeige_schritte(self):
        """Zeigt alle Verarbeitungsschritte"""
        for i, schritt in enumerate(self.history):
            print(f"  Schritt {i}: '{schritt}'")

def verarbeite_batch_texte(texte, verarbeitungs_funktion):
    """Verarbeitet mehrere Texte mit derselben Funktion"""
    return [verarbeitungs_funktion(text) for text in texte]

def text_pipeline(*verarbeitungsschritte):
    """Erstellt eine wiederverwendbare Pipeline"""
    def pipeline(text):
        result = text
        for schritt in verarbeitungsschritte:
            result = schritt(result)
        return result
    return pipeline
