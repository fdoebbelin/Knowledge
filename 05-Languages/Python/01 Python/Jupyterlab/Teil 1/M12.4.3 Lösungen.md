## Aufgabe 1: Grundlegende Set-Operationen

```python
# Gegebene Sets
mathe_schueler = {"Anna", "Bob", "Clara", "David", "Eva"}
physik_schueler = {"Bob", "Clara", "Frank", "Greta", "Hans"}

print("=== Schüler-Analyse ===")
print(f"Mathe-Schüler: {mathe_schueler}")
print(f"Physik-Schüler: {physik_schueler}")

# 1. Vereinigung: Alle Schüler die Mathe ODER Physik haben
alle_schueler_op = mathe_schueler | physik_schueler
alle_schueler_method = mathe_schueler.union(physik_schueler)

print(f"\n1. Vereinigung (Mathe ODER Physik):")
print(f"   Mit |: {alle_schueler_op}")
print(f"   Mit union(): {alle_schueler_method}")
print(f"   Anzahl: {len(alle_schueler_op)} Schüler")

# 2. Schnittmenge: Schüler die BEIDE Fächer haben
beide_faecher_op = mathe_schueler & physik_schueler
beide_faecher_method = mathe_schueler.intersection(physik_schueler)

print(f"\n2. Schnittmenge (Mathe UND Physik):")
print(f"   Mit &: {beide_faecher_op}")
print(f"   Mit intersection(): {beide_faecher_method}")
print(f"   Anzahl: {len(beide_faecher_op)} Schüler")

# 3. Differenz: Schüler die nur Mathe haben
nur_mathe_op = mathe_schueler - physik_schueler
nur_mathe_method = mathe_schueler.difference(physik_schueler)

print(f"\n3. Differenz (nur Mathe):")
print(f"   Mit -: {nur_mathe_op}")
print(f"   Mit difference(): {nur_mathe_method}")
print(f"   Anzahl: {len(nur_mathe_op)} Schüler")

# Umgekehrt: Nur Physik
nur_physik = physik_schueler - mathe_schueler
print(f"   Nur Physik: {nur_physik} ({len(nur_physik)} Schüler)")

# 4. Symmetrische Differenz: Schüler die nur EINES der Fächer haben
nur_ein_fach_op = mathe_schueler ^ physik_schueler
nur_ein_fach_method = mathe_schueler.symmetric_difference(physik_schueler)

print(f"\n4. Symmetrische Differenz (nur ein Fach):")
print(f"   Mit ^: {nur_ein_fach_op}")
print(f"   Mit symmetric_difference(): {nur_ein_fach_method}")
print(f"   Anzahl: {len(nur_ein_fach_op)} Schüler")

# Zusätzliche Analyse
print(f"\n=== Zusammenfassung ===")
print(f"Gesamt Schüler: {len(alle_schueler_op)}")
print(f"Beide Fächer: {len(beide_faecher_op)}")
print(f"Nur Mathe: {len(nur_mathe_op)}")
print(f"Nur Physik: {len(nur_physik)}")
print(f"Nur ein Fach: {len(nur_ein_fach_op)}")

# Verifikation: Summe muss stimmen
check = len(beide_faecher_op) + len(nur_ein_fach_op)
print(f"Verifikation: {len(beide_faecher_op)} + {len(nur_ein_fach_op)} = {check} = {len(alle_schueler_op)} ✓")
```

```python
# Gegebene Berechtigungen
admin_rechte = {"lesen", "schreiben", "löschen", "admin", "backup"}

viewer_rechte = {"lesen", "download"}

benutzer = {

    "bob": editor_rechte,
    "charlie": viewer_rechte,
    "diana": admin_rechte
}

print("=== Benutzerrechte-System ===")

# 1. Gemeinsame Rechte zwischen zwei Benutzern
def gemeinsame_rechte(benutzer1, benutzer2):
    """Findet Rechte die beide Benutzer haben"""
    if benutzer1 not in benutzer or benutzer2 not in benutzer:
        return set()
    
    rechte1 = benutzer[benutzer1]
    rechte2 = benutzer[benutzer2]
    return rechte1 & rechte2

# 2. Einzigartige Rechte eines Benutzers
def einzigartige_rechte(benutzer_name):
    """Findet Rechte die nur dieser Benutzer hat"""
    if benutzer_name not in benutzer:
        return set()
    
    user_rechte = benutzer[benutzer_name]
    andere_rechte = set()
    
    # Alle Rechte der anderen Benutzer sammeln
    for name, rechte in benutzer.items():
        if name != benutzer_name:
            andere_rechte |= rechte
    
    return user_rechte - andere_rechte

# 3. Prüfen ob Aktion erlaubt ist
def kann_aktion(benutzer_name, aktion):
    """Prüft ob Benutzer eine Aktion durchführen kann"""
    if benutzer_name not in benutzer:
        return False
    
    return aktion in benutzer[benutzer_name]

# 4. Alle möglichen Aktionen im System
def alle_aktionen():
    """Gibt alle Aktionen zurück die irgendein Benutzer hat"""
    alle_rechte = set()
    for rechte in benutzer.values():
        alle_rechte |= rechte
    return alle_rechte

# 5. Benutzer mit bestimmtem Recht
def benutzer_mit_recht(recht):
    """Findet alle Benutzer die ein bestimmtes Recht haben"""
    return [name for name, rechte in benutzer.items() if recht in rechte]

print("\n2. Einzigartige Rechte:")
for name in benutzer:
    einzigartig = einzigartige_rechte(name)
    print(f"   Nur {name}: {einzigartig}")

# Test 3: Aktions-Tests
print("\n3. Aktions-Tests:")
test_aktionen = ["lesen", "löschen", "backup", "upload", "admin"]
for aktion in test_aktionen:
    print(f"   '{aktion}' können: {benutzer_mit_recht(aktion)}")

# Test 4: Alle Aktionen
print(f"\n4. Alle möglichen Aktionen: {alle_aktionen()}")

print("\n5. Beispiel-Berechtigungsprüfungen:")
tests = [
```

## Aufgabe 4: E-Mail-Domain-Analyse

```python
# Gegebene E-Mail-Sets
kunden_emails = {
    "anna@gmail.com", "bob@yahoo.com", "clara@company.de",

}

mitarbeiter_emails = {
    "frank@company.de", "greta@company.de", "hans@gmail.com",

}

print("=== E-Mail-Domain-Analyse ===")
print(f"Kunden E-Mails: {kunden_emails}")
print(f"Mitarbeiter E-Mails: {mitarbeiter_emails}")

# 1. Domains aus beiden Listen extrahieren
def extrahiere_domains(email_set):
    """Extrahiert Domains aus E-Mail-Adressen"""
    return {email.split("@")[1] for email in email_set}

kunden_domains = extrahiere_domains(kunden_emails)
mitarbeiter_domains = extrahiere_domains(mitarbeiter_emails)

print(f"\n1. Domain-Extraktion:")
print(f"   Kunden-Domains: {kunden_domains}")
print(f"   Mitarbeiter-Domains: {mitarbeiter_domains}")

# 2. Domains die beide Gruppen nutzen
gemeinsame_domains = kunden_domains & mitarbeiter_domains
print(f"\n2. Gemeinsame Domains: {gemeinsame_domains}")

# 3. Domains nur von Kunden genutzt
nur_kunden_domains = kunden_domains - mitarbeiter_domains
print(f"\n3. Nur Kunden-Domains: {nur_kunden_domains}")

# 4. Domains nur von Mitarbeitern genutzt
nur_mitarbeiter_domains = mitarbeiter_domains - kunden_domains
print(f"\n4. Nur Mitarbeiter-Domains: {nur_mitarbeiter_domains}")

# 5. Statistiken über Domain-Verteilung
print(f"\n5. Domain-Statistiken:")

# Alle Domains
alle_domains = kunden_domains | mitarbeiter_domains
print(f"   Gesamt verschiedene Domains: {len(alle_domains)}")

# Für jede Domain: Anzahl Benutzer
def zaehle_benutzer_pro_domain():
    """Zählt Benutzer pro Domain"""
    domain_stats = {}
    
    # Alle E-Mails zusammen
    alle_emails = kunden_emails | mitarbeiter_emails
    
    for email in alle_emails:
        domain = email.split("@")[1]
        if domain not in domain_stats:
            domain_stats[domain] = {"kunden": 0, "mitarbeiter": 0, "gesamt": 0}
        
        # Prüfen ob Kunde oder Mitarbeiter
        if email in kunden_emails:
            domain_stats[domain]["kunden"] += 1
        if email in mitarbeiter_emails:
            domain_stats[domain]["mitarbeiter"] += 1
        
        domain_stats[domain]["gesamt"] += 1
    
    return domain_stats

domain_statistics = zaehle_benutzer_pro_domain()

print(f"\n   Benutzer pro Domain:")
print(f"   {'Domain':<15} {'Kunden':<8} {'Mitarbeiter':<12} {'Gesamt':<8}")
print(f"   {'-'*50}")

for domain in sorted(domain_statistics.keys()):
    stats = domain_statistics[domain]
    print(f"   {domain:<15} {stats['kunden']:<8} {stats['mitarbeiter']:<12} {stats['gesamt']:<8}")

# Beliebteste Domain
beliebteste_domain = max(domain_statistics.keys(), 
                        key=lambda d: domain_statistics[d]['gesamt'])
print(f"\n   Beliebteste Domain: {beliebteste_domain} "
      f"({domain_statistics[beliebteste_domain]['gesamt']} Benutzer)")

# Domain-Typ-Analyse
print(f"\n=== Domain-Typ-Analyse ===")
firmen_domains = {domain for domain in alle_domains if ".de" in domain or "company" in domain}
public_domains = alle_domains - firmen_domains

print(f"Firmen-Domains: {firmen_domains}")
print(f"Öffentliche Domains: {public_domains}")

# Sicherheits-Check: Mitarbeiter mit öffentlichen E-Mails
mitarbeiter_public = {email for email in mitarbeiter_emails 
                     if email.split("@")[1] in public_domains}
if mitarbeiter_public:
    print(f"\n⚠️  Mitarbeiter mit öffentlichen E-Mails: {mitarbeiter_public}")
else:
    print(f"\n✅ Alle Mitarbeiter nutzen Firmen-E-Mails")
```

## Aufgabe 6: Wort-Analyse in Texten

```python
# Gegebene Texte
text1 = "Python ist eine einfache und mächtige Programmiersprache"
text2 = "Java ist eine robuste und weit verbreitete Programmiersprache"  
text3 = "JavaScript ist eine flexible und dynamische Programmiersprache"

texte = {
    "Text 1 (Python)": text1,
    "Text 2 (Java)": text2,
    "Text 3 (JavaScript)": text3
}

print("=== Wort-Analyse in Texten ===")
for name, text in texte.items():
    print(f"{name}: '{text}'")

# Texte in Wort-Sets umwandeln
def text_zu_wort_set(text):
    """Wandelt Text in Set von Wörtern um (lowercase, ohne Satzzeichen)"""
    return set(text.lower().replace(",", "").replace(".", "").split())

wort_sets = {
    "Python": text_zu_wort_set(text1),
    "Java": text_zu_wort_set(text2),
    "JavaScript": text_zu_wort_set(text3)
}

print(f"\n=== Wort-Sets ===")
for name, woerter in wort_sets.items():
    print(f"{name}: {woerter}")

# 1. Wörter die in allen drei Texten vorkommen
gemeinsame_woerter = wort_sets["Python"] & wort_sets["Java"] & wort_sets["JavaScript"]
print(f"\n1. Wörter in allen Texten: {gemeinsame_woerter}")

# 2. Wörter die nur in einem Text vorkommen
nur_python = wort_sets["Python"] - wort_sets["Java"] - wort_sets["JavaScript"]
nur_java = wort_sets["Java"] - wort_sets["Python"] - wort_sets["JavaScript"]
nur_javascript = wort_sets["JavaScript"] - wort_sets["Python"] - wort_sets["Java"]

print(f"\n2. Wörter nur in einem Text:")
print(f"   Nur Python: {nur_python}")
print(f"   Nur Java: {nur_java}")
print(f"   Nur JavaScript: {nur_javascript}")

# 3. Wort-Häufigkeits-Analyse über alle Texte
alle_woerter = wort_sets["Python"] | wort_sets["Java"] | wort_sets["JavaScript"]
print(f"\n3. Wort-Häufigkeitsanalyse:")
print(f"   Verschiedene Wörter gesamt: {len(alle_woerter)}")

# Für jedes Wort prüfen in wie vielen Texten es vorkommt
wort_haeufigkeit = {}
for wort in alle_woerter:
    count = 0
    texte_mit_wort = []
    for name, woerter in wort_sets.items():
        if wort in woerter:
            count += 1
            texte_mit_wort.append(name)
    wort_haeufigkeit[wort] = {"count": count, "texte": texte_mit_wort}

# Nach Häufigkeit sortiert ausgeben
print(f"\n   Wörter nach Häufigkeit:")
for haeufigkeit in [3, 2, 1]:  # Von häufig zu selten
    woerter_mit_haeufigkeit = [wort for wort, info in wort_haeufigkeit.items() 
                              if info["count"] == haeufigkeit]
    if woerter_mit_haeufigkeit:
        print(f"   {haeufigkeit}x: {sorted(woerter_mit_haeufigkeit)}")

# 4. Charakteristische Wörter für jeden Text
print(f"\n4. Charakteristische Wörter:")
for name in wort_sets:
    andere_sets = [s for n, s in wort_sets.items() if n != name]
    andere_woerter = set()
    for s in andere_sets:
        andere_woerter |= s
    
    charakteristisch = wort_sets[name] - andere_woerter
    print(f"   Charakteristisch für {name}: {charakteristisch}")

# 5. Ähnlichkeit zwischen Texten berechnen
print(f"\n5. Text-Ähnlichkeit (Jaccard-Index):")

def jaccard_aehnlichkeit(set1, set2):
    """Berechnet Jaccard-Ähnlichkeit zwischen zwei Sets"""
    intersection = len(set1 & set2)
    union = len(set1 | set2)
    return intersection / union if union > 0 else 0

text_paare = [
    ("Python", "Java"),
    ("Python", "JavaScript"),
    ("Java", "JavaScript")
]

aehnlichkeiten = []
for text1, text2 in text_paare:
    similarity = jaccard_aehnlichkeit(wort_sets[text1], wort_sets[text2])
    aehnlichkeiten.append((similarity, text1, text2))
    
    gemeinsam = len(wort_sets[text1] & wort_sets[text2])
    gesamt = len(wort_sets[text1] | wort_sets[text2])
    
    print(f"   {text1} ↔ {text2}:")
    print(f"     Jaccard-Index: {similarity:.3f}")
    print(f"     Gemeinsame Wörter: {gemeinsam}/{gesamt}")
    print(f"     Gemeinsame Wörter: {wort_sets[text1] & wort_sets[text2]}")

# Ähnlichste Texte
aehnlichste = max(aehnlichkeiten, key=lambda x: x[0])
print(f"\n   Ähnlichste Texte: {aehnlichste[1]} & {aehnlichste[2]} "
      f"({aehnlichste[0]:.3f})")

# Zusätzliche Analyse: Wortlängen
print(f"\n=== Zusätzliche Wort-Analyse ===")
for name, woerter in wort_sets.items():
    wortlaengen = [len(wort) for wort in woerter]
    avg_laenge = sum(wortlaengen) / len(wortlaengen) if wortlaengen else 0
    laengstes_wort = max(woerter, key=len) if woerter else ""
    
    print(f"{name}:")
    print(f"   Anzahl Wörter: {len(woerter)}")
    print(f"   Durchschnittliche Wortlänge: {avg_laenge:.1f}")
    print(f"   Längstes Wort: '{laengstes_wort}' ({len(laengstes_wort)} Zeichen)")
```
