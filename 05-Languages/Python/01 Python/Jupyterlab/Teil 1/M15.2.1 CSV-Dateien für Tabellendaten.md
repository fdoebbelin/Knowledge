## 1. Was sind CSV-Dateien?
CSV = "Comma Separated Values" - Tabellendaten in Textform

```python
def erklaere_csv_format():
    """Erklärt das CSV-Format anhand eines einfachen Beispiels"""
    csv_beispiel = """Name,Alter,Stadt
Anna,25,Berlin
Bob,30,München
Charlie,28,Hamburg"""
    
    print("📊 CSV-FORMAT VERSTEHEN")
    print("=" * 30)
    print("CSV = Comma Separated Values (Komma-getrennte Werte)")
    print("Erste Zeile = Header (Spaltenüberschriften)")
    print("Weitere Zeilen = Daten")
    print("\nBeispiel einer CSV-Datei:")
    print(csv_beispiel)
    print("\n✅ Jede Zeile = ein Datensatz")
    print("✅ Kommas trennen die Spalten")
    print("✅ Wie eine Excel-Tabelle, aber als Text")
    return csv_beispiel

print(erklaere_csv_format())
```
## 2. Erste CSV-Datei erstellen

```python
def erstelle_erste_csv():
    """Erstellt eine einfache CSV-Datei mit csv.writer()"""
    import csv
    
    # Daten als Liste von Listen vorbereiten
    tabellendaten = [
        ["Name", "Alter", "Stadt"],      # Header-Zeile
        ["Anna", "25", "Berlin"],        # Datenzeile 1
        ["Bob", "30", "München"],        # Datenzeile 2
        ["Charlie", "28", "Hamburg"]     # Datenzeile 3
    ]
    
    # CSV-Datei schreiben
    with open("personen.csv", "w", newline="", encoding="utf-8") as datei:
        writer = csv.writer(datei)
        for zeile in tabellendaten:
            writer.writerow(zeile)
    
    print("📝 ERSTE CSV-DATEI ERSTELLT")
    print("=" * 35)
    print("✅ Datei 'personen.csv' wurde erstellt")
    print("✅ csv.writer() verwendet")
    print("✅ newline='' verhindert leere Zeilen")
    print(f"✅ {len(tabellendaten)} Zeilen geschrieben")
    return f"CSV-Datei mit {len(tabellendaten)} Zeilen erstellt"

print(erstelle_erste_csv())
```

## 3. CSV-Dateien lesen

```python
def lese_csv_komplett():
    """Liest eine CSV-Datei komplett mit csv.reader()"""
    import csv
    
    alle_zeilen = []
    with open("personen.csv", "r", encoding="utf-8") as datei:
        reader = csv.reader(datei)
        for zeile in reader:
            alle_zeilen.append(zeile)
    
    print("\n📖 CSV-DATEI LESEN")
    print("=" * 23)
    print("✅ csv.reader() verwendet")
    print(f"✅ {len(alle_zeilen)} Zeilen gelesen")
    print("\nGelesene Daten:")
    for i, zeile in enumerate(alle_zeilen):
        print(f"  Zeile {i}: {zeile}")
    
    return f"CSV gelesen: {len(alle_zeilen)} Zeilen"

print(lese_csv_komplett())
```

```python
def lese_csv_mit_header():
    """Liest CSV und trennt Header von Daten"""
    import csv
    
    with open("personen.csv", "r", encoding="utf-8") as datei:
        reader = csv.reader(datei)
        
        # Erste Zeile als Header lesen
        header = next(reader)
        
        # Restliche Zeilen als Daten lesen
        daten = []
        for zeile in reader:
            daten.append(zeile)
    
    print("\n📖 CSV MIT HEADER-TRENNUNG")
    print("=" * 33)
    print(f"Header: {header}")
    print(f"Datenzeilen: {len(daten)}")
    print("\nDaten-Zeilen:")
    for i, zeile in enumerate(daten, 1):
        print(f"  Person {i}: {zeile}")
    
    return f"Header: {header}, Daten: {len(daten)} Zeilen"

print(lese_csv_mit_header())
```

## 4. Mit DictReader arbeiten (benannte Spalten)

```python
def lese_csv_als_dict():
    """Liest CSV als Dictionary mit Spaltennamen als Keys"""
    import csv
    
    personen_liste = []
    with open("personen.csv", "r", encoding="utf-8") as datei:
        dict_reader = csv.DictReader(datei)
        
        # Jede Zeile wird als Dictionary gelesen
        for zeile in dict_reader:
            personen_liste.append(zeile)
    
    print("\n📖 CSV ALS DICTIONARY LESEN")
    print("=" * 35)
    print("✅ csv.DictReader() verwendet")
    print("✅ Header wird automatisch als Keys verwendet")
    print(f"✅ {len(personen_liste)} Personen gelesen")
    print("\nDictionary-Daten:")
    for i, person in enumerate(personen_liste, 1):
        print(f"  Person {i}: {person}")
    
    return f"DictReader: {len(personen_liste)} Personen als Dictionaries"

print(lese_csv_als_dict())
```

## 5. Mit DictWriter schreiben

```python
def schreibe_csv_als_dict():
    """Schreibt CSV mit DictWriter und benannten Spalten"""
    import csv
    
    # Daten als Liste von Dictionaries
    schulklasse = [
        {"Name": "Lisa", "Fach": "Mathematik", "Note": "1.3"},
        {"Name": "Tom", "Fach": "Physik", "Note": "2.1"},
        {"Name": "Sarah", "Fach": "Chemie", "Note": "1.8"},
        {"Name": "Max", "Fach": "Informatik", "Note": "1.0"}
    ]
    
    # CSV mit DictWriter schreiben
    with open("noten.csv", "w", newline="", encoding="utf-8") as datei:
        fieldnames = ["Name", "Fach", "Note"]
        dict_writer = csv.DictWriter(datei, fieldnames=fieldnames)
        
        # Header schreiben
        dict_writer.writeheader()
        
        # Datenzeilen schreiben
        for schueler in schulklasse:
            dict_writer.writerow(schueler)
    
    print("\n📝 CSV MIT DICTWRITER SCHREIBEN")
    print("=" * 40)
    print("✅ csv.DictWriter() verwendet")
    print("✅ fieldnames definiert: ['Name', 'Fach', 'Note']")
    print("✅ writeheader() für automatischen Header")
    print(f"✅ {len(schulklasse)} Schüler-Datensätze geschrieben")
    return f"DictWriter: {len(schulklasse)} Datensätze geschrieben"

print(schreibe_csv_als_dict())
```
## 6. Praktisches Beispiel: Verkaufsdaten

```python
def erstelle_verkaufsdaten():
    """Erstellt eine CSV-Datei mit Verkaufsdaten"""
    import csv
    from datetime import datetime
    
    verkaufs_daten = [
        {"Datum": "2024-01-15", "Produkt": "Laptop", "Anzahl": "2", "Preis": "1799.98"},
        {"Datum": "2024-01-16", "Produkt": "Maus", "Anzahl": "5", "Preis": "127.50"},
        {"Datum": "2024-01-17", "Produkt": "Tastatur", "Anzahl": "3", "Preis": "239.70"},
        {"Datum": "2024-01-18", "Produkt": "Monitor", "Anzahl": "1", "Preis": "299.99"}
    ]
    
    with open("verkaufe.csv", "w", newline="", encoding="utf-8") as datei:
        fieldnames = ["Datum", "Produkt", "Anzahl", "Preis"]
        writer = csv.DictWriter(datei, fieldnames=fieldnames)
        
        writer.writeheader()
        writer.writerows(verkaufs_daten)  # Alle auf einmal schreiben
    
    print("\n📊 VERKAUFSDATEN-CSV ERSTELLT")
    print("=" * 35)
    print("✅ Realistische Geschäftsdaten")
    print("✅ writerows() für alle Daten auf einmal")
    print(f"✅ {len(verkaufs_daten)} Verkäufe gespeichert")
    return f"Verkaufsdaten: {len(verkaufs_daten)} Einträge"

print(erstelle_verkaufsdaten())
```

```python
def analysiere_verkaufsdaten():
    """Liest und analysiert die Verkaufsdaten"""
    import csv
    
    verkaufe = []
    with open("verkaufe.csv", "r", encoding="utf-8") as datei:
        reader = csv.DictReader(datei)
        for verkauf in reader:
            verkaufe.append(verkauf)
    
    # Einfache Analyse
    gesamtumsatz = sum(float(verkauf["Preis"]) for verkauf in verkaufe)
    produktanzahl = len(set(verkauf["Produkt"] for verkauf in verkaufe))
    
    print("\n📈 VERKAUFSDATEN-ANALYSE")
    print("=" * 30)
    print(f"✅ {len(verkaufe)} Verkäufe analysiert")
    print(f"✅ Gesamtumsatz: {gesamtumsatz:.2f}€")
    print(f"✅ {produktanzahl} verschiedene Produkte")
    print("\nTop-Verkäufe:")
    for verkauf in verkaufe:
        print(f"  {verkauf['Produkt']}: {verkauf['Preis']}€")
    
    return f"Analyse: {gesamtumsatz:.2f}€ Umsatz mit {produktanzahl} Produkten"

print(analysiere_verkaufsdaten())
```

## 7. CSV-Fehlerbehandlung

```python
def sichere_csv_funktion():
    """Liest CSV-Datei mit Fehlerbehandlung"""
    import csv
    
    try:
        with open("noten.csv", "r", encoding="utf-8") as datei:
            reader = csv.DictReader(datei)
            noten_liste = list(reader)
        
        ergebnis = f"✅ {len(noten_liste)} Noten erfolgreich gelesen"
        
    except FileNotFoundError:
        ergebnis = "❌ CSV-Datei nicht gefunden!"
    except csv.Error as csv_fehler:
        ergebnis = f"❌ CSV-Fehler: {csv_fehler}"
    except Exception as fehler:
        ergebnis = f"❌ Allgemeiner Fehler: {fehler}"
    
    print("\n🛡️ SICHERE CSV-VERARBEITUNG")
    print("=" * 35)
    print("✅ FileNotFoundError abgefangen")
    print("✅ csv.Error für CSV-spezifische Probleme")
    print("✅ Exception als allgemeiner Fallback")
    return ergebnis

print(sichere_csv_funktion())
```

```python
def wann_csv_verwenden():
    """Erklärt wann CSV-Dateien sinnvoll sind"""
    anwendungen = {
        "✅ Gut für CSV": [
            "Tabellendaten (wie Excel)",
            "Datenexport/-import zwischen Programmen",
            "Einfache Listen und Kataloge",
            "Kleine bis mittelgroße Datenmengen",
            "Daten für Tabellenkalkulationen"
        ],
        "❌ Nicht gut für CSV": [
            "Verschachtelte Datenstrukturen",
            "Komplexe Beziehungen zwischen Daten",
            "Sehr große Datenmengen (> 1GB)",
            "Daten mit Kommas im Text",
            "Binary-Daten (Bilder, Videos)"
        ]
    }
    
    print("\n🤔 WANN CSV-DATEIEN VERWENDEN?")
    print("=" * 40)
    
    for kategorie, liste in anwendungen.items():
        print(f"\n{kategorie}:")
        for punkt in liste:
            print(f"  • {punkt}")
    
    return "CSV-Anwendungsfälle erklärt"

print(wann_csv_verwenden())
```

## Zusammenfassung

### Wichtige Erkenntnisse:

1. **CSV = Tabellendaten in Textform** - wie Excel, aber einfacher
2. **csv-Modul verwenden** - nicht manuell mit Kommas arbeiten
3. **newline=""** - verhindert leere Zeilen in CSV-Dateien
4. **DictReader/DictWriter** - bequemer Zugriff über Spaltennamen
5. **Fehlerbehandlung** - csv.Error für CSV-spezifische Probleme
6. **encoding="utf-8"** - für deutsche Umlaute in CSV-Dateien

### Nächste Schritte:
- JSON-Dateien für verschachtelte Daten
- SQLite-Datenbanken für komplexe Abfragen
- pandas-Library für große CSV-Dateien
