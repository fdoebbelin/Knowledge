def hole_json_daten():
    """Ruft JSON-Daten von einer API ab und verarbeitet sie"""
    
    api_url = "https://httpbin.org/json"
    
    try:
        import requests
        response = requests.get(api_url)
        
        if response.status_code == 200:
            data = response.json()
            daten_typ = type(data).__name__
            erster_schluessel = list(data.keys())[0] if data else "Keine Schlüssel"
            ergebnis = f"JSON-Typ: {daten_typ}, erster Schlüssel: {erster_schluessel}"
        else:
            ergebnis = f"Fehler: Status {response.status_code}"
            
    except ImportError:
        ergebnis = "requests nicht verfügbar"
    except Exception as fehler:
        ergebnis = f"Fehler: {fehler}"
    
    print("\n📄 SCHRITT 2.2: JSON-Daten verarbeiten")
    print("=" * 45)
    print("✅ response.json() für automatisches Parsing")
    print("✅ Datentyp und Schlüssel ermittelt")
    print(f"✅ Ergebnis: {ergebnis}")
    return ergebnis

# Zelle 3.3: Funktion für beliebige Farben
def farbiger_text(text, farbe):
    """Färbt einen Text in der gewünschten Farbe ein"""
    
    try:
        from colorama import Fore, Style, init
        init()
        
        farb_mapping = {
            "rot": Fore.RED,
            "grün": Fore.GREEN,  
            "blau": Fore.BLUE,
            "gelb": Fore.YELLOW
        }
        
        if farbe.lower() in farb_mapping:
            farb_code = farb_mapping[farbe.lower()]
            ergebnis = f"{farb_code}{text}{Style.RESET_ALL}"
        else:
            ergebnis = text  # Unbekannte Farbe = normaler Text
            
    except ImportError:
        ergebnis = text  # Fallback ohne Farbe
    
    return ergebnis

# Tests der Funktion
print("\n🎯 SCHRITT 3.3: Dynamische Farb-Funktion")
print("=" * 48)
print("✅ Farb-Mapping Dictionary implementiert")
print("✅ Fallback für unbekannte Farben")
print(f"✅ Test rot: {farbiger_text('Test rot', 'rot')}")
print(f"✅ Test grün: {farbiger_text('Test grün', 'grün')}")
print(f"✅ Test unbekannt: {farbiger_text('Test unbekannt', 'lila')}")

# Zelle 4.2: Mehrere URLs mit farbigem Status
def website_monitor():
    """Überwacht mehrere Websites und zeigt Status farbig an"""
    
    websites = {
        "Google": "https://www.google.com",
        "GitHub": "https://github.com", 
        "Test-API": "https://httpbin.org/status/200"
    }
    
    ergebnisse = []
    
    try:
        import requests
        from colorama import Fore, Style, init
        init()
        
        for name, url in websites.items():
            try:
                response = requests.get(url, timeout=5)
                if response.status_code == 200:
                    status = f"{Fore.GREEN}✅ OK{Style.RESET_ALL}"
                else:
                    status = f"{Fore.YELLOW}⚠️ {response.status_code}{Style.RESET_ALL}"
            except:
                status = f"{Fore.RED}❌ Fehler{Style.RESET_ALL}"
            
            ergebnisse.append(f"{name}: {status}")
            
    except ImportError:
        for name in websites.keys():
            ergebnisse.append(f"{name}: ❌ Bibliotheken fehlen")
    
    print("\n📊 SCHRITT 4.2: Website-Monitor")
    print("=" * 37)
    print(f"✅ {len(websites)} Websites überwacht")
    print("✅ Farbkodierte Status-Anzeige")
    for ergebnis in ergebnisse:
        print(f"   {ergebnis}")
    
    return f"Monitor: {len(ergebnisse)} Websites geprüft"

# Test der Funktion
print(website_monitor())
