## Aufgabe 1: Daten-Extraktor

```python
# Zelle 1: Daten aus Text extrahieren
kontakt_text = """
Firma: TechCorp GmbH
Ansprechpartner: Max Mustermann
E-Mail: max.mustermann@techcorp.de
Telefon: +49 30 12345678
Handy: 0171 9876543
Website: https://www.techcorp.de
Adresse: Hauptstraße 123, 10115 Berlin
Fax: 030-98765432
Zweit-E-Mail: info@techcorp.de
"""

import re

print("DATEN-EXTRAKTOR")
print("=" * 30)
print("Original Text:")
print(kontakt_text)

# 1. Alle E-Mail-Adressen finden
email_muster = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b'
emails = re.findall(email_muster, kontakt_text)
print("📧 E-Mail-Adressen:", emails)

# 2. Alle Telefonnummern (verschiedene Formate)
telefon_muster = r'(?:\+49|0)\s?\d{2,4}\s?[\d\s-]{6,10}'
telefonnummern = re.findall(telefon_muster, kontakt_text)
print("📞 Telefonnummern:", telefonnummern)

# 3. Website-URL extrahieren
url_muster = r'https?://[^\s]+'
websites = re.findall(url_muster, kontakt_text)
print("🌐 Website:", websites)

# 4. Postleitzahl finden
plz_muster = r'\b\d{5}\b'
postleitzahlen = re.findall(plz_muster, kontakt_text)
print("📍 Postleitzahl:", postleitzahlen)

# 5. Alle Zahlen einzeln
zahlen_muster = r'\d+'
alle_zahlen = re.findall(zahlen_muster, kontakt_text)
print("🔢 Alle Zahlen:", alle_zahlen)

# Zusätzliche Analyse
print("\n📊 Statistiken:")
print(f"Anzahl E-Mails: {len(emails)}")
print(f"Anzahl Telefonnummern: {len(telefonnummern)}")
print(f"Anzahl Websites: {len(websites)}")
print(f"Anzahl Zahlen insgesamt: {len(alle_zahlen)}")
```

**Erklärung:**
- `\b` sorgt für Wortgrenzen (verhindert Teilmatches)
- `(?:\+49|0)` ist eine Gruppe die +49 oder 0 matcht (?: macht sie zu einer Non-Capturing Group)
- `[^\s]+` bedeutet "alle Zeichen außer Whitespace" - gut für URLs
- `{2,}` bedeutet "mindestens 2 Zeichen" - wichtig bei Domains
## Aufgabe 2: Passwort-Validator

```python
# Zelle 2: Passwort-Validation
passwoerter = [
    "123456",                    # Zu einfach
    "passwort",                  # Nur Buchstaben
    "Password123",               # Gut, aber kein Sonderzeichen
    "P@ssw0rd123",              # Perfekt
    "abc",                      # Zu kurz
    "VeryLongPasswordButNoSpecialChar123",  # Lang, aber kein Sonderzeichen
    "Short1!",                  # Kurz, aber alle Kriterien
    "MySecureP@ssw0rd2024!"     # Sehr gut
]

def validiere_passwort(passwort):
    """Validiert Passwort nach strengen Regeln"""
    
    regeln = {
        "Mindestens 8 Zeichen": len(passwort) >= 8,
        "Mindestens 1 Großbuchstabe": bool(re.search(r'[A-Z]', passwort)),
        "Mindestens 1 Kleinbuchstabe": bool(re.search(r'[a-z]', passwort)),
        "Mindestens 1 Ziffer": bool(re.search(r'\d', passwort)),
        "Mindestens 1 Sonderzeichen": bool(re.search(r'[!@#$%^&*]', passwort))
    }
    
    erfuellt = sum(regeln.values())
    ist_sicher = all(regeln.values())
    
    return regeln, erfuellt, ist_sicher

print("PASSWORT-VALIDATOR")
print("=" * 40)

# Alle Passwörter testen
for passwort in passwoerter:
    regeln, erfuellt, ist_sicher = validiere_passwort(passwort)
    
    status = "🔒 SICHER" if ist_sicher else "🔓 UNSICHER"
    print(f"\n{status} '{passwort}' ({erfuellt}/5 Regeln erfüllt)")
    
    # Details zeigen für unsichere Passwörter
    if not ist_sicher:
        for regel, erfuellt in regeln.items():
            symbol = "✅" if erfuellt else "❌"
            print(f"  {symbol} {regel}")

# Statistik
sichere_passwoerter = 0
for passwort in passwoerter:
    _, _, ist_sicher = validiere_passwort(passwort)
    if ist_sicher:
        sichere_passwoerter += 1

print(f"\n📊 Gesamt: {sichere_passwoerter}/{len(passwoerter)} Passwörter sind sicher")

# Passwort-Stärke-Bewertung
def bewerte_staerke(passwort):
    """Bewertet Passwort-Stärke von 0-100"""
    _, erfuellt, _ = validiere_passwort(passwort)
    
    # Bonus für Länge
    laenge_bonus = min(len(passwort) - 8, 12) * 2  # Max 24 Punkte
    basis_punkte = erfuellt * 15  # Max 75 Punkte
    
    staerke = min(basis_punkte + laenge_bonus, 100)
    return staerke

print(f"\n💪 Passwort-Stärke-Bewertung:")
for passwort in passwoerter:
    staerke = bewerte_staerke(passwort)
    if staerke >= 80:
        level = "Sehr stark"
        emoji = "🟢"
    elif staerke >= 60:
        level = "Stark"
        emoji = "🟡"
    elif staerke >= 40:
        level = "Mittel"
        emoji = "🟠"
    else:
        level = "Schwach"
        emoji = "🔴"
    
    print(f"{emoji} '{passwort[:15]:<15}': {staerke:3d}% - {level}")
```

**Erklärung:**
- `bool(re.search(...))` wandelt Match-Objekt in True/False um
- `all(regeln.values())` prüft ob alle Regeln erfüllt sind
- `sum(regeln.values())` zählt erfüllte Regeln (True = 1, False = 0)
- Zusätzliche Bewertungslogik zeigt Passwort-Stärke in Prozent
## Aufgabe 3: Log-File Analyzer

```python
# Zelle 3: Log-Datei analysieren
log_eintraege = [
    "2024-01-15 10:30:25 [INFO] User login successful: user123",
    "2024-01-15 10:31:42 [ERROR] Database connection failed: timeout",
    "2024-01-15 10:32:15 [WARNING] Low disk space: 85% used",
    "2024-01-15 10:33:01 [INFO] File uploaded: document.pdf (2.5MB)",
    "2024-01-15 10:34:18 [ERROR] Authentication failed: invalid_password",
    "2024-01-15 10:35:33 [INFO] User logout: user123",
    "2024-01-15 10:36:45 [DEBUG] Cache cleared: 150 items removed"
]

def analysiere_log_eintrag(eintrag):
    """Analysiert einen einzelnen Log-Eintrag"""
    
    # RegEx-Muster für die verschiedenen Teile
    datum_zeit_muster = r'(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2})'
    level_muster = r'\[(INFO|ERROR|WARNING|DEBUG)\]'
    nachricht_muster = r'\] (.+)$'
    
    # Daten extrahieren
    datum_zeit_match = re.search(datum_zeit_muster, eintrag)
    level_match = re.search(level_muster, eintrag)
    nachricht_match = re.search(nachricht_muster, eintrag)
    
    return {
        'datum_zeit': datum_zeit_match.group(1) if datum_zeit_match else None,
        'level': level_match.group(1) if level_match else None,
        'nachricht': nachricht_match.group(1) if nachricht_match else None,
        'original': eintrag
    }

print("LOG-FILE ANALYZER")
print("=" * 30)

# Alle Log-Einträge analysieren
analysierte_logs = []
log_level_counts = {}
alle_nachrichten = []

for eintrag in log_eintraege:
    analyse = analysiere_log_eintrag(eintrag)
    analysierte_logs.append(analyse)
    
    # Statistiken sammeln
    level = analyse['level']
    if level:
        log_level_counts[level] = log_level_counts.get(level, 0) + 1
    
    if analyse['nachricht']:
        alle_nachrichten.append(analyse['nachricht'])

# Ergebnisse anzeigen
print("📋 Extrahierte Daten:")
for i, log in enumerate(analysierte_logs, 1):
    print(f"\nEintrag {i}:")
    print(f"  📅 Datum/Zeit: {log['datum_zeit']}")
    print(f"  🏷️  Level: {log['level']}")
    print(f"  💬 Nachricht: {log['nachricht']}")

# Statistiken
print(f"\n📊 LOG-LEVEL STATISTIKEN:")
for level, anzahl in sorted(log_level_counts.items()):
    prozent = (anzahl / len(log_eintraege)) * 100
    print(f"  {level:8}: {anzahl:2d} ({prozent:4.1f}%)")

# Zeitspanne ermitteln
zeiten = [log['datum_zeit'] for log in analysierte_logs if log['datum_zeit']]
if zeiten:
    erste_zeit = min(zeiten)
    letzte_zeit = max(zeiten)
    print(f"\n⏰ Zeitspanne: {erste_zeit} bis {letzte_zeit}")

# Häufigste Wörter in Nachrichten
alle_woerter = []
for nachricht in alle_nachrichten:
    # Einfache Wort-Extraktion (nur Buchstaben)
    woerter = re.findall(r'\b[A-Za-z]+\b', nachricht.lower())
    alle_woerter.extend(woerter)

# Häufigkeits-Analyse
wort_counts = {}
for wort in alle_woerter:
    if len(wort) > 3:  # Nur Wörter mit mehr als 3 Buchstaben
        wort_counts[wort] = wort_counts.get(wort, 0) + 1

# Top 5 häufigste Wörter
haeufigste_woerter = sorted(wort_counts.items(), key=lambda x: x[1], reverse=True)[:5]

print(f"\n🔤 Häufigste Wörter in Nachrichten:")
for wort, anzahl in haeufigste_woerter:
    print(f"  '{wort}': {anzahl}x")

# Fehler-Analyse
fehler_nachrichten = [log['nachricht'] for log in analysierte_logs if log['level'] == 'ERROR']
print(f"\n🚨 Fehler-Details ({len(fehler_nachrichten)} Fehler):")
for i, fehler in enumerate(fehler_nachrichten, 1):
    print(f"  {i}. {fehler}")
```

**Erklärung:**
- `(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2})` - Klammern erstellen eine Capture Group
- `group(1)` holt den Inhalt der ersten Capture Group
- `(INFO|ERROR|WARNING|DEBUG)` - Alternation mit | für mehrere Möglichkeiten
- `.get(key, 0)` für sichere Dictionary-Zugriffe mit Standardwert
## Aufgabe 4: Text-Cleaner

```python
# Zelle 4: Text bereinigen
unordentlicher_text = """
Hallo!!! Das ist ein Text mit vielen    Leerzeichen.
E-Mail: spam@werbung.com oder marketing@ads.net

print("✅ Premium-Telefonnummern entfernt")

# 3. HTTP-URLs entfernen (nur http, nicht https)
http_muster = r'http://[^\s]+'
gereinigter_text = re.sub(http_muster, '[UNSICHERE-URL ENTFERNT]', gereinigter_text)
print("✅ Unsichere HTTP-URLs entfernt")

# 4. Mehrfache Leerzeichen reduzieren
mehrfache_leerzeichen_muster = r'\s{2,}'
gereinigter_text = re.sub(mehrfache_leerzeichen_muster, ' ', gereinigter_text)
print("✅ Mehrfache Leerzeichen reduziert")

# 5. Mehrfache Ausrufezeichen reduzieren
mehrfache_ausrufe_muster = r'!{3,}'
gereinigter_text = re.sub(mehrfache_ausrufe_muster, '!!', gereinigter_text)
print("✅ Excessive Ausrufezeichen reduziert")

print("\nNACHHER:")
print("-" * 40)
print(gereinigter_text)

# Vergleichsstatistiken
print("\n📊 BEREINIGUNGS-STATISTIKEN:")

# Was wurde entfernt?
spam_emails = re.findall(spam_email_muster, unordentlicher_text)
premium_nummern = re.findall(premium_telefon_muster, unordentlicher_text)
http_urls = re.findall(http_muster, unordentlicher_text)

print(f"Entfernte Spam-E-Mails: {len(spam_emails)}")
if spam_emails:
    for email in spam_emails:
        print(f"  - {email}")

print(f"Entfernte Premium-Nummern: {len(premium_nummern)}")
if premium_nummern:
    for nummer in premium_nummern:
        print(f"  - {nummer}")

print(f"Entfernte HTTP-URLs: {len(http_urls)}")
if http_urls:
    for url in http_urls:
        print(f"  - {url}")

# Längenvergleich
original_laenge = len(unordentlicher_text)
bereinigte_laenge = len(gereinigter_text)
ersparnis = original_laenge - bereinigte_laenge

print(f"\nText-Länge vorher: {original_laenge} Zeichen")
print(f"Text-Länge nachher: {bereinigte_laenge} Zeichen")
print(f"Einsparung: {ersparnis} Zeichen ({ersparnis/original_laenge*100:.1f}%)")

# Verbliebene gute Inhalte prüfen
print(f"\n✅ VERBLIEBENE GUTE INHALTE:")
gute_emails = re.findall(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b', gereinigter_text)
https_urls = re.findall(r'https://[^\s]+', gereinigter_text)
normale_telefone = re.findall(r'030-\d+', gereinigter_text)

print(f"Seriöse E-Mails: {gute_emails}")
print(f"Sichere HTTPS-URLs: {https_urls}")
print(f"Normale Telefonnummern: {normale_telefone}")
```

**Erklärung:**
- `(?:spam|marketing|ads)` - Non-Capturing Group für Alternation
- `[A-Za-z0-9._%+-]*` - Beliebige Anzahl von E-Mail-Zeichen vor/nach den Spam-Wörtern
- `\s{2,}` - Mindestens 2 aufeinanderfolgende Whitespace-Zeichen
- `re.sub()` ersetzt alle Matches durch den angegebenen Text
- Schrittweise Bereinigung zeigt jeden Schritt einzeln

```python
# Bonus: Smart Data Extractor für Rechnungen
rechnung_text = """
RECHNUNG Nr. R-2024-001523

Rechnungsdatum: 15.01.2024
Fälligkeitsdatum: 29.01.2024

Kunde: Max Mustermann GmbH
Adresse: Hauptstraße 123, 10115 Berlin
USt-IdNr.: DE123456789

Artikel:
- Python Kurs (40h à 85,00 €)     3.400,00 €
- Materialien                        150,00 €
- Fahrtkosten                         75,50 €

Nettobetrag:                      3.625,50 €
MwSt (19%):                         688,85 €
Gesamtbetrag:                     4.314,35 €

Bankverbindung: IBAN DE89 1234 5678 9012 3456 78
"""

def extrahiere_rechnungsdaten(text):
    """Extrahiert strukturierte Daten aus einer Rechnung"""
    
    daten = {}
    
    # Rechnungsnummer
    rechnung_nr_match = re.search(r'RECHNUNG Nr\.\s+([A-Z]-\d{4}-\d+)', text)
    daten['rechnungsnummer'] = rechnung_nr_match.group(1) if rechnung_nr_match else None
    
    # Daten
    rech_datum_match = re.search(r'Rechnungsdatum:\s+(\d{2}\.\d{2}\.\d{4})', text)
    faellig_datum_match = re.search(r'Fälligkeitsdatum:\s+(\d{2}\.\d{2}\.\d{4})', text)
    
    daten['rechnungsdatum'] = rech_datum_match.group(1) if rech_datum_match else None
    daten['faelligkeitsdatum'] = faellig_datum_match.group(1) if faellig_datum_match else None
    
    # Kunde und Adresse
    kunde_match = re.search(r'Kunde:\s+(.+)', text)
    adresse_match = re.search(r'Adresse:\s+(.+)', text)
    ust_match = re.search(r'USt-IdNr\.:\s+([A-Z]{2}\d+)', text)
    
    daten['kunde'] = kunde_match.group(1) if kunde_match else None
    daten['adresse'] = adresse_match.group(1) if adresse_match else None
    daten['ust_idnr'] = ust_match.group(1) if ust_match else None
    
    # Artikel extrahieren
    artikel_muster = r'-\s+([^(]+(?:\([^)]+\))?)\s+([\d.,]+\s+€)'
    artikel = re.findall(artikel_muster, text)
    daten['artikel'] = [(name.strip(), preis.strip()) for name, preis in artikel]
    
    # Geldbeträge
    netto_match = re.search(r'Nettobetrag:\s+([\d.,]+\s+€)', text)
    mwst_match = re.search(r'MwSt.*?:\s+([\d.,]+\s+€)', text)
    gesamt_match = re.search(r'Gesamtbetrag:\s+([\d.,]+\s+€)', text)
    
    daten['nettobetrag'] = netto_match.group(1) if netto_match else None
    daten['mwst'] = mwst_match.group(1) if mwst_match else None
    daten['gesamtbetrag'] = gesamt_match.group(1) if gesamt_match else None
    
    # IBAN
    iban_match = re.search(r'IBAN\s+([A-Z]{2}\d{2}\s+(?:\d{4}\s+){4,}\d{2})', text)
    daten['iban'] = iban_match.group(1) if iban_match else None
    
    return daten

print("SMART DATA EXTRACTOR - RECHNUNG")
print("=" * 45)

extrahierte_daten = extrahiere_rechnungsdaten(rechnung_text)

# Strukturierte Ausgabe
print("📄 RECHNUNGSINFORMATIONEN:")
print(f"  Nummer: {extrahierte_daten['rechnungsnummer']}")
print(f"  Datum: {extrahierte_daten['rechnungsdatum']}")
print(f"  Fällig: {extrahierte_daten['faelligkeitsdatum']}")

print(f"\n👤 KUNDE:")
print(f"  Name: {extrahierte_daten['kunde']}")
print(f"  Adresse: {extrahierte_daten['adresse']}")
print(f"  USt-IdNr: {extrahierte_daten['ust_idnr']}")

print(f"\n🛒 ARTIKEL ({len(extrahierte_daten['artikel'])} Stück):")
for i, (artikel, preis) in enumerate(extrahierte_daten['artikel'], 1):
    print(f"  {i}. {artikel} - {preis}")

print(f"\n💰 BETRÄGE:")
print(f"  Netto: {extrahierte_daten['nettobetrag']}")
print(f"  MwSt: {extrahierte_daten['mwst']}")
print(f"  Gesamt: {extrahierte_daten['gesamtbetrag']}")

print(f"\n🏦 BANKVERBINDUNG:")
print(f"  IBAN: {extrahierte_daten['iban']}")

# Validierung der extrahierten Daten
print(f"\n✅ DATENQUALITÄT:")
vollstaendige_felder = sum(1 for wert in extrahierte_daten.values() if wert)
gesamt_felder = len(extrahierte_daten)
print(f"Vollständigkeit: {vollstaendige_felder}/{gesamt_felder} Felder ({vollstaendige_felder/gesamt_felder*100:.0f}%)")

# Datum-Validierung
if extrahierte_daten['rechnungsdatum'] and extrahierte_daten['faelligkeitsdatum']:
    print("Datumsformat: Korrekt (DD.MM.YYYY)")
else:
    print("Datumsformat: Probleme erkannt")
```

**Erklärung:**
- Komplexere RegEx-Muster für strukturierte Datenextraktion
- `re.findall()` für wiederholende Elemente (Artikel)
- Non-greedy Matching `.*?` für präzise Erfassung
- Strukturierte Datenrückgabe als Dictionary
- Datenqualitäts-Prüfung der extrahierten Informationen
