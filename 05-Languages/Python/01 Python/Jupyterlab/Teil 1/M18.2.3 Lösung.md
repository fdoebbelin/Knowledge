# Zelle 3.1: Pure Function für Rabattberechnung entwickeln
def entwickle_testbare_rabattlogik():
    """Trennt Seiteneffekte von Business-Logik für bessere Testbarkeit"""
    
    # Pure Function - keine Seiteneffekte, nur Logik
    def berechne_rabatt(bisherige_kaeufe, kunde_typ):
        """Pure Function - berechnet Rabatt ohne Seiteneffekte"""
        
        # Basis-Rabatt basierend auf Kaufhistorie
        if bisherige_kaeufe > 1000:
            basis_rabatt = 15  # Großkunde
        elif bisherige_kaeufe > 500:
            basis_rabatt = 10  # Stammkunde
        else:
            basis_rabatt = 5   # Neukunde
        
        # Premium-Bonus
        if kunde_typ == "premium":
            end_rabatt = basis_rabatt + 5
        else:
            end_rabatt = basis_rabatt
        
        # Maximal 25% Rabatt
        return min(end_rabatt, 25)
    
    # Testbarkeit demonstrieren - deterministisch
    test_faelle = [
        {'kaeufe': 100, 'typ': 'standard'},
        {'kaeufe': 600, 'typ': 'premium'},

            'kaeufe': test['kaeufe'],
            'typ': test['typ'],
            'rabatt': rabatt
        })

# Zelle 4.1: Robuste E-Commerce Funktionen mit Eingabevalidierung
def implementiere_defensive_programmierung():
    """Implementiert defensive Programmierung für robuste Funktionen"""
    
    def verarbeite_ecommerce_bestellung(produkt_name, preis, menge, rabatt_prozent, kunde_typ, bisherige_kaeufe):
        """Robuste E-Commerce Funktion mit vollständiger Eingabevalidierung"""
        
        # Eingabevalidierung - Defensive Programmierung
        if not isinstance(produkt_name, str) or not produkt_name.strip():
            return {'fehler': 'Produktname muss ein nicht-leerer Text sein'}
        
        if not isinstance(preis, (int, float)) or preis <= 0:
            return {'fehler': 'Preis muss eine positive Zahl sein'}
        
        if not isinstance(menge, int) or menge <= 0:
            return {'fehler': 'Menge muss eine positive ganze Zahl sein'}
        
        if not isinstance(rabatt_prozent, (int, float)) or not (0 <= rabatt_prozent <= 100):
            return {'fehler': 'Rabatt muss zwischen 0 und 100 Prozent liegen'}
        
        if kunde_typ not in ['standard', 'premium']:
            return {'fehler': 'Kundentyp muss "standard" oder "premium" sein'}
        
        if not isinstance(bisherige_kaeufe, (int, float)) or bisherige_kaeufe < 0:
            return {'fehler': 'Bisherige Käufe müssen eine nicht-negative Zahl sein'}
        
        # Hilfsfunktionen (Pure Functions aus vorherigen Lösungen)
        def berechne_produktpreis(preis, menge, rabatt_prozent):
            rabatt_betrag = preis * menge * (rabatt_prozent / 100)
            return (preis * menge) - rabatt_betrag
        
        def berechne_kundenrabatt(bisherige_kaeufe, kunde_typ):
            if bisherige_kaeufe > 1000:
                basis_rabatt = 15
            elif bisherige_kaeufe > 500:
                basis_rabatt = 10
            else:
                basis_rabatt = 5
            end_rabatt = basis_rabatt + 5 if kunde_typ == "premium" else basis_rabatt
            return min(end_rabatt, 25)
        
        def berechne_bestellung_total(artikel_summe, kunde_typ, versandkosten=5.99, steuersatz=0.19):
            versand = 0.0 if kunde_typ == "premium" else versandkosten
            steuer = artikel_summe * steuersatz
            return artikel_summe + versand + steuer
        
        # Berechnung mit validierten Eingaben
        try:
            produkt_preis = berechne_produktpreis(preis, menge, rabatt_prozent)
            kunden_rabatt = berechne_kundenrabatt(bisherige_kaeufe, kunde_typ)
            
            # Kundenrabatt anwenden
            preis_nach_kundenrabatt = produkt_preis * (1 - kunden_rabatt / 100)
            
            # Gesamtbestellung berechnen
            endpreis = berechne_bestellung_total(preis_nach_kundenrabatt, kunde_typ)
            
            return {
                'erfolg': True,
                'produkt': produkt_name,
                'basis_preis': produkt_preis,
                'kundenrabatt_prozent': kunden_rabatt,
                'preis_nach_rabatt': preis_nach_kundenrabatt,
                'endpreis': endpreis,
                'kunde_typ': kunde_typ
            }
            
        except Exception as e:
            return {'fehler': f'Fehler bei der Berechnung: {str(e)}'}
    
    # Robustheit testen - gültige und ungültige Eingaben
    test_faelle = [
        # Gültiger Fall
        {'name': 'Gaming Maus', 'preis': 49.99, 'menge': 2, 'rabatt': 10, 'typ': 'premium', 'kaeufe': 600},
        # Ungültige Fälle
        {'name': '', 'preis': 49.99, 'menge': 2, 'rabatt': 10, 'typ': 'premium', 'kaeufe': 600},
        {'name': 'Maus', 'preis': -10, 'menge': 2, 'rabatt': 10, 'typ': 'premium', 'kaeufe': 600},
        {'name': 'Maus', 'preis': 49.99, 'menge': 2, 'rabatt': 150, 'typ': 'premium', 'kaeufe': 600},
    ]
    
    test_ergebnisse = []
    for i, test in enumerate(test_faelle, 1):
        ergebnis = verarbeite_ecommerce_bestellung(
            test['name'], test['preis'], test['menge'], 
            test['rabatt'], test['typ'], test['kaeufe']
        )
        test_ergebnisse.append({'test': i, 'ergebnis': ergebnis})
    
    print("\n🛡️ SCHRITT 4.1: Defensive Programmierung")
    print("=" * 50)
    print("✅ Vollständige Eingabevalidierung implementiert")
    print("✅ Hilfreiche Fehlermeldungen bei ungültigen Eingaben")
    print("✅ Robuste Fehlerbehandlung mit try-catch")
    print("\n🧪 Robustheitstests:")
    
    for test_ergebnis in test_ergebnisse:
        test_nr = test_ergebnis['test']
        ergebnis = test_ergebnis['ergebnis']
        
        if 'fehler' in ergebnis:
            print(f"   Test {test_nr}: ❌ {ergebnis['fehler']}")

    """Führt alle Clean Code Komponenten zu einem funktionalen System zusammen"""
    
    # Alle Pure Functions zusammengefasst
    def berechne_produktpreis(preis, menge, rabatt_prozent):
        """Pure Function - Produktpreis berechnen"""
        rabatt_betrag = preis * menge * (rabatt_prozent / 100)
        return (preis * menge) - rabatt_betrag
    
    def berechne_kundenrabatt(bisherige_kaeufe, kunde_typ):
        """Pure Function - Kundenrabatt basierend auf Historie"""
        if bisherige_kaeufe > 1000:
            basis_rabatt = 15
        elif bisherige_kaeufe > 500:
            basis_rabatt = 10
        else:
            basis_rabatt = 5
        end_rabatt = basis_rabatt + 5 if kunde_typ == "premium" else basis_rabatt
        return min(end_rabatt, 25)
    
    def berechne_versand_und_steuer(betrag, kunde_typ, versandkosten=5.99, steuersatz=0.19):
        """Pure Function - Versand und Steuer berechnen"""
        versand = 0.0 if kunde_typ == "premium" else versandkosten
        steuer = betrag * steuersatz
        return versand, steuer
    
    def formatiere_bestellung(produkt, preis_details, kunde_info):
        """Pure Function - Bestellung formatieren"""
        return {
            'produkt_name': produkt['name'],
            'menge': produkt['menge'],
            'basis_preis': preis_details['basis_preis'],
            'produkt_rabatt': preis_details['produkt_rabatt_prozent'],
            'preis_nach_produkt_rabatt': preis_details['preis_nach_produkt_rabatt'],
            'kundenrabatt_prozent': preis_details['kundenrabatt_prozent'],
            'preis_nach_kundenrabatt': preis_details['preis_nach_kundenrabatt'],
            'versandkosten': preis_details['versandkosten'],
            'steuer': preis_details['steuer'],
            'endpreis': preis_details['endpreis'],
            'kunde_typ': kunde_info['typ'],
            'kunde_kategorie': kunde_info['kategorie']
        }
    
    # Vollständiger E-Commerce Workflow
    def verarbeite_vollstaendige_bestellung(produkt_info, kunde_info):
        """Integriert alle Clean Code Komponenten zu einem vollständigen System"""
        
        # 1. Produktpreis berechnen (mit Produktrabatt)
        basis_preis = berechne_produktpreis(
            produkt_info['preis'], 
            produkt_info['menge'], 
            produkt_info['rabatt_prozent']
        )
        
        # 2. Kundenrabatt berechnen
        kundenrabatt_prozent = berechne_kundenrabatt(
            kunde_info['bisherige_kaeufe'], 
            kunde_info['typ']
        )
        
        # 3. Kundenrabatt anwenden
        preis_nach_kundenrabatt = basis_preis * (1 - kundenrabatt_prozent / 100)
        
        # 4. Versand und Steuer berechnen
        versandkosten, steuer = berechne_versand_und_steuer(
            preis_nach_kundenrabatt, 
            kunde_info['typ']
        )
        
        # 5. Endpreis ermitteln
        endpreis = preis_nach_kundenrabatt + versandkosten + steuer
        
        # 6. Kundenkategorie bestimmen
        if kunde_info['bisherige_kaeufe'] > 1000:
            kunde_kategorie = "Großkunde"
        elif kunde_info['bisherige_kaeufe'] > 500:
            kunde_kategorie = "Stammkunde"
        else:
            kunde_kategorie = "Neukunde"
        
        # 7. Alle Details sammeln
        preis_details = {
            'basis_preis': basis_preis,
            'produkt_rabatt_prozent': produkt_info['rabatt_prozent'],
            'preis_nach_produkt_rabatt': basis_preis,
            'kundenrabatt_prozent': kundenrabatt_prozent,
            'preis_nach_kundenrabatt': preis_nach_kundenrabatt,
            'versandkosten': versandkosten,
            'steuer': steuer,
            'endpreis': endpreis
        }
        
        kunde_info_erweitert = {
            'typ': kunde_info['typ'],
            'kategorie': kunde_kategorie
        }
        
        # 8. Formatierte Bestellung erstellen
        bestellung = formatiere_bestellung(produkt_info, preis_details, kunde_info_erweitert)
        
        return bestellung
    
    # Beispiel-Bestellung mit dem integrierten System
    beispiel_produkt = {
        'name': 'Gaming Maus Pro',
        'preis': 79.99,
        'menge': 2,
        'rabatt_prozent': 15
    }
    
    beispiel_kunde = {
        'typ': 'premium',
        'bisherige_kaeufe': 750
    }
    
    bestellung = verarbeite_vollstaendige_bestellung(beispiel_produkt, beispiel_kunde)
    
    print("\n🛒 SCHRITT 4.2: Integriertes Clean E-Commerce System")
    print("=" * 60)
    print("✅ Alle refaktorierten Komponenten integriert")
    print("✅ Hohe Kohäsion: Jede Funktion hat eine klare Aufgabe")
    print("✅ Lose Kopplung: Parameter statt globale Variablen")
    print("✅ Testbar: Pure Functions ohne Seiteneffekte")
    print("✅ Defensive Programmierung mit Validierung")
    
    print(f"\n🧾 BEISPIEL-BESTELLUNG:")
    print(f"   📦 Produkt: {bestellung['produkt_name']} ({bestellung['menge']}x)")
    print(f"   💰 Basis-Preis: {bestellung['basis_preis']:.2f}€")
    print(f"   🏷️ Produkt-Rabatt: {bestellung['produkt_rabatt']}%")
    print(f"   👤 Kunde: {bestellung['kunde_kategorie']} ({bestellung['kunde_typ']})")
    print(f"   🎯 Kundenrabatt: {bestellung['kundenrabatt_prozent']}%")
    print(f"   📦 Versandkosten: {bestellung['versandkosten']:.2f}€")
    print(f"   🧾 Steuer: {bestellung['steuer']:.2f}€")
    print(f"   💳 GESAMT: {bestellung['endpreis']:.2f}€")
    
    print(f"\n📈 CLEAN CODE ERFOLG:")
    print(f"   ✅ 6 Pure Functions ohne Seiteneffekte")
    print(f"   ✅ Modulare Architektur mit klaren Verantwortlichkeiten")
    print(f"   ✅ Vollständig testbar und wartbar")
    print(f"   ✅ Flexibel erweiterbar für neue Anforderungen")
    
    return f"Clean E-Commerce System: Bestellung {bestellung['endpreis']:.2f}€ erfolgreich verarbeitet"

print(integriere_clean_ecommerce_system())

## Zusammenfassung - Clean Code Transformation

### Vorher vs. Nachher Vergleich:
**❌ Legacy-Code Probleme:**
- Niedrige Kohäsion: Eine Funktion machte alles
- Starke Kopplung: Globale Variablen überall
- Seiteneffekte: print-Statements in Business-Logik
- Schwer testbar: Keine klaren Ein-/Ausgaben
- Nicht wartbar: Code-Duplikation und Magic Numbers

**✅ Clean Code Lösung:**
- Hohe Kohäsion: Jede Funktion hat eine klare Aufgabe
- Lose Kopplung: Parameter statt globale Abhängigkeiten
- Pure Functions: Seiteneffekte isoliert
- Vollständig testbar: Deterministische Funktionen
- Wartbar: Modulare Struktur und defensive Programmierung

### Angewendete Best Practices:
1. **Single Responsibility Principle** - Eine Funktion, eine Aufgabe
2. **Dependency Injection** - Parameter statt globale Variablen
3. **Pure Functions** - Keine Seiteneffekte in Business-Logik
4. **Defensive Programming** - Eingabevalidierung und Fehlerbehandlung
5. **Separation of Concerns** - Logik, Formatierung und Ausgabe getrennt

### Lernergebnisse:
- ✅ Kohäsion und Kopplung praktisch verstanden
- ✅ Testbaren Code mit Pure Functions entwickelt
- ✅ Seiteneffekte erfolgreich isoliert
- ✅ Defensive Programmierung implementiert
- ✅ Komplettes System nach Clean Code Prinzipien refaktoriert
