## Aufgabe 3: Einkaufsliste mit Preisen

```python
# 1. Dictionary mit 5 Produkten erstellen
einkaufsliste = {
    "Milch": 1.20,
    "Brot": 2.50,
    "Äpfel": 3.80,
    "Käse": 4.20,
    "Butter": 2.30
}

print("Ursprüngliche Einkaufsliste:")
for produkt, preis in einkaufsliste.items():
    print(f"  {produkt}: {preis:.2f}€")

# 2. Gesamtsumme berechnen
gesamtsumme = sum(einkaufsliste.values())
print(f"\nGesamtsumme: {gesamtsumme:.2f}€")

# 3. Teuerstes Produkt finden
teuerster_preis = max(einkaufsliste.values())
# Produkt mit diesem Preis finden
for produkt, preis in einkaufsliste.items():
    if preis == teuerster_preis:
        teuerstes_produkt = produkt
        break

print(f"Teuerstes Produkt: {teuerstes_produkt} ({teuerster_preis:.2f}€)")

# 4. Zwei neue Produkte hinzufügen
einkaufsliste["Eier"] = 2.80
einkaufsliste["Joghurt"] = 1.60

# 5. Alle Produkte nach Preis sortiert ausgeben
print("\n--- Sortiert nach Preis ---")
# Sortierung: items() gibt Tupel zurück, wir sortieren nach dem 2. Element (Preis)
sortierte_produkte = sorted(einkaufsliste.items(), key=lambda x: x[1])

for produkt, preis in sortierte_produkte:
    print(f"  {produkt}: {preis:.2f}€")

print(f"\nNeue Gesamtsumme: {sum(einkaufsliste.values()):.2f}€")
```

## Aufgabe 4: Wörterzähler

```python
# Gegebener Text
text = "Python ist toll Python macht Spaß Python ist einfach"

# Text in Wörter aufteilen (Kleinbuchstaben für einheitliche Zählung)
woerter = text.lower().split()

print(f"Gefundene Wörter: {woerter}")

# Wörter zählen
woerter_count = {}

# Methode 1: Mit if-Bedingung
for wort in woerter:
    if wort in woerter_count:
        woerter_count[wort] += 1  # Wort schon vorhanden, +1
    else:
        woerter_count[wort] = 1   # Neues Wort, erste Zählung

print(f"\nWörter-Häufigkeit: {woerter_count}")

# Alternative Methode 2: Mit get() (eleganter)
woerter_count_alt = {}
for wort in woerter:
    # get() gibt 0 zurück falls Wort nicht existiert, sonst aktueller Wert
    woerter_count_alt[wort] = woerter_count_alt.get(wort, 0) + 1

print(f"Alternative Zählung: {woerter_count_alt}")

# Die 3 häufigsten Wörter finden
# Sortieren nach Häufigkeit (absteigend)
haeufigste_woerter = sorted(woerter_count.items(), key=lambda x: x[1], reverse=True)

print("\n--- Top 3 häufigste Wörter ---")
for i in range(min(3, len(haeufigste_woerter))):  # Falls weniger als 3 verschiedene Wörter
    wort, anzahl = haeufigste_woerter[i]
    print(f"{i+1}. '{wort}': {anzahl}x")
```
## Aufgabe 5: Schülerverwaltung

```python
# Startdaten
schueler = {
    "Anna": [1.3, 2.0, 1.7],
    "Bob": [2.3, 1.7, 2.0], 
    "Clara": [1.0, 1.3, 1.7]
}

print("Ursprüngliche Schülerdaten:")
for name, noten in schueler.items():
    print(f"  {name}: {noten}")

# Funktion: Neue Note hinzufügen
def note_hinzufuegen(name, note):
    """Fügt einem Schüler eine neue Note hinzu"""
    if name in schueler:
        schueler[name].append(note)
        print(f"Note {note} für {name} hinzugefügt!")
    else:
        print(f"Schüler {name} nicht gefunden!")

# Funktion: Durchschnitt eines Schülers
def durchschnitt_schueler(name):
    """Berechnet den Notendurchschnitt eines Schülers"""
    if name in schueler:
        noten = schueler[name]
        durchschnitt = sum(noten) / len(noten)
        return durchschnitt
    else:
        print(f"Schüler {name} nicht gefunden!")
        return None

# Funktion: Klassendurchschnitt
def klassendurchschnitt():
    """Berechnet den Durchschnitt der ganzen Klasse"""
    alle_noten = []
    # Alle Noten aller Schüler sammeln
    for noten_liste in schueler.values():
        alle_noten.extend(noten_liste)  # extend() fügt alle Elemente hinzu
    
    if alle_noten:
        return sum(alle_noten) / len(alle_noten)
    else:
        return 0

# Funktion: Besten Schüler finden
def bester_schueler():
    """Findet den Schüler mit dem besten Durchschnitt"""
    bester_name = None
    bester_durchschnitt = float('inf')  # Unendlich, damit jeder Wert besser ist
    
    for name in schueler:
        durchschnitt = durchschnitt_schueler(name)
        if durchschnitt < bester_durchschnitt:  # Kleinere Note = besser
            bester_durchschnitt = durchschnitt
            bester_name = name
    
    return bester_name, bester_durchschnitt

# Funktionen testen:
print("\n--- Funktionen testen ---")

# Neue Note hinzufügen
note_hinzufuegen("Anna", 1.0)
print(f"Anna's Noten jetzt: {schueler['Anna']}")

# Durchschnitte berechnen
print("\n--- Individuelle Durchschnitte ---")
for name in schueler:
    durchschnitt = durchschnitt_schueler(name)
    print(f"{name}: {durchschnitt:.2f}")

# Klassendurchschnitt
klassen_durchschnitt = klassendurchschnitt()
print(f"\nKlassendurchschnitt: {klassen_durchschnitt:.2f}")

# Bester Schüler
bester_name, bester_wert = bester_schueler()
print(f"Bester Schüler: {bester_name} mit Durchschnitt {bester_wert:.2f}")
```

```python
alle_produkte = set(preise.keys()) | set(lager.keys())  # Vereinigung der Schlüssel
print(f"\nAlle Produkte: {sorted(alle_produkte)}")

# 2. + 3. Kombiniertes Dictionary erstellen und Übersicht ausgeben
print("\n--- Produktübersicht ---")

kombiniert = {}
for produkt in alle_produkte:
    # Sichere Zugriffe mit get() und Standardwerten
    preis = preise.get(produkt, "Nicht verfügbar")
```

## Bonus-Aufgabe: Mini-Datenbank

```python
# Mini-Film-Datenbank
filme_db = {}

def film_hinzufuegen(titel, jahr, genre, bewertung):
    """Fügt einen neuen Film zur Datenbank hinzu"""
    filme_db[titel] = {
        "jahr": jahr,
        "genre": genre, 
        "bewertung": bewertung
    }
    print(f"Film '{titel}' hinzugefügt!")

def film_suchen(titel):
    """Sucht einen Film nach Titel"""
    if titel in filme_db:
        film = filme_db[titel]
        print(f"\nGefunden: {titel}")
        print(f"  Jahr: {film['jahr']}")
        print(f"  Genre: {film['genre']}")
        print(f"  Bewertung: {film['bewertung']}/10")
        return filme_db[titel]
    else:
        print(f"Film '{titel}' nicht gefunden!")
        return None

def filme_nach_genre(genre):
    """Zeigt alle Filme eines bestimmten Genres"""
    gefundene_filme = []
    for titel, film in filme_db.items():
        if film["genre"].lower() == genre.lower():
            gefundene_filme.append(titel)
    
    if gefundene_filme:
        print(f"\n--- {genre}-Filme ---")
        for titel in gefundene_filme:
            bewertung = filme_db[titel]["bewertung"]
            jahr = filme_db[titel]["jahr"]
            print(f"  {titel} ({jahr}) - {bewertung}/10")
    else:
        print(f"Keine {genre}-Filme gefunden!")
    
    return gefundene_filme

def top_filme(anzahl=3):
    """Zeigt die Top-Filme nach Bewertung"""
    # Liste aller Filme mit Bewertungen erstellen
    film_bewertungen = [(titel, daten["bewertung"]) for titel, daten in filme_db.items()]
    
    # Nach Bewertung sortieren (absteigend)
    sortierte_filme = sorted(film_bewertungen, key=lambda x: x[1], reverse=True)
    
    print(f"\n--- Top {anzahl} Filme ---")
    for i in range(min(anzahl, len(sortierte_filme))):
        titel, bewertung = sortierte_filme[i]
        jahr = filme_db[titel]["jahr"] 
        genre = filme_db[titel]["genre"]
        print(f"{i+1}. {titel} ({jahr}, {genre}) - {bewertung}/10")

def durchschnittsbewertung():
    """Berechnet die Durchschnittsbewertung aller Filme"""
    if not filme_db:
        return 0
    
    alle_bewertungen = [film["bewertung"] for film in filme_db.values()]
    durchschnitt = sum(alle_bewertungen) / len(alle_bewertungen)
    return durchschnitt

# Testdaten hinzufügen
print("=== Film-Datenbank Test ===")
film_hinzufuegen("Der Herr der Ringe", 2001, "Fantasy", 9)
film_hinzufuegen("Matrix", 1999, "Sci-Fi", 8)
film_hinzufuegen("Titanic", 1997, "Drama", 7)
film_hinzufuegen("Avatar", 2009, "Sci-Fi", 8)
film_hinzufuegen("Der Hobbit", 2012, "Fantasy", 6)

# Funktionen testen
film_suchen("Matrix")
filme_nach_genre("Fantasy")
top_filme(3)

durchschnitt = durchschnittsbewertung()
print(f"\nDurchschnittsbewertung aller Filme: {durchschnitt:.1f}/10")

print(f"\nGesamtanzahl Filme in Datenbank: {len(filme_db)}")
```
## Zusammenfassung: Wichtige Dictionary-Operationen

```python
# Die wichtigsten Dictionary-Operationen im Überblick:

d = {"a": 1, "b": 2}

# Zugriff
wert = d["a"]           # Direkt (kann KeyError verursachen)
wert = d.get("a", 0)    # Sicher mit Standardwert

# Hinzufügen/Ändern  
d["c"] = 3              # Neuen Schlüssel hinzufügen
d["a"] = 10             # Bestehenden Wert ändern

# Löschen
del d["a"]              # Löschen (kann KeyError verursachen)
wert = d.pop("b", 0)    # Löschen und Wert zurückgeben

# Iteration
for schluessel in d:                    # Über Schlüssel
for wert in d.values():                 # Über Werte  
for schluessel, wert in d.items():      # Über Paare

# Prüfungen
if "a" in d:            # Schlüssel vorhanden?
if d:                   # Dictionary nicht leer?

# Nützliche Methoden
d.clear()               # Alles löschen
d.copy()                # Kopie erstellen
d.update(other)         # Anderen Dict hinzufügen
```
