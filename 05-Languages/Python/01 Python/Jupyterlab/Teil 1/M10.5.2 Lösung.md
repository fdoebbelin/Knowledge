def berechne_ordner_groesse(ordner_struktur):
    """
    Berechnet rekursiv die Gesamtgröße einer Ordnerstruktur.
    
    Args:
        ordner_struktur (dict): Dictionary mit 'dateien' und 'unterordner'
        
    Returns:
        int: Gesamtgröße in KB
    """
    print(f"🔄 Berechne Ordnergröße...")
    
    # Basisfall: Größe der direkten Dateien berechnen
    dateien = ordner_struktur.get("dateien", [])
    direkte_groesse = sum(dateien)
    print(f"   📄 Direkte Dateien: {dateien} → {direkte_groesse} KB")
    
    # Rekursiver Fall: Größe aller Unterordner berechnen
    unterordner = ordner_struktur.get("unterordner", {})
    unterordner_groesse = 0
    
    for ordner_name, unter_ordner in unterordner.items():
        print(f"   📂 Berechne Unterordner '{ordner_name}':")
        ordner_groesse = berechne_ordner_groesse(unter_ordner)  # REKURSION!
        print(f"   📂 '{ordner_name}' hat {ordner_groesse} KB")
        unterordner_groesse += ordner_groesse
    
    # Gesamtgröße = direkte Dateien + alle Unterordner
    gesamtgroesse = direkte_groesse + unterordner_groesse
    print(f"   ✨ Gesamtgröße: {direkte_groesse} + {unterordner_groesse} = {gesamtgroesse} KB")
    
    return gesamtgroesse

def kb_zu_mb(kb_wert):
    """Konvertiert KB zu MB mit 1 Dezimalstelle."""
    return round(kb_wert / 1024, 1)

def formatiere_groesse(kb_wert):
    """Formatiert Dateigröße benutzerfreundlich."""
    if kb_wert < 1024:
        return f"{kb_wert:,} KB"
    else:
        mb_wert = kb_zu_mb(kb_wert)
        return f"{kb_wert:,} KB ({mb_wert} MB)"

def formatiere_dateiliste(dateien):
    """Formatiert eine Dateiliste als String."""
    if not dateien:
        return "keine Dateien"
    elif len(dateien) <= 5:
        return " + ".join(map(str, dateien))
    else:
        ersten_drei = " + ".join(map(str, dateien[:3]))
        return f"{ersten_drei} + ... ({len(dateien)} Dateien total)"

# Test der Formatierungsfunktionen

def berechne_ordner_groesse_clean(ordner_struktur):

    Args:
        ordner_struktur (dict): Dictionary mit 'dateien' und 'unterordner'

        int: Gesamtgröße in KB

    # Rekursiver Fall: Größe aller Unterordner
    unterordner = ordner_struktur.get("unterordner", {})
    unterordner_groesse = 0

def zeige_ordner_details(ordner_struktur, ordner_name="Hauptordner", tiefe=0):
    """
    Zeigt Ordnerstruktur hierarchisch mit Größenangaben an.
    
    Args:
        ordner_struktur (dict): Ordnerstruktur
        ordner_name (str): Name des aktuellen Ordners
        tiefe (int): Verschachtelungstiefe für Einrückung
        
    Returns:
        int: Größe des aktuellen Ordners
    """
    einrueckung = "  " * tiefe
    symbol = "📁" if tiefe == 0 else "📂"
    
    # Dateien in diesem Ordner
    dateien = ordner_struktur.get("dateien", [])
    direkte_groesse = sum(dateien)
    
    # Unterordner berechnen
    unterordner = ordner_struktur.get("unterordner", {})
    unterordner_groesse = 0
    
    # Rekursiv alle Unterordner verarbeiten
    unterordner_details = []
    for unter_name, unter_ordner in unterordner.items():
        unter_groesse = zeige_ordner_details(unter_ordner, unter_name, tiefe + 1)
        unterordner_groesse += unter_groesse
        unterordner_details.append((unter_name, unter_groesse))
    
    # Gesamtgröße berechnen
    gesamtgroesse = direkte_groesse + unterordner_groesse
    
    # Ausgabe für aktuellen Ordner
    print(f"{einrueckung}{symbol} {ordner_name}: {formatiere_groesse(gesamtgroesse)}")
    
    if dateien:
        dateien_str = formatiere_dateiliste(dateien)
        print(f"{einrueckung}   📄 Dateien: {formatiere_groesse(direkte_groesse)} ({dateien_str})")
    
    if unterordner_details:
        gesamt_unterordner = sum(groesse for _, groesse in unterordner_details)
        print(f"{einrueckung}   📂 Unterordner: {formatiere_groesse(gesamt_unterordner)}")
    
    return gesamtgroesse

def erweiterte_ordner_analyse(ordner_struktur, ordner_name="Hauptordner"):
    """
    Erweiterte Analyse mit Statistiken.
    
    Returns:
        dict: Ausführliche Statistiken über die Ordnerstruktur
    """
    def sammle_statistiken(ordner, tiefe=0):
        """Rekursive Hilfsfunktion zum Sammeln von Statistiken."""
        dateien = ordner.get("dateien", [])
        unterordner = ordner.get("unterordner", {})
        
        stats = {
            'groesse': sum(dateien),
            'anzahl_dateien': len(dateien),
            'anzahl_ordner': len(unterordner),
            'max_tiefe': tiefe,
            'dateien_nach_groesse': dateien.copy()
        }
        
        # Rekursiv durch Unterordner
        for unter_ordner in unterordner.values():
            unter_stats = sammle_statistiken(unter_ordner, tiefe + 1)
            stats['groesse'] += unter_stats['groesse']
            stats['anzahl_dateien'] += unter_stats['anzahl_dateien']
            stats['anzahl_ordner'] += unter_stats['anzahl_ordner']
            stats['max_tiefe'] = max(stats['max_tiefe'], unter_stats['max_tiefe'])
            stats['dateien_nach_groesse'].extend(unter_stats['dateien_nach_groesse'])
        
        return stats
    
    # Statistiken sammeln
    stats = sammle_statistiken(ordner_struktur)
    
    # Erweiterte Berechnungen
    dateien_groessen = stats['dateien_nach_groesse']
    if dateien_groessen:
        stats['groesste_datei'] = max(dateien_groessen)
        stats['kleinste_datei'] = min(dateien_groessen)
        stats['durchschnitt_datei'] = round(sum(dateien_groessen) / len(dateien_groessen), 1)
    else:
        stats['groesste_datei'] = 0
        stats['kleinste_datei'] = 0
        stats['durchschnitt_datei'] = 0
    
    return stats

   - Direkte Berechnung ohne weitere Rekursion
