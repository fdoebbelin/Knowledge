## 3.1 f-Strings - Die moderne Art

f-Strings sind die beste und einfachste Art, Variablen in Text einzubauen:

```python
# Grundlagen von f-Strings
name = "Anna"
alter = 25
stadt = "Berlin"

# Variablen in Text einbauen
nachricht = f"Hallo {name}! Du bist {alter} Jahre alt und wohnst in {stadt}."
print(nachricht)

# Mit Berechnungen
preis = 19.99
anzahl = 3
gesamt = f"Gesamt: {preis * anzahl} Euro"
print(gesamt)
```

```python
# f-Strings mit Formatierung
zahl = 123.456789

print(f"Original: {zahl}")
print(f"2 Stellen nach Komma: {zahl:.2f}")
print(f"Ohne Nachkommastellen: {zahl:.0f}")

# Große Zahlen lesbar machen
große_zahl = 1234567
print(f"Mit Tausender-Punkten: {große_zahl:,}")
print(f"Geld formatiert: {große_zahl:.2f} Euro")
```

```python
# Prozentangaben und andere Formate
anteil = 0.75
note = 2.3

print(f"Anteil als Prozent: {anteil:.1%}")

# Text ausrichten
wort = "Python"
print(f"Links:     '{wort:<10}'")     # 10 Zeichen breit, links
print(f"Rechts:    '{wort:>10}'")     # 10 Zeichen breit, rechts  
print(f"Zentriert: '{wort:^10}'")     # 10 Zeichen breit, zentriert
```
## 3.2 Praktische Beispiele

```python
# Persönliche Informationen formatieren
vorname = "Max"
nachname = "Mustermann"
gehalt = 3500.50
arbeitszeit = 0.8  # 80% Stelle

vollname = f"{vorname} {nachname}"
jahresgehalt = gehalt * 12
prozent_stelle = arbeitszeit * 100

print(f"Name: {vollname}")
print(f"Monatsgehalt: {gehalt:.2f} Euro")
print(f"Jahresgehalt: {jahresgehalt:,.2f} Euro")
print(f"Arbeitszeit: {prozent_stelle:.0f}% Stelle")
```

```python
# Tabellen-ähnliche Ausgabe erstellen
personen = [
    ("Anna", 25, 2800),
    ("Bob", 30, 3200),
    ("Charlie", 28, 2950)
]

print("Name      Alter  Gehalt")
print("-" * 25)

for name, alter, gehalt in personen:
    print(f"{name:<9} {alter:>5}  {gehalt:>6}")

# Oder mit besserer Formatierung
print("\n--- Schönere Tabelle ---")
print("│ Name     │ Alter │ Gehalt    │")
print("├──────────┼───────┼───────────┤")

for name, alter, gehalt in personen:
    print(f"│ {name:<8} │ {alter:>5} │ {gehalt:>7.0f} € │")
```

```python
# Einkaufszettel erstellen
einkäufe = [
    ("Äpfel", 2.5, 1.99),
    ("Brot", 1, 2.49),
    ("Milch", 2, 1.29)
]

print("🛒 EINKAUFSZETTEL")
print("=" * 35)

gesamt_preis = 0
for artikel, menge, preis_pro_einheit in einkäufe:
    gesamt_artikel = menge * preis_pro_einheit
    gesamt_preis += gesamt_artikel
    
    print(f"{artikel:<10} {menge:>4.1f}x {preis_pro_einheit:>5.2f}€ = {gesamt_artikel:>6.2f}€")

print("-" * 35)
print(f"{'GESAMT':<10}                  {gesamt_preis:>6.2f}€")
```

## 3.4 Datum und Zeit formatieren

```python
# Datum und Zeit schön darstellen
from datetime import datetime

jetzt = datetime.now()
geburtstag = datetime(1995, 7, 15)

# Verschiedene Datumsformate
print("Datum formatieren:")
print(f"Vollständig: {jetzt}")
print(f"Nur Datum: {jetzt.strftime('%d.%m.%Y')}")
print(f"Nur Zeit: {jetzt.strftime('%H:%M')}")
print(f"Wochentag: {jetzt.strftime('%A')}")

# Alter berechnen und anzeigen
alter_tage = (jetzt - geburtstag).days
alter_jahre = alter_tage / 365.25

print(f"Geburtstag: {geburtstag.strftime('%d.%m.%Y')}")
print(f"Alter: {alter_jahre:.1f} Jahre")
print(f"Das sind {alter_tage:,} Tage!")
```
## 3.5 Alternative Formatierungsmethoden

```python
# .format() Methode (ältere Art, aber noch wichtig)
name = "Lisa"
punkte = 87.5

# Verschiedene Schreibweisen
text1 = "Spieler: {}, Punkte: {}".format(name, punkte)
text2 = "Spieler: {0}, Punkte: {1:.1f}".format(name, punkte)
text3 = "Spieler: {spieler}, Punkte: {score:.1f}".format(spieler=name, score=punkte)

print("Mit .format():")
print(text1)
print(text2) 
print(text3)

# Wann .format() statt f-String?
# - Wenn der Format-String aus einer Datei/Datenbank kommt
format_vorlage = "Hallo {name}, dein Kontostand ist {betrag:.2f} Euro"
nachricht = format_vorlage.format(name="Max", betrag=1234.56)
print(f"\nAus Vorlage: {nachricht}")
```

```python
# Prozent-Formatierung (sehr alte Art - nur zum Verstehen)
name = "Tom"
alter = 30

# Alte % Formatierung (nicht mehr empfohlen)
alte_art = "Name: %s, Alter: %d Jahre" % (name, alter)
print(f"Alte Art: {alte_art}")

# Besser: f-String verwenden
neue_art = f"Name: {name}, Alter: {alter} Jahre"
print(f"Neue Art: {neue_art}")
```
## 3.6 Debugging mit f-Strings

```python
# f-Strings für Debugging (sehr nützlich!)
x = 10
y = 20
ergebnis = x * y + 5

# Debug-Information anzeigen
print(f"{x=}")  # Zeigt: x=10
print(f"{y=}")  # Zeigt: y=20
print(f"{ergebnis=}")  # Zeigt: ergebnis=205

# Bei Berechnungen
print(f"{x * y=}")  # Zeigt: x * y=200
print(f"{x + y=}")  # Zeigt: x + y=30

# Mit Formatierung
pi = 3.14159
print(f"{pi=:.2f}")  # Zeigt: pi=3.14
```

```python
# Debugging einer Funktion
def berechne_rabatt(preis, rabatt_prozent):
    rabatt_betrag = preis * (rabatt_prozent / 100)
    endpreis = preis - rabatt_betrag
    
    # Debug-Ausgaben
    print(f"Debug: {preis=:.2f}")
    print(f"Debug: {rabatt_prozent=}%")
    print(f"Debug: {rabatt_betrag=:.2f}")
    print(f"Debug: {endpreis=:.2f}")
    
    return endpreis

# Funktion testen
endpreis = berechne_rabatt(100, 15)
print(f"Endpreis: {endpreis:.2f} Euro")
```
## 3.7 Tipps und Tricks

```python
# Mehrzeilige f-Strings
name = "Anna"
alter = 25
beruf = "Lehrerin"

# Übersichtliche Formatierung
info = f"""
Persönliche Daten:
------------------
Name:  {name}
Alter: {alter} Jahre
Beruf: {beruf}
"""
print(info)

# Bedingte Formatierung
anzahl_nachrichten = 3

if anzahl_nachrichten == 0:
    nachricht = "Keine neuen Nachrichten"
elif anzahl_nachrichten == 1:
    nachricht = "1 neue Nachricht"
else:
    nachricht = f"{anzahl_nachrichten} neue Nachrichten"

print(f"Status: {nachricht}")
```

```python
# Listen in f-Strings
hobbys = ["Lesen", "Sport", "Musik"]
noten = [1.3, 2.1, 1.7, 2.4]

# Listen direkt anzeigen
print(f"Hobbys: {hobbys}")
print(f"Noten: {noten}")

# Listen schöner formatieren
hobbys_text = ", ".join(hobbys)
noten_durchschnitt = sum(noten) / len(noten)

print(f"Hobbys: {hobbys_text}")
print(f"Notendurchschnitt: {noten_durchschnitt:.2f}")
```
