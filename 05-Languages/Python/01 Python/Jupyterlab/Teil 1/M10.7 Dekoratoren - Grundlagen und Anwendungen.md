Dekoratoren sind eine elegante Möglichkeit, die Funktionalität von Funktionen zu erweitern, ohne deren Code zu ändern. Sie nutzen alle bisher gelernten Konzepte: Higher-order Functions, Closures und Scope.

## 7.1 Einfache Dekoratoren

Ein Dekorator ist eine Funktion, die eine andere Funktion als Parameter nimmt und eine erweiterte Version zurückgibt.

### Grundlagen: Dekorator ohne @-Syntax

```python
def einfacher_dekorator(func):
    """Ein einfacher Dekorator, der Funktionsaufrufe protokolliert."""
    
    def wrapper(*args, **kwargs):
        print(f"🔧 Rufe {func.__name__} auf mit args={args}, kwargs={kwargs}")
        result = func(*args, **kwargs)
        print(f"✅ {func.__name__} fertig, Ergebnis: {result}")
        return result
    
    return wrapper

# Funktion definieren
def addiere(a, b):
    """Addiert zwei Zahlen."""
    return a + b

# Dekorator manuell anwenden
dekorierte_addiere = einfacher_dekorator(addiere)

print("=== Dekorator ohne @-Syntax ===")
print(f"Original: {addiere(3, 4)}")
print(f"Dekoriert: {dekorierte_addiere(3, 4)}")
```

### Die @-Syntax verstehen

```python
def protokolliere(func):
    """Dekorator der Funktionsaufrufe protokolliert."""
    
    def wrapper(*args, **kwargs):
        print(f"📞 Starte {func.__name__}")
        result = func(*args, **kwargs)
        print(f"🏁 Beende {func.__name__}")
        return result
    
    return wrapper

# @-Syntax ist syntactic sugar für: funktion = protokolliere(funktion)
@protokolliere
def multipliziere(x, y):
    """Multipliziert zwei Zahlen."""
    print(f"   Berechne {x} * {y}")
    return x * y

@protokolliere
def sage_hallo(name):
    """Begrüßt eine Person."""
    print(f"   Erstelle Begrüßung für {name}")
    return f"Hallo, {name}!"

print("\n=== @-Syntax ===")
print(f"Ergebnis: {multipliziere(4, 5)}")
print(f"Ergebnis: {sage_hallo('Anna')}")
```

```python
import functools

def besserer_dekorator(func):

    @functools.wraps(func)  # Erhält __name__, __doc__, etc.

@besserer_dekorator
def dokumentierte_funktion(x, y):

    Args:

    return x + y
```

### Mehrere Dekoratoren verketten

```python
def zeitmessung(func):
    """Misst die Ausführungszeit einer Funktion."""
    import time
    
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        ende = time.time()
        print(f"⏱️ {func.__name__} brauchte {ende - start:.4f} Sekunden")
        return result
    
    return wrapper

def eingabe_validierung(func):
    """Validiert dass alle Argumente Zahlen sind."""
    
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        # Alle args müssen Zahlen sein
        for arg in args:
            if not isinstance(arg, (int, float)):
                raise TypeError(f"Argument {arg} ist keine Zahl")
        
        # Alle kwarg-Werte müssen Zahlen sein
        for key, value in kwargs.items():
            if not isinstance(value, (int, float)):
                raise TypeError(f"Parameter {key}={value} ist keine Zahl")
        
        return func(*args, **kwargs)
    
    return wrapper

# Mehrere Dekoratoren anwenden (von unten nach oben)
@zeitmessung
@eingabe_validierung
@protokolliere
def komplexe_berechnung(x, y, z):
    """Eine rechenintensive Funktion."""
    import time
    time.sleep(0.1)  # Simuliert komplexe Berechnung
    return x * y + z

print("\n=== Verkettete Dekoratoren ===")
try:
    print(f"Ergebnis: {komplexe_berechnung(2, 3, 4)}")
    print(f"Mit Fehler: {komplexe_berechnung(2, '3', 4)}")
except TypeError as e:
    print(f"Validierungsfehler: {e}")
```

## 7.2 Dekoratoren mit Parametern

Parametrisierte Dekoratoren sind Funktionen, die Dekoratoren zurückgeben.

### Einfacher parametrisierter Dekorator

```python
def wiederhole(anzahl):
    """Dekorator-Factory: Wiederholt eine Funktion n-mal."""
    
    def dekorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            ergebnisse = []
            for i in range(anzahl):
                print(f"🔄 Durchlauf {i+1}/{anzahl}")
                ergebnis = func(*args, **kwargs)
                ergebnisse.append(ergebnis)
            return ergebnisse
        
        return wrapper
    
    return dekorator

@wiederhole(3)
def würfeln():
    """Simuliert einen Würfelwurf."""
    import random
    wurf = random.randint(1, 6)
    print(f"   🎲 Gewürfelt: {wurf}")
    return wurf

print("=== Parametrisierter Dekorator ===")
alle_würfe = würfeln()
print(f"Alle Würfe: {alle_würfe}")
```

### Flexibler Retry-Dekorator

```python
def retry(max_versuche=3, ausnahmen=(Exception,), delay=0):
    """
    Wiederholt eine Funktion bei Fehlern.
    
    Args:
        max_versuche: Maximale Anzahl Versuche
        ausnahmen: Tuple von Exception-Typen die einen Retry auslösen
        delay: Wartezeit zwischen Versuchen in Sekunden
    """
    
    def dekorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            import time
            import random
            
            for versuch in range(max_versuche):
                try:
                    print(f"🔄 Versuch {versuch + 1}/{max_versuche}: {func.__name__}")
                    result = func(*args, **kwargs)
                    print(f"✅ Erfolgreich nach {versuch + 1} Versuch(en)")
                    return result
                
                except ausnahmen as e:
                    print(f"❌ Versuch {versuch + 1} fehlgeschlagen: {e}")
                    
                    if versuch < max_versuche - 1:  # Nicht beim letzten Versuch warten
                        if delay > 0:
                            print(f"⏳ Warte {delay} Sekunden...")
                            time.sleep(delay)
                    else:
                        print(f"💥 Alle {max_versuche} Versuche fehlgeschlagen")
                        raise
            
        return wrapper
    
    return dekorator

@retry(max_versuche=4, ausnahmen=(ValueError, ConnectionError), delay=0.5)
def unzuverlässige_funktion(erfolgsrate=0.3):
    """Simuliert eine unzuverlässige Netzwerk-Operation."""
    import random
    
    if random.random() < erfolgsrate:
        return "🎉 Erfolg!"
    else:
        # Zufälligen Fehler werfen
        fehler_typ = random.choice([ValueError, ConnectionError])
        raise fehler_typ("Temporärer Fehler aufgetreten")

print("\n=== Retry-Dekorator ===")
try:
    ergebnis = unzuverlässige_funktion(0.4)  # 40% Erfolgsrate
    print(f"Endergebnis: {ergebnis}")
except Exception as e:
    print(f"Endgültiger Fehler: {e}")
```

### Konfigurierbarer Cache-Dekorator

```python
def cache(max_size=128, ttl_seconds=None):
    """
    Cache-Dekorator mit konfigurierbarer Größe und TTL.
    
    Args:
        max_size: Maximale Anzahl gecachter Ergebnisse
        ttl_seconds: Time-to-live in Sekunden (None = kein Ablauf)
    """
    
    def dekorator(func):
        import time
        
        # Cache-Storage als Closure-Variable
        cache_storage = {}
        access_order = []  # Für LRU (Least Recently Used)
        
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            # Cache-Key erstellen
            key = str(args) + str(sorted(kwargs.items()))
            current_time = time.time()
            
            # Cache-Hit prüfen
            if key in cache_storage:
                entry = cache_storage[key]
                
                # TTL prüfen
                if ttl_seconds is None or (current_time - entry['time']) < ttl_seconds:
                    # Cache-Hit! Access-Order aktualisieren
                    access_order.remove(key)
                    access_order.append(key)
                    print(f"💾 Cache-Hit für {func.__name__}{args}")
                    return entry['value']
                else:
                    # Abgelaufen
                    del cache_storage[key]
                    access_order.remove(key)
                    print(f"⏰ Cache-Entry abgelaufen für {func.__name__}{args}")
            
            # Cache-Miss: Funktion ausführen
            print(f"🔄 Cache-Miss, berechne {func.__name__}{args}")
            result = func(*args, **kwargs)
            
            # Im Cache speichern
            if len(cache_storage) >= max_size:
                # LRU Eviction: Ältesten Eintrag entfernen
                oldest_key = access_order.pop(0)
                del cache_storage[oldest_key]
                print(f"🗑️ Cache-Eviction: {oldest_key}")
            
            cache_storage[key] = {
                'value': result,
                'time': current_time
            }
            access_order.append(key)
            
            return result
        
        # Cache-Verwaltungsfunktionen hinzufügen
        def cache_info():
            return {
                'size': len(cache_storage),
                'max_size': max_size,
                'ttl': ttl_seconds,
                'keys': list(cache_storage.keys())
            }
        
        def cache_clear():
            cache_storage.clear()
            access_order.clear()
            print("🧹 Cache geleert")
        
        wrapper.cache_info = cache_info
        wrapper.cache_clear = cache_clear
        
        return wrapper
    
    return dekorator

@cache(max_size=3, ttl_seconds=5)
def fibonacci(n):
    """Berechnet Fibonacci-Zahl (ineffizient ohne Cache)."""
    if n <= 1:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

@cache(max_size=5, ttl_seconds=None)  # Kein TTL
def schwere_berechnung(x, y):
    """Simuliert eine rechenintensive Operation."""
    import time
    time.sleep(1)  # Simuliert Berechnungszeit
    return x ** y

print("\n=== Cache-Dekorator ===")
print(f"Fibonacci(10): {fibonacci(10)}")
print(f"Fibonacci(10) (cached): {fibonacci(10)}")
print(f"Cache-Info: {fibonacci.cache_info()}")

print(f"\nSchwere Berechnung 2^10: {schwere_berechnung(2, 10)}")
print(f"Schwere Berechnung 2^10 (cached): {schwere_berechnung(2, 10)}")
```

### Zeitmessung und Performance-Monitoring

```python
def performance_monitor(include_args=False, include_memory=False):
    """Umfassender Performance-Monitor."""
    
    def dekorator(func):
        import time
        import sys
        
        # Statistiken sammeln
        stats = {
            'calls': 0,
            'total_time': 0,
            'min_time': float('inf'),
            'max_time': 0,
            'errors': 0
        }
        
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            stats['calls'] += 1
            
            # Memory vor Aufruf (optional)
            if include_memory:
                memory_before = sys.getsizeof(locals())
            
            start_time = time.time()
            
            try:
                result = func(*args, **kwargs)
                
                # Zeitmessung
                execution_time = time.time() - start_time
                stats['total_time'] += execution_time
                stats['min_time'] = min(stats['min_time'], execution_time)
                stats['max_time'] = max(stats['max_time'], execution_time)
                
                # Ausgabe
                avg_time = stats['total_time'] / stats['calls']
                print(f"⏱️ {func.__name__}: {execution_time:.4f}s (Ø {avg_time:.4f}s)")
                
                if include_args and (args or kwargs):
                    print(f"   📝 Args: {args}, Kwargs: {kwargs}")
                
                if include_memory:
                    memory_after = sys.getsizeof(locals())
                    print(f"   💾 Memory: {memory_after - memory_before} bytes")
                
                return result
                
            except Exception as e:
                stats['errors'] += 1
                execution_time = time.time() - start_time
                print(f"❌ {func.__name__} Error nach {execution_time:.4f}s: {e}")
                raise
        
        # Statistik-Funktionen hinzufügen
        def get_stats():
            return stats.copy()
        
        def reset_stats():
            stats.update({
                'calls': 0,
                'total_time': 0,
                'min_time': float('inf'),
                'max_time': 0,
                'errors': 0
            })
        
        wrapper.get_stats = get_stats
        wrapper.reset_stats = reset_stats
        
        return wrapper
    
    return dekorator

@performance_monitor(include_args=True, include_memory=True)
def aufwendige_liste_operation(n):
    """Erstellt und verarbeitet eine große Liste."""
    import time
    
    # Liste erstellen
    große_liste = list(range(n))
    
    # Simuliere komplexe Verarbeitung
    result = []
    for i in große_liste:
        if i % 2 == 0:
            result.append(i ** 2)
        time.sleep(0.0001)  # Kleine Verzögerung
    
    return len(result)

# Performance-Tests
print("=== Performance-Monitoring ===")
for size in [100, 500, 1000]:
    aufwendige_liste_operation(size)

print(f"\nGesamtstatistiken: {aufwendige_liste_operation.get_stats()}")
```

### Eingabevalidierung automatisieren

```python
def validate_types(**type_hints):
    """
    Validiert Funktionsparameter basierend auf Typ-Hints.
    
    Usage:
        @validate_types(x=int, y=int, name=str)
        def func(x, y, name): ...
    """
    
    def dekorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            import inspect
            
            # Funktionssignatur analysieren
            sig = inspect.signature(func)
            bound_args = sig.bind(*args, **kwargs)
            bound_args.apply_defaults()
            
            # Typ-Validierung
            for param_name, expected_type in type_hints.items():
                if param_name in bound_args.arguments:
                    value = bound_args.arguments[param_name]
                    
                    if not isinstance(value, expected_type):
                        raise TypeError(
                            f"Parameter '{param_name}' muss {expected_type.__name__} sein, "
                            f"ist aber {type(value).__name__}: {value}"
                        )
            
            print(f"✅ Validierung erfolgreich für {func.__name__}")
            return func(*args, **kwargs)
        
        return wrapper
    
    return dekorator

def validate_range(**range_hints):
    """
    Validiert numerische Parameter gegen Min/Max-Werte.
    
    Usage:
        @validate_range(x=(0, 100), y=(1, None))  # None = keine Obergrenze
        def func(x, y): ...
    """
    
    def dekorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            import inspect
            
            sig = inspect.signature(func)
            bound_args = sig.bind(*args, **kwargs)
            bound_args.apply_defaults()
            
            for param_name, (min_val, max_val) in range_hints.items():
                if param_name in bound_args.arguments:
                    value = bound_args.arguments[param_name]
                    
                    # Min-Wert prüfen
                    if min_val is not None and value < min_val:
                        raise ValueError(
                            f"Parameter '{param_name}' ({value}) muss >= {min_val} sein"
                        )
                    
                    # Max-Wert prüfen
                    if max_val is not None and value > max_val:
                        raise ValueError(
                            f"Parameter '{param_name}' ({value}) muss <= {max_val} sein"
                        )
            
            return func(*args, **kwargs)
        
        return wrapper
    
    return dekorator

# Validierte Funktionen
@validate_types(name=str, alter=int, gehalt=float)
@validate_range(alter=(0, 150), gehalt=(0, None))
def erstelle_mitarbeiter(name, alter, gehalt=50000.0):
    """Erstellt einen Mitarbeiter mit Validierung."""
    return {
        "name": name,
        "alter": alter,
        "gehalt": gehalt,
        "id": hash(name) % 10000
    }

print("\n=== Eingabe-Validierung ===")

# Gültige Eingaben
try:
    mitarbeiter1 = erstelle_mitarbeiter("Anna", 28, 60000.0)
    print(f"Mitarbeiter erstellt: {mitarbeiter1}")
except (TypeError, ValueError) as e:
    print(f"Fehler: {e}")

# Ungültige Eingaben

    ("Charlie", -5, 50000),  # Negatives Alter
    ("Diana", 200, 50000),  # Zu hohes Alter
    (123, 30, 50000),  # Name nicht String
]

for name, alter, gehalt in test_cases:

        print(f"✅ {name} akzeptiert")
    except (TypeError, ValueError) as e:
        print(f"❌ {name}: {e}")
```

### Logging und Debugging

```python
def debug_logger(log_level="INFO", include_traceback=False):
    """
    Erweiterte Logging-Funktionalität.
    
    Args:
        log_level: DEBUG, INFO, WARNING, ERROR
        include_traceback: Ob bei Fehlern Traceback ausgegeben werden soll
    """
    
    def dekorator(func):
        import datetime
        import traceback
        
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            
            # Funktion-Start loggen
            if log_level in ["DEBUG"]:
                print(f"[{timestamp}] DEBUG: Starte {func.__name__}")
                print(f"[{timestamp}] DEBUG: Args: {args}")
                print(f"[{timestamp}] DEBUG: Kwargs: {kwargs}")
            elif log_level in ["DEBUG", "INFO"]:
                print(f"[{timestamp}] INFO: Ausführung {func.__name__}")
            
            try:
                # Funktion ausführen
                result = func(*args, **kwargs)
                
                # Erfolg loggen
                if log_level in ["DEBUG"]:
                    print(f"[{timestamp}] DEBUG: {func.__name__} erfolgreich, Ergebnis: {result}")
                elif log_level in ["DEBUG", "INFO"]:
                    print(f"[{timestamp}] INFO: {func.__name__} abgeschlossen")
                
                return result
                
            except Exception as e:
                # Fehler loggen
                print(f"[{timestamp}] ERROR: {func.__name__} fehlgeschlagen: {e}")
                
                if include_traceback:
                    print(f"[{timestamp}] ERROR: Traceback:")
                    traceback.print_exc()
                
                raise  # Exception weiterwerfen
        
        return wrapper
    
    return dekorator

@debug_logger(log_level="DEBUG", include_traceback=True)
def division(a, b):
    """Teilt zwei Zahlen mit ausführlichem Logging."""
    if b == 0:
        raise ZeroDivisionError("Division durch Null nicht erlaubt")
    
    result = a / b
    print(f"   Berechnung: {a} / {b} = {result}")
    return result

def datei_verarbeitung(dateiname):

    return f"Datei {dateiname} erfolgreich verarbeitet"
```
