## 2.1 *args - Variable Positionsargumente

Mit *args können Funktionen eine beliebige Anzahl von Positionsargumenten akzeptieren.

### Grundlagen von *args

```python
def addiere_zahlen(*args):
    """Addiert beliebig viele Zahlen."""
    print(f"Erhalten: {args}")
    print(f"Typ von args: {type(args)}")
    
    summe = sum(args)
    return summe

# Tests mit verschiedenen Anzahlen von Argumenten
print(addiere_zahlen(1, 2, 3))
print(addiere_zahlen(10, 20))
print(addiere_zahlen(5))
print(addiere_zahlen())  # Keine Argumente
```

### Praktisches Beispiel: Statistiken berechnen

```python
def berechne_statistiken(*zahlen):
    """Berechnet verschiedene Statistiken für eine Zahlenreihe."""
    if not zahlen:
        return {"fehler": "Keine Zahlen übergeben"}
    
    return {
        "anzahl": len(zahlen),
        "summe": sum(zahlen),
        "durchschnitt": sum(zahlen) / len(zahlen),
        "minimum": min(zahlen),
        "maximum": max(zahlen)
    }

# Tests
print(berechne_statistiken(1, 2, 3, 4, 5))
print(berechne_statistiken(10, 5, 8, 12, 3, 7))
print(berechne_statistiken(42))
```

### Kombination von normalen Parametern und *args

```python
def protokolliere_nachrichten(log_level, *nachrichten):
    """Protokolliert mehrere Nachrichten mit einem bestimmten Level."""
    zeitstempel = "2024-01-15 10:30:00"
    
    print(f"=== LOG LEVEL: {log_level.upper()} ===")
    for i, nachricht in enumerate(nachrichten, 1):
        print(f"[{zeitstempel}] {i}. {nachricht}")
    print()

# Tests
protokolliere_nachrichten("info", "Anwendung gestartet")
protokolliere_nachrichten("error", "Verbindung fehlgeschlagen", "Retry in 5 Sekunden")
protokolliere_nachrichten("debug", "Variable x = 42", "Schleife betreten", "Funktion beendet")
```

### Liste als Argumente mit * entpacken

```python
def finde_maximum(*zahlen):
    """Findet das Maximum von beliebig vielen Zahlen."""
    if not zahlen:
        return None
    return max(zahlen)

# Liste als einzelne Argumente übergeben
meine_zahlen = [15, 3, 9, 27, 8]
print(f"Liste: {meine_zahlen}")
print(f"Maximum: {finde_maximum(*meine_zahlen)}")

# Direkte Übergabe
print(f"Direkt: {finde_maximum(15, 3, 9, 27, 8)}")
```

## 2.2 `**kwargs` - Variable Schlüsselwortargumente

Mit **kwargs können Funktionen beliebige benannte Parameter akzeptieren.

### Grundlagen von `**kwargs`

```python
def zeige_informationen(**kwargs):
    """Zeigt beliebige Informationen an."""
    print(f"Erhalten: {kwargs}")
    print(f"Typ von kwargs: {type(kwargs)}")
    
    print("Informationen:")
    for schlüssel, wert in kwargs.items():
        print(f"  {schlüssel}: {wert}")
    print()

# Tests
zeige_informationen(name="Anna", alter=25, stadt="Berlin")
zeige_informationen(produkt="Laptop", preis=999.99, verfügbar=True)
zeige_informationen(titel="Python Kurs")
```

### Praktisches Beispiel: Flexibles Benutzerprofil

```python
def erstelle_benutzerprofil(name, **zusatz_infos):
    """Erstellt ein Benutzerprofil mit flexiblen Zusatzinformationen."""
    profil = {"name": name}
    
    # Zusatzinformationen hinzufügen
    profil.update(zusatz_infos)
    
    # Profil anzeigen
    print(f"=== Profil für {name} ===")
    for feld, wert in profil.items():
        if feld != "name":
            print(f"{feld.capitalize()}: {wert}")
    
    return profil

# Tests
profil1 = erstelle_benutzerprofil("Max", alter=30, beruf="Entwickler", hobby="Fotografie")
print()

profil2 = erstelle_benutzerprofil("Lisa", 
                                 alter=28, 
                                 stadt="München", 
                                 ausbildung="Informatik",
                                 sprachen=["Deutsch", "Englisch", "Spanisch"])
```

### Dictionary als kwargs übergeben

```python
def konfiguriere_server(**einstellungen):
    """Konfiguriert einen Server mit verschiedenen Einstellungen."""
    defaults = {
        "port": 8000,
        "host": "localhost",
        "debug": False,
        "workers": 1
    }
    
    # Defaults mit übergebenen Einstellungen überschreiben
```

## 2.3 Kombination von *args und **kwargs

Maximale Flexibilität durch Kombination beider Techniken.

### Vollständige Flexibilität

```python
def flexible_funktion(pflicht_param, optional_param="default", *args, **kwargs):
    """Demonstriert alle Parameter-Typen zusammen."""
    print(f"Pflichtparameter: {pflicht_param}")
    print(f"Optionaler Parameter: {optional_param}")
    print(f"Zusätzliche Positionsargs: {args}")
    print(f"Zusätzliche Keyword-Args: {kwargs}")
    print("-" * 40)

# Verschiedene Aufrufe
flexible_funktion("Wichtig")
flexible_funktion("Wichtig", "Geändert")
flexible_funktion("Wichtig", "Geändert", 1, 2, 3)
flexible_funktion("Wichtig", 1, 2, 3, name="Test", wert=42)
flexible_funktion("Wichtig", extra1="A", extra2="B")
```

### Praktisches Beispiel: API-Wrapper

```python
def api_request(endpoint, method="GET", *path_parts, **params):
    """Simuliert eine flexible API-Anfrage."""
    # URL zusammenbauen
    base_url = "https://api.example.com"
    
    # Pfad-Teile hinzufügen
    full_path = endpoint
    if path_parts:
        full_path += "/" + "/".join(str(part) for part in path_parts)
    
    # URL zusammensetzen
    url = f"{base_url}/{full_path}"
    
    # Parameter als Query-String
    if params:
        query_params = "&".join(f"{k}={v}" for k, v in params.items())
        url += f"?{query_params}"
    
    # Anfrage simulieren
    print(f"🌐 {method} {url}")
    return {"status": "success", "url": url}

# Verschiedene API-Aufrufe
api_request("users")
api_request("users", "POST", name="Anna", age=25)
api_request("users", "GET", 123, "posts", limit=10, offset=20)
api_request("products", category="electronics", price_min=100, price_max=500)
```

### Argumentweiterleitung an andere Funktionen

```python
def basis_berechnung(x, y, operation="add"):
    """Basis-Berechnungsfunktion."""
    if operation == "add":
        return x + y
    elif operation == "multiply":
        return x * y
    elif operation == "power":
        return x ** y
    else:
        return "Unbekannte Operation"

def erweiterte_berechnung(*args, **kwargs):
    """Erweitert die Basis-Berechnung um Logging."""
    print(f"🔢 Berechnung mit args={args}, kwargs={kwargs}")
    
    # Argumente an Basis-Funktion weiterleiten
    ergebnis = basis_berechnung(*args, **kwargs)
    
    print(f"✅ Ergebnis: {ergebnis}")
    return ergebnis

# Tests der Argumentweiterleitung
erweiterte_berechnung(5, 3)
erweiterte_berechnung(4, 2, operation="multiply")
erweiterte_berechnung(2, 8, operation="power")
```

### Decorator-Pattern Vorbereitung

```python
def protokolliere_aufrufe(func):
    """Einfacher Decorator, der Funktionsaufrufe protokolliert."""
    def wrapper(*args, **kwargs):
        print(f"📞 Rufe {func.__name__} auf")
        print(f"   Args: {args}")
        print(f"   Kwargs: {kwargs}")
        
        # Ursprüngliche Funktion aufrufen
        ergebnis = func(*args, **kwargs)
        
        print(f"   Ergebnis: {ergebnis}")
        return ergebnis
    
    return wrapper

# Decorator verwenden
@protokolliere_aufrufe
def multipliziere(a, b, runden=False):
    """Multipliziert zwei Zahlen."""
    ergebnis = a * b
    return round(ergebnis) if runden else ergebnis

# Test des dekorierten Funktions
multipliziere(3.7, 2.1)
multipliziere(3.7, 2.1, runden=True)
```

```python
    },

    }

print("Zusammenfassung der Konzepte:")
```
