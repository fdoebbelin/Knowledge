## 2.1 Auf Dictionary-Werte zugreifen

```python
# Dictionary erstellen
student = {
    "name": "Anna",
    "alter": 20,
    "noten": [1.3, 2.0, 1.7]
}

# Zugriff auf Werte
print(student["name"])          # Anna
print(student["alter"])         # 20

# Sicherer Zugriff mit get()
print(student.get("name"))      # Anna
print(student.get("email"))     # None (kein Fehler!)
print(student.get("email", "Keine Angabe"))  # "Keine Angabe"
```

## 2.2 Dictionary-Werte ändern und hinzufügen

```python
# Werte ändern
student["alter"] = 21
print(student["alter"])         # 21

# Neue Einträge hinzufügen
student["email"] = "anna@uni.de"
student["semester"] = 3

print(student)
# {'name': 'Anna', 'alter': 21, 'noten': [1.3, 2.0, 1.7], 'email': 'anna@uni.de', 'semester': 3}
```

## 2.3 Dictionary-Einträge löschen

```python
# Einzelnen Eintrag löschen
del student["semester"]

# Mit pop() löschen und Wert zurückgeben
email = student.pop("email")
print(email)                    # anna@uni.de

# Mit pop() und Standardwert
telefon = student.pop("telefon", "Nicht vorhanden")
print(telefon)                  # "Nicht vorhanden"

# Alles löschen
student.clear()
print(student)                  # {}
```

## 2.4 Durch Dictionaries iterieren

```python
person = {
    "name": "Max",
    "alter": 25,
    "stadt": "Berlin"
}

# Über Schlüssel iterieren
print("Schlüssel:")
for schluessel in person:
    print(schluessel)

# Über Werte iterieren
print("\nWerte:")
for wert in person.values():
    print(wert)

# Über Schlüssel-Wert-Paare iterieren
print("\nSchlüssel und Werte:")
for schluessel, wert in person.items():
    print(f"{schluessel}: {wert}")
```

## 2.5 Dictionary-Methoden im Überblick

```python
# Beispiel-Dictionary
zahlen = {"eins": 1, "zwei": 2, "drei": 3}

# Wichtige Methoden
print(zahlen.keys())            # dict_keys(['eins', 'zwei', 'drei'])
print(zahlen.values())          # dict_values([1, 2, 3])
print(zahlen.items())           # dict_items([('eins', 1), ('zwei', 2), ('drei', 3)])

# Prüfen ob Schlüssel existiert
print("eins" in zahlen)         # True
print("vier" in zahlen)         # False

# Dictionary kopieren
zahlen_kopie = zahlen.copy()
print(zahlen_kopie)             # {'eins': 1, 'zwei': 2, 'drei': 3}

# Dictionaries zusammenführen
mehr_zahlen = {"vier": 4, "fünf": 5}
zahlen.update(mehr_zahlen)
print(zahlen)                   # {'eins': 1, 'zwei': 2, 'drei': 3, 'vier': 4, 'fünf': 5}
```

## 2.6 Praktisches Beispiel: Notenverwaltung

```python
# Notenverwaltung für eine Klasse
klasse = {
    "Anna": [1.3, 2.0, 1.7],
    "Bob": [2.3, 1.7, 2.0],
    "Clara": [1.0, 1.3, 1.7]
}

# Neue Note hinzufügen
klasse["Anna"].append(1.0)
print(f"Anna's Noten: {klasse['Anna']}")

# Durchschnitt berechnen
for name, noten in klasse.items():
    durchschnitt = sum(noten) / len(noten)
    print(f"{name}: Durchschnitt {durchschnitt:.2f}")

# Neuen Schüler hinzufügen
klasse["David"] = [2.7, 2.3, 3.0]

# Schüler entfernen
entfernt = klasse.pop("Bob")
print(f"Bob entfernt mit Noten: {entfernt}")

print(f"Aktuelle Klasse: {list(klasse.keys())}")
```
