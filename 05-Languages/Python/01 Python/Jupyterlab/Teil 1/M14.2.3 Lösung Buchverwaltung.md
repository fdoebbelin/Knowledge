# Zelle 4.3: Vollständige Autor-Statistik
def erstelle_autor_uebersicht():
    """Erstellt Übersicht: Autor → Anzahl Bücher"""
    buecher = [
        ("Python Basics", "Max Müller", 250, "Programmierung"),
        ("Java Guide", "Anna Schmidt", 180, "Programmierung"),
        ("Geschichte Europas", "Peter Wagner", 420, "Geschichte"),
        ("Python Advanced", "Max Müller", 380, "Programmierung"),
        ("Weltkrieg", "Lisa Fischer", 290, "Geschichte"),
        ("HTML & CSS", "Tom Brown", 150, "Programmierung"),
        ("Antike Kulturen", "Peter Wagner", 310, "Geschichte")
    ]
    
    # Zuerst eindeutige Autoren sammeln
    alle_autoren = {autor for titel, autor, seitenzahl, genre in buecher}
    
    # Dictionary Comprehension: für jeden Autor die Anzahl seiner Bücher zählen
    # sum(1 for ... if ...) zählt wie oft ein Autor vorkommt
    autor_uebersicht = {
        autor: sum(1 for titel, a, seitenzahl, genre in buecher if a == autor) 
        for autor in alle_autoren
    }
    
    print("\n🧪 SCHRITT 4.3: Autor-Übersicht")
    print("=" * 35)
    print("Autor → Anzahl Bücher:")
    for autor, anzahl in sorted(autor_uebersicht.items()):
        plural = "Bücher" if anzahl > 1 else "Buch"
        print(f"  {autor}: {anzahl} {plural}")
    
    # Zusätzliche Statistiken
    produktivster = max(autor_uebersicht.items(), key=lambda x: x[1])
    print(f"\nProduktivster Autor: {produktivster[0]} mit {produktivster[1]} Büchern")
    
    # Autoren mit nur einem Buch
    einzelautoren = [autor for autor, anzahl in autor_uebersicht.items() if anzahl == 1]
    print(f"Autoren mit nur einem Buch: {len(einzelautoren)} ({', '.join(sorted(einzelautoren))})")
    
    print("✅ Vollständige Autor-Übersicht erstellt!")
    return autor_uebersicht

# Zelle 6: Comprehension-Patterns verstehen
def erklaere_comprehension_patterns():
    """Erklärt die verschiedenen Comprehension-Patterns"""
    buecher = [
        ("Python Basics", "Max Müller", 250, "Programmierung"),
        ("Java Guide", "Anna Schmidt", 180, "Programmierung"),
        ("Geschichte Europas", "Peter Wagner", 420, "Geschichte")
    ]
    
    print("\n🧠 COMPREHENSION-PATTERNS VERSTEHEN")
    print("=" * 45)
    
    # Pattern 1: Dictionary ohne Filter
    print("1️⃣ DICTIONARY COMPREHENSION (ohne Filter):")
    print("   Syntax: {key: value for tuple in liste}")
    titel_seiten = {titel: seitenzahl for titel, autor, seitenzahl, genre in buecher}
    print(f"   Beispiel: {titel_seiten}")
    
    # Pattern 2: Dictionary mit Filter
    print("\n2️⃣ DICTIONARY COMPREHENSION (mit Filter):")
    print("   Syntax: {key: value for tuple in liste if bedingung}")
    dicke_titel_genre = {titel: genre for titel, autor, seitenzahl, genre in buecher if seitenzahl > 200}
    print(f"   Beispiel: {dicke_titel_genre}")
    
    # Pattern 3: Set ohne Filter
    print("\n3️⃣ SET COMPREHENSION (ohne Filter):")
    print("   Syntax: {ausdruck for tuple in liste}")
    alle_autoren = {autor for titel, autor, seitenzahl, genre in buecher}
    print(f"   Beispiel: {alle_autoren}")
    
    # Pattern 4: Set mit Filter
    print("\n4️⃣ SET COMPREHENSION (mit Filter):")
    print("   Syntax: {ausdruck for tuple in liste if bedingung}")
    dicke_titel = {titel for titel, autor, seitenzahl, genre in buecher if seitenzahl > 200}
    print(f"   Beispiel: {dicke_titel}")
    
    # Pattern 5: Zählung mit sum()
    print("\n5️⃣ ZÄHLUNG MIT sum() und Generator Expression:")
    print("   Syntax: sum(1 for tuple in liste if bedingung)")
    anzahl_prog_buecher = sum(1 for titel, autor, seitenzahl, genre in buecher if genre == "Programmierung")
    print(f"   Beispiel: Programmierung-Bücher = {anzahl_prog_buecher}")
    
    print("\n💡 MERKSÄTZE:")
    print("   • Dictionary = Key-Value-Beziehungen mit {key: value}")
    print("   • Set = Eindeutige Werte mit {ausdruck}")
    print("   • Filter mit 'if' am Ende")
    print("   • Tupel-Entpackung: for a, b, c, d in liste")
    print("   • Zählung: sum(1 for ... if ...)")
    
    return titel_seiten, dicke_titel_genre, alle_autoren, dicke_titel

    buecher = ["Python Basics", "Java Guide", "Geschichte Europas"]
    
    # List Comprehension (alle Werte sofort im Speicher)
    titel_gross_list = [titel.upper() for titel in buecher]
    print(f"   List: {titel_gross_list} (alle Werte im Speicher)")
    
    # Generator Expression (lazy evaluation)
    titel_gross_gen = (titel.upper() for titel in buecher)
    print(f"   Generator: {titel_gross_gen} (Werte bei Bedarf)")
    print(f"   Erste Werte: {[next(titel_gross_gen), next(titel_gross_gen)]}")
    
    print("\n2️⃣ VERSCHACHTELTE COMPREHENSIONS:")
    print("   Für 2D-Datenstrukturen (Listen von Listen)")
    
    print("\n3️⃣ COMPREHENSIONS MIT FUNKTIONEN:")
    print("   Integration in eigene Funktionen und Module")
    
    print("\n💪 Du bist bereit für memory-effiziente Datenverarbeitung!")

# Ausblick zeigen
zeige_naechste_schritte()
