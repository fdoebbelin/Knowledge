## 1. Der große Unterschied: Veränderlich vs. Unveränderlich

```python
# Listen sind veränderlich (mutable)
meine_liste = [1, 2, 3]
print(f"Liste: {meine_liste}")

meine_liste[0] = 99     # Funktioniert!
meine_liste.append(4)   # Funktioniert!
print(f"Liste geändert: {meine_liste}")

# Tupel sind unveränderlich (immutable)
mein_tupel = (1, 2, 3)
print(f"Tupel: {mein_tupel}")

# Das funktioniert NICHT:
try:
    mein_tupel[0] = 99  # Fehler!
except TypeError as e:
    print(f"Tupel-Fehler: {e}")

try:
    mein_tupel.append(4)  # Fehler!
except AttributeError as e:
    print(f"Tupel hat keine append(): {e}")
```

## 3. Tupel als Dictionary-Keys (Listen gehen nicht!)

```python
# Tupel können als Dictionary-Keys verwendet werden
positionen = {}
positionen[(0, 0)] = "Start"
positionen[(5, 3)] = "Schatz"
positionen[(10, 8)] = "Ziel"

print("Spielfeld-Positionen:")
for (x, y), inhalt in positionen.items():
    print(f"  Position ({x}, {y}): {inhalt}")

# Listen können NICHT als Keys verwendet werden
try:
    fehlerhaft = {[1, 2]: "Wert"}  # Fehler!
except TypeError as e:
    print(f"\nListen als Keys - Fehler: {e}")

print("Warum? Nur 'hashbare' Objekte können Dictionary-Keys sein.")
print(f"Tupel ist hashbar: {hash((1, 2, 3))}")
try:
    hash([1, 2, 3])
except TypeError:
    print("Liste ist nicht hashbar!")
```

## 6. Funktionen mit Tupel-Rückgaben

```python
# Tupel für mehrere Rückgabewerte
def berechne_kreis(radius):
    """Berechnet Umfang und Fläche eines Kreises"""
    import math
    umfang = 2 * math.pi * radius
    flaeche = math.pi * radius * radius
    return umfang, flaeche  # Tupel-Rückgabe!

# Verwenden mit Unpacking
radius = 5
u, f = berechne_kreis(radius)
print(f"Kreis mit Radius {radius}:")
print(f"  Umfang: {u:.2f}")
print(f"  Fläche: {f:.2f}")

# Oder als ganzes Tupel
ergebnis = berechne_kreis(3)
print(f"Kreis-Ergebnis: {ergebnis}")
```

# Weitere Beispiele für Tupel-Rückgaben
def analysiere_text(text):
    """Analysiert Text und gibt Statistiken zurück"""
    woerter = text.split()
    zeichen = len(text)
    saetze = text.count('.') + text.count('!') + text.count('?')
    
    return len(woerter), zeichen, saetze  # Drei Werte als Tupel

def min_max_durchschnitt(zahlen):
    """Gibt Min, Max und Durchschnitt zurück"""
    if not zahlen:
        return None, None, None
    
    minimum = min(zahlen)
    maximum = max(zahlen)
    durchschnitt = sum(zahlen) / len(zahlen)
    
    return minimum, maximum, durchschnitt

# Tests
text = "Python ist toll! Es macht Spaß. Wirklich!"
woerter_anz, zeichen_anz, saetze_anz = analysiere_text(text)
print(f"\nText-Analyse:")
print(f"  Wörter: {woerter_anz}")
print(f"  Zeichen: {zeichen_anz}")
print(f"  Sätze: {saetze_anz}")

zahlen = [10, 5, 8, 12, 3, 15, 7]
min_val, max_val, avg_val = min_max_durchschnitt(zahlen)
print(f"\nZahlen-Analyse von {zahlen}:")
print(f"  Minimum: {min_val}")
print(f"  Maximum: {max_val}")
print(f"  Durchschnitt: {avg_val:.1f}")

### Kontaktbuch mit beiden Datentypen

```python
# Kombination: Listen UND Tupel sinnvoll verwenden
class Kontaktbuch:
    def __init__(self):
        # Liste für veränderliche Kontakt-Sammlung
        self.kontakte = []
    
    def kontakt_hinzufuegen(self, vorname, nachname, telefon):
        # Kontakt-Info als Tupel (unveränderliche Grunddaten)
        kontakt = (vorname, nachname, telefon)
        self.kontakte.append(kontakt)
        print(f"Kontakt hinzugefügt: {vorname} {nachname}")
    
    def alle_kontakte(self):
        print("\n📞 Kontaktbuch:")
        if not self.kontakte:
            print("  Keine Kontakte vorhanden")
            return
        
        for i, (vorname, nachname, telefon) in enumerate(self.kontakte, 1):
            print(f"  {i}. {vorname} {nachname}: {telefon}")
    
    def suche_kontakt(self, suchname):
        gefunden = []
        for kontakt in self.kontakte:
            vorname, nachname, telefon = kontakt
            if suchname.lower() in vorname.lower() or suchname.lower() in nachname.lower():
                gefunden.append(kontakt)
        return gefunden

# Demonstration
buch = Kontaktbuch()
buch.kontakt_hinzufuegen("Anna", "Müller", "0123-456789")
buch.kontakt_hinzufuegen("Bob", "Schmidt", "0123-987654")
buch.kontakt_hinzufuegen("Anna", "Wagner", "0123-555666")

buch.alle_kontakte()

# Suchen
print(f"\nSuche nach 'Anna':")
gefunden = buch.suche_kontakt("Anna")
for vorname, nachname, telefon in gefunden:
    print(f"  {vorname} {nachname}: {telefon}")
```

### Spiel mit Koordinaten

```python
# Spiel-Beispiel: Kombinierte Verwendung
class EinfachesSpiel:
    def __init__(self):
        # Tupel für feste Spielfeld-Grenzen
        self.spielfeld_groesse = (10, 10)
        
        # Liste für veränderliche Spieler-Positionen
        self.spieler = []
        
        # Tupel für feste Hindernisse
        self.hindernisse = {
            (2, 2): "Stein",
            (5, 5): "Baum", 
            (8, 3): "Wasser"
        }
    
    def spieler_hinzufuegen(self, name, start_position):
        # Position als Tupel (unveränderliche Koordinate)
        spieler_info = {
            "name": name,
            "position": start_position,
            "punkte": 0
        }
        self.spieler.append(spieler_info)
        print(f"Spieler {name} startet bei {start_position}")
    
    def spieler_bewegen(self, spieler_name, neue_position):
        # Neue Position als Tupel
        x, y = neue_position
        max_x, max_y = self.spielfeld_groesse
        
        # Grenzen prüfen
        if not (0 <= x < max_x and 0 <= y < max_y):
            print(f"❌ Position {neue_position} außerhalb des Spielfelds!")
            return
        
        # Hindernis prüfen
        if neue_position in self.hindernisse:
            hindernis = self.hindernisse[neue_position]
            print(f"❌ Bewegung blockiert durch {hindernis} bei {neue_position}")
            return
        
        # Spieler bewegen
        for spieler in self.spieler:
            if spieler["name"] == spieler_name:
                alte_position = spieler["position"]
                spieler["position"] = neue_position
                spieler["punkte"] += 1
                print(f"✅ {spieler_name}: {alte_position} → {neue_position}")
                return
        
        print(f"❌ Spieler {spieler_name} nicht gefunden")
    
    def zeige_spielstatus(self):
        print(f"\n🎮 Spielfeld: {self.spielfeld_groesse}")
        print("Hindernisse:")
        for pos, hindernis in self.hindernisse.items():
            print(f"  {pos}: {hindernis}")
        
        print("Spieler:")
        for spieler in self.spieler:
            print(f"  {spieler['name']}: {spieler['position']} ({spieler['punkte']} Punkte)")

# Spiel demonstrieren
spiel = EinfachesSpiel()
spiel.spieler_hinzufuegen("Anna", (0, 0))
spiel.spieler_hinzufuegen("Bob", (9, 9))

spiel.zeige_spielstatus()

# Bewegungen
spiel.spieler_bewegen("Anna", (1, 1))
spiel.spieler_bewegen("Anna", (2, 2))  # Hindernis!
spiel.spieler_bewegen("Anna", (1, 2))
spiel.spieler_bewegen("Bob", (8, 8))

spiel.zeige_spielstatus()
```
## Zusammenfassung
### 🎯 Entscheidungsregeln:

**Listen verwenden für:**
- ✅ Daten, die sich ändern können/sollen
- ✅ Unbekannte finale Größe
- ✅ Wenn Sortierung/Manipulation nötig
- ✅ Sammlung gleichartiger Objekte

**Tupel verwenden für:**
- ✅ Unveränderliche, feste Datenstrukturen
- ✅ Koordinaten und mathematische Werte
- ✅ Funktions-Rückgabewerte (mehrere Werte)
- ✅ Dictionary-Keys (hashbar erforderlich)
- ✅ Konfigurationswerte und Konstanten

### 📊 Performance-Fakten:
- **Tupel sind schneller** bei Erstellung und Zugriff
- **Tupel brauchen weniger Speicher**
- **Tupel sind thread-safe** (unveränderlich)
- **Listen haben mehr Methoden** für Manipulation

### 🔧 Praktische Tipps:
- **Dokumentieren Sie Ihre Entscheidung** (warum Liste/Tupel?)
- **Verwenden Sie Type Hints:** `list[str]` vs `tuple[int, int]`
- **Kombinieren Sie beide Typen** sinnvoll in komplexen Strukturen
- **Denken Sie an die Semantik:** Was drückt der Datentyp aus?
