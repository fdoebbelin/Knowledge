```python
# Einfache Wrapper-Funktionen
def bereinige_text(text, gross_klein="klein", leerzeichen=True):
    """Bereinigt Text nach verschiedenen Regeln"""
    
    if not text:  # Leeren Text abfangen
        return ""
    
    # Leerzeichen entfernen (optional)
    if leerzeichen:
        text = text.strip()
    
    # Groß-/Kleinschreibung
    if gross_klein == "klein":
        text = text.lower()
    elif gross_klein == "gross":
        text = text.upper()
    elif gross_klein == "title":
        text = text.title()
    
    return text

# Funktion testen
test_texte = ["  PYTHON ist toll  ", "", "   JavaScript   ", "HTML/CSS"]

print("Text-Bereinigung:")
for text in test_texte:
    bereinigt = bereinige_text(text, gross_klein="klein", leerzeichen=True)
    print(f"'{text}' → '{bereinigt}'")
```

```python
# Funktionen für häufige Aufgaben
def extrahiere_zahlen(text):
    """Holt alle Zahlen aus einem Text"""
    import re
    zahlen = re.findall(r'\d+', text)
    return [int(zahl) for zahl in zahlen]

def extrahiere_woerter(text, min_laenge=3):
    """Holt Wörter mit Mindestlänge"""
    import re
    woerter = re.findall(r'\b[A-Za-z]+\b', text)
    return [wort for wort in woerter if len(wort) >= min_laenge]

def ist_email(text):
    """Prüft ob Text eine E-Mail ist"""
    import re
    muster = r'^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$'
    return bool(re.match(muster, text.strip()))

# Funktionen testen

print("kein-email ist E-Mail:", ist_email("kein-email"))
```

```python
# Text-Pipeline - mehrere Schritte hintereinander
def text_pipeline(text, schritte):
    """Führt mehrere Text-Verarbeitungsschritte aus"""
    ergebnis = text
    
    for schritt in schritte:
        if schritt == "bereinigen":
            ergebnis = ergebnis.strip()
        elif schritt == "klein":
            ergebnis = ergebnis.lower()
        elif schritt == "gross":
            ergebnis = ergebnis.upper()
        elif schritt == "keine_zahlen":
            import re
            ergebnis = re.sub(r'\d+', '', ergebnis)
        elif schritt == "keine_sonderzeichen":
            import re
            ergebnis = re.sub(r'[^A-Za-z0-9\s]', '', ergebnis)
    
    return ergebnis

# Pipeline testen
roher_text = "  HALLO!!! Ich bin 25 Jahre alt... @#$  "
schritte = ["bereinigen", "klein", "keine_zahlen", "keine_sonderzeichen"]

ergebnis = text_pipeline(roher_text, schritte)
print(f"Vorher: '{roher_text}'")
print(f"Nachher: '{ergebnis}'")

# Verschiedene Pipelines ausprobieren
pipelines = {
    "Nur bereinigen": ["bereinigen"],
    "Bereinigen + klein": ["bereinigen", "klein"],
    "Komplett säubern": ["bereinigen", "klein", "keine_zahlen", "keine_sonderzeichen"]
}

for name, schritte in pipelines.items():
    result = text_pipeline(roher_text, schritte)
    print(f"{name}: '{result}'")
```

```python
# Schlüssel-Wert-Paare aus Text extrahieren
def parse_key_value_text(text):
    """Extrahiert Key-Value Paare aus Text wie 'Name: Max, Alter: 30'"""
    import re
    
    # Pattern für "Schlüssel: Wert" oder "Schlüssel = Wert"
    muster = r'([A-Za-z_]+)[:=]\s*([^,\n]+)'
    matches = re.findall(muster, text)
    
    # Als Dictionary zurückgeben
    daten = {}
    for schluessel, wert in matches:
        # Bereinigen
        schluessel = schluessel.strip().lower()
        wert = wert.strip()
        daten[schluessel] = wert
    
    return daten

# Testen
personen_text = """
Name: Max Mustermann
Alter: 30
Stadt: Berlin
E-Mail: max@beispiel.de
Beruf: Entwickler
"""

config_text = "host=localhost, port=3306, user=admin, password=secret123"

print("Personen-Daten:")
personen_daten = parse_key_value_text(personen_text)
for key, value in personen_daten.items():
    print(f"  {key}: {value}")

print("\nConfig-Daten:")
config_daten = parse_key_value_text(config_text)
for key, value in config_daten.items():
    print(f"  {key}: {value}")
```

```python
# Tabellen-ähnliche Daten extrahieren
def parse_tabelle(text, trennzeichen="|"):
    """Extrahiert Tabellendaten aus Text"""
    zeilen = text.strip().split('\n')
    
    # Leere Zeilen überspringen
    zeilen = [zeile.strip() for zeile in zeilen if zeile.strip()]
    
    if not zeilen:
        return [], []
    
    # Erste Zeile als Header
    header = [spalte.strip() for spalte in zeilen[0].split(trennzeichen)]
    
    # Rest als Daten
    daten = []
    for zeile in zeilen[1:]:
        if not zeile.startswith('-'):  # Trennzeilen überspringen
            werte = [wert.strip() for wert in zeile.split(trennzeichen)]
            # Nur wenn genug Werte vorhanden
            if len(werte) == len(header):
                daten.append(werte)
    
    return header, daten

# Testen
tabellen_text = """
Name | Alter | Stadt | Beruf
-----|-------|-------|-------
Max Mustermann | 30 | Berlin | Entwickler
Anna Schmidt | 25 | München | Designer
Tom Weber | 35 | Hamburg | Manager
"""

header, daten = parse_tabelle(tabellen_text)

print("Tabellen-Daten:")
print("Header:", header)
print("Daten:")
for zeile in daten:
    for i, wert in enumerate(zeile):
        print(f"  {header[i]}: {wert}")
    print()
```

```python
# Log-Dateien analysieren
def analysiere_log_zeile(zeile):
    """Analysiert eine einzelne Log-Zeile"""
    import re
    
    # Pattern für Standard-Log: "YYYY-MM-DD HH:MM:SS [LEVEL] Nachricht"
    log_pattern = r'(\d{4}-\d{2}-\d{2}\s+\d{2}:\d{2}:\d{2})\s+\[(\w+)\]\s+(.+)'
    
    match = re.match(log_pattern, zeile)
    if match:
        zeitstempel, level, nachricht = match.groups()
        return {
            'zeitstempel': zeitstempel,
            'level': level,
            'nachricht': nachricht,
            'gueltig': True
        }
    else:
        return {
            'zeitstempel': None,
            'level': None,
            'nachricht': zeile,
            'gueltig': False
        }

def analysiere_logs(log_text):
    """Analysiert komplette Log-Datei"""
    zeilen = log_text.strip().split('\n')
    
    log_eintraege = []
    level_counts = {}
    
    for zeile in zeilen:
        if zeile.strip():
            analyse = analysiere_log_zeile(zeile.strip())
            log_eintraege.append(analyse)
            
            # Statistiken sammeln
            if analyse['gueltig'] and analyse['level']:
                level = analyse['level']
                level_counts[level] = level_counts.get(level, 0) + 1
    
    return log_eintraege, level_counts

# Log-Analyse testen
log_text = """
2024-01-15 10:30:15 [INFO] Anwendung gestartet
2024-01-15 10:30:20 [DEBUG] Datenbank verbunden
2024-01-15 10:31:45 [WARNING] Speicher wird knapp
2024-01-15 10:32:10 [ERROR] Fehler beim Laden der Datei
2024-01-15 10:32:15 [INFO] Benutzer angemeldet: user123
Ungültige Zeile ohne Format
2024-01-15 10:33:00 [ERROR] Netzwerk-Timeout
"""

print("Log-Analyse:")
print(f"Gefundene Einträge: {len(eintraege)}")
```

```python
# String-Verbindung optimieren
import time

def teste_string_performance():
    """Vergleicht verschiedene Methoden der String-Verbindung"""
    
    woerter = ["Python", "ist", "eine", "tolle"] * 1000  # 4000 Wörter
    
    # Methode 1: + Operator (langsam)
    start = time.time()
    ergebnis1 = ""
    for wort in woerter:
        ergebnis1 += wort + " "
    zeit1 = time.time() - start
    
    # Methode 2: join() (schnell)
    start = time.time()
    ergebnis2 = " ".join(woerter)
    zeit2 = time.time() - start
    
    # Methode 3: f-String in Liste (mittel)
    start = time.time()
    teile = [f"{wort} " for wort in woerter]
    ergebnis3 = "".join(teile)
    zeit3 = time.time() - start
    
    print("String-Performance Test (4000 Wörter):")
    print(f"+ Operator:     {zeit1:.6f} Sekunden")
    print(f"join():         {zeit2:.6f} Sekunden")  
    print(f"List + join():  {zeit3:.6f} Sekunden")
    print(f"join() ist {zeit1/zeit2:.1f}x schneller als +")

teste_string_performance()
```

```python
# Generator für große Textdateien (Memory-schonend)
def verarbeite_text_chunks(text, chunk_groesse=1000):
    """Verarbeitet Text in kleinen Stücken"""
    for i in range(0, len(text), chunk_groesse):
        chunk = text[i:i + chunk_groesse]
        yield chunk

def zaehle_woerter_in_chunks(text):
    """Zählt Wörter in großen Texten Memory-schonend"""
    wort_count = 0
    
    for chunk in verarbeite_text_chunks(text, 500):
        woerter = chunk.split()
        wort_count += len(woerter)
    
    return wort_count

# Großen Text simulieren
```

```python
# Caching für teure Text-Operationen
from functools import lru_cache

@lru_cache(maxsize=128)
def teure_text_operation(text):
    """Simuliert teure Text-Verarbeitung (z.B. komplexe RegEx)"""
    import time
    import re
    
    # Simulierte Verzögerung
    time.sleep(0.01)  
    
    # Komplexe Verarbeitung
    woerter = re.findall(r'\b[A-Za-z]+\b', text.lower())
    statistiken = {
        'wort_anzahl': len(woerter),
        'eindeutige_woerter': len(set(woerter)),
        'durchschnittliche_laenge': sum(len(w) for w in woerter) / len(woerter) if woerter else 0
    }
    
    return statistiken

def demonstriere_caching():
    """Zeigt Vorteile von Caching"""
    test_texte = [
        "Python ist eine tolle Programmiersprache",
        "JavaScript ist auch sehr beliebt",
        "Python ist eine tolle Programmiersprache",  # Wiederholung
        "C++ ist sehr performant",
        "Python ist eine tolle Programmiersprache"   # Wiederholung
    ]
    
    print("Caching-Demonstration:")
    
    for i, text in enumerate(test_texte, 1):
        start = time.time()
        stats = teure_text_operation(text)
        dauer = time.time() - start
        
        print(f"Text {i}: {dauer:.4f}s - {stats['wort_anzahl']} Wörter")
        if dauer < 0.005:  # Sehr schnell = Cache-Hit
            print("  → Aus Cache geladen! 🚀")
    
    # Cache-Informationen
    cache_info = teure_text_operation.cache_info()
    print(f"\nCache-Statistiken:")
    print(f"  Hits: {cache_info.hits}")
    print(f"  Misses: {cache_info.misses}")
    print(f"  Effizienz: {cache_info.hits/(cache_info.hits+cache_info.misses)*100:.1f}%")

demonstriere_caching()
```

```python
# Kompletter Text-Analyzer als Klasse
class TextAnalyzer:
    """Sammlung nützlicher Text-Analyse-Tools"""
    
    def __init__(self):
        self.stopwoerter = {'der', 'die', 'das', 'und', 'oder', 'aber', 'ist', 'sind', 'ein', 'eine'}
    
    def analysiere_text(self, text):
        """Vollständige Text-Analyse"""
        import re
        
        # Grundlegende Statistiken
        zeichen_anzahl = len(text)
        zeichen_ohne_leerzeichen = len(text.replace(' ', ''))
        
        # Wörter extrahieren
        woerter = re.findall(r'\b[A-Za-z]+\b', text.lower())
        saetze = re.split(r'[.!?]+', text)
        saetze = [s.strip() for s in saetze if s.strip()]
        
        # Eindeutige Wörter (ohne Stopwörter)
        bedeutsame_woerter = [w for w in woerter if w not in self.stopwoerter]
        eindeutige_woerter = set(bedeutsame_woerter)
        
        # Wort-Häufigkeiten
        wort_haeufigkeit = {}
        for wort in bedeutsame_woerter:
            wort_haeufigkeit[wort] = wort_haeufigkeit.get(wort, 0) + 1
        
        # Top 5 häufigste Wörter
        top_woerter = sorted(wort_haeufigkeit.items(), key=lambda x: x[1], reverse=True)[:5]
        
        return {
            'zeichen_gesamt': zeichen_anzahl,
            'zeichen_ohne_leerzeichen': zeichen_ohne_leerzeichen,
            'woerter_gesamt': len(woerter),
            'saetze': len(saetze),
            'eindeutige_woerter': len(eindeutige_woerter),
            'durchschnittliche_wortlaenge': sum(len(w) for w in woerter) / len(woerter) if woerter else 0,
            'durchschnittliche_satzlaenge': len(woerter) / len(saetze) if saetze else 0,
            'top_woerter': top_woerter
        }
    
    def finde_emails(self, text):
        """Findet alle E-Mail-Adressen"""
        import re
        pattern = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b'
        return re.findall(pattern, text)
    
    def finde_telefonnummern(self, text):
        """Findet deutsche Telefonnummern"""
        import re
        pattern = r'(?:\+49|0)\s?\d{2,4}\s?\d{6,8}'
        return re.findall(pattern, text)
    
    def bereinige_fuer_analyse(self, text):
        """Bereinigt Text für bessere Analyse"""
        import re
        
        # Mehrfache Leerzeichen reduzieren
        text = re.sub(r'\s+', ' ', text)
        
        # Überflüssige Zeichen entfernen
        text = re.sub(r'[^\w\s.!?äöüÄÖÜß-]', '', text)
        
        # Trim
        text = text.strip()
        
        return text

# Text-Analyzer testen
analyzer = TextAnalyzer()

beispiel_text = """
Hallo! Mein Name ist Max Mustermann und ich bin Entwickler.
Sie können mich unter max@beispiel.de erreichen oder
telefonisch unter 030 12345678. Ich arbeite hauptsächlich
mit Python und JavaScript. Python ist eine tolle Sprache!
Python macht Spaß und ist sehr vielseitig.
"""

print("TEXT-ANALYZER DEMO")
print("=" * 30)

# Text bereinigen
bereinigter_text = analyzer.bereinige_fuer_analyse(beispiel_text)
print("Bereinigter Text:")
print(bereinigter_text)

# Vollanalyse
analyse = analyzer.analysiere_text(bereinigter_text)
print(f"\n📊 TEXT-STATISTIKEN:")
print(f"Zeichen gesamt: {analyse['zeichen_gesamt']}")
print(f"Wörter gesamt: {analyse['woerter_gesamt']}")
print(f"Sätze: {analyse['saetze']}")
print(f"Eindeutige Wörter: {analyse['eindeutige_woerter']}")
print(f"Ø Wortlänge: {analyse['durchschnittliche_wortlaenge']:.1f}")
print(f"Ø Satzlänge: {analyse['durchschnittliche_satzlaenge']:.1f} Wörter")

print(f"\n🔝 Häufigste Wörter:")
for wort, anzahl in analyse['top_woerter']:
    print(f"  '{wort}': {anzahl}x")

# Kontaktdaten finden
```

```python
# Nützliche Helfer-Funktionen für den Alltag
def sichere_dateiname(text, max_laenge=50):
    """Macht aus Text einen sicheren Dateinamen"""
    import re
    
    # Ungültige Zeichen ersetzen
    sicher = re.sub(r'[<>:"/\\|?*]', '_', text)
    
    # Mehrfache Unterstriche reduzieren
    sicher = re.sub(r'_+', '_', sicher)
    
    # Kürzen wenn zu lang
    if len(sicher) > max_laenge:
        sicher = sicher[:max_laenge-3] + "..."
    
    # Rand bereinigen
    sicher = sicher.strip('_. ')
    
    return sicher if sicher else "unnamed"

def formatiere_dateigröße(bytes_anzahl):
    """Formatiert Bytes in lesbare Größe"""
    einheiten = ['B', 'KB', 'MB', 'GB', 'TB']
    größe = float(bytes_anzahl)
    
    for einheit in einheiten:
        if größe < 1024.0:
            return f"{größe:.1f} {einheit}"
        größe /= 1024.0
    
    return f"{größe:.1f} PB"

def maskiere_persönliche_daten(text):
    """Maskiert E-Mails und Telefonnummern für Anonymisierung"""
    import re
    
    # E-Mails maskieren
    text = re.sub(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b', '[EMAIL]', text)
    
    # Telefonnummern maskieren
    text = re.sub(r'(?:\+49|0)\s?\d{2,4}\s?\d{6,8}', '[TELEFON]', text)
    
    return text

# Praktische Tools testen
```
