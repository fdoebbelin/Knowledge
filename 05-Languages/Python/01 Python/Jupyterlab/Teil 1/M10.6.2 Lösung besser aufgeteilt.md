```python
def demonstriere_scopes():
    # Enclosing scope (äußere Funktion)
    enclosing_variable = "Ich bin im umschließenden Scope"
    
    print(f"\n📍 In äußerer Funktion:")
    print(f"   Global: {global_variable}")
    print(f"   Enclosing: {enclosing_variable}")
    
    def inner_function():
        # Local scope (innere Funktion)
        local_variable = "Ich bin lokal"
        
        print(f"\n📍 In innerer Funktion:")
        print(f"   Global: {global_variable}")
        print(f"   Enclosing: {enclosing_variable}")
        print(f"   Local: {local_variable}")
        print(f"   Built-in: {type(local_variable)}")
        
        # Python sucht in dieser Reihenfolge:
        # 1. Local -> 2. Enclosing -> 3. Global -> 4. Built-in
    
    return inner_function

# Test der Scope-Hierarchie
inner_func = demonstriere_scopes()
inner_func()

print(f"\n🔍 Variable Resolution Order: Local → Enclosing → Global → Built-in")
```

```python
def erstelle_zaehler(start=0):
    """
    Erstellt einen konfigurierbaren Zähler mit verschiedenen Operationen.
    
    Args:
        start (int): Startwert des Zählers
        
    Returns:
        function: Zähler-Funktion mit verschiedenen Befehlen
    """
    aktueller_wert = start
    start_wert = start
    
    def zaehler_operation(befehl, schritt=1):
        """
        Führt verschiedene Zähler-Operationen aus.
        
        Args:
            befehl (str): 'hoch', 'runter', 'wert', 'reset'
            schritt (int): Schrittweite für hoch/runter
            
        Returns:
            int oder None: Aktueller Wert oder None bei Operationen
        """
        nonlocal aktueller_wert  # Zugriff auf umschließende Variable
        
        if befehl == 'hoch':
            aktueller_wert += schritt
            print(f"📈 +{schritt} → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'runter':
            aktueller_wert -= schritt
            print(f"📉 -{schritt} → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'wert':
            return aktueller_wert
            
        elif befehl == 'reset':
            aktueller_wert = start_wert
            print(f"🔄 Reset → {aktueller_wert}")
            return aktueller_wert
            
        else:
            print(f"❌ Unbekannter Befehl: {befehl}")
            print("   Verfügbare Befehle: 'hoch', 'runter', 'wert', 'reset'")
            return None
    
    return zaehler_operation
```

```python
def erstelle_bereichs_zaehler(start=0, minimum=None, maximum=None):
    """
    Erstellt einen Zähler mit Bereichsbegrenzung.
    
    Args:
        start (int): Startwert
        minimum (int): Untere Grenze (optional)
        maximum (int): Obere Grenze (optional)
        
    Returns:
        function: Bereichs-Zähler-Funktion
    """
    aktueller_wert = start
    start_wert = start
    
    def validiere_bereich(neuer_wert):
        """Hilfsfunktion zur Bereichsvalidierung."""
        if minimum is not None and neuer_wert < minimum:
            print(f"⚠️  Minimum {minimum} erreicht, bleibe bei {minimum}")
            return minimum
        if maximum is not None and neuer_wert > maximum:
            print(f"⚠️  Maximum {maximum} erreicht, bleibe bei {maximum}")
            return maximum
        return neuer_wert
    
    def bereichs_operation(befehl, schritt=1):
        nonlocal aktueller_wert
        
        if befehl == 'hoch':
            neuer_wert = aktueller_wert + schritt
            aktueller_wert = validiere_bereich(neuer_wert)
            print(f"📈 +{schritt} → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'runter':
            neuer_wert = aktueller_wert - schritt
            aktueller_wert = validiere_bereich(neuer_wert)
            print(f"📉 -{schritt} → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'wert':
            return aktueller_wert
            
        elif befehl == 'reset':
            aktueller_wert = validiere_bereich(start_wert)
            print(f"🔄 Reset → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'info':
            print(f"📋 Bereichs-Zähler Info:")
            print(f"   Aktuell: {aktueller_wert}")
            print(f"   Minimum: {minimum if minimum is not None else 'unbegrenzt'}")
            print(f"   Maximum: {maximum if maximum is not None else 'unbegrenzt'}")
            print(f"   Startwert: {start_wert}")
            return aktueller_wert
            
        else:
            print(f"❌ Unbekannter Befehl: {befehl}")
            return None
    
    return bereichs_operation

print("\n📊 BEREICHS-ZÄHLER TEST")
```

```python
def erstelle_zyklus_zaehler(start=0, minimum=0, maximum=9):
    """
    Erstellt einen zyklischen Zähler, der bei Grenzüberschreitung umspringt.
    
    Args:
        start (int): Startwert
        minimum (int): Untere Grenze
        maximum (int): Obere Grenze
        
    Returns:
        function: Zyklus-Zähler-Funktion
    """
    aktueller_wert = start
    start_wert = start
    
    def berechne_zyklus_wert(neuer_wert):
        """Berechnet zyklischen Wert zwischen minimum und maximum."""
        bereich = maximum - minimum + 1
        
        if neuer_wert > maximum:
            # Überlauf: springe zum Minimum
            ueberlauf = neuer_wert - maximum
            zyklus_wert = minimum + (ueberlauf - 1) % bereich
            print(f"🔄 Überlauf bei {maximum} → springe zu {zyklus_wert}")
            return zyklus_wert
            
        elif neuer_wert < minimum:
            # Unterlauf: springe zum Maximum
            unterlauf = minimum - neuer_wert
            zyklus_wert = maximum - (unterlauf - 1) % bereich
            print(f"🔄 Unterlauf bei {minimum} → springe zu {zyklus_wert}")
            return zyklus_wert
            
        return neuer_wert
    
    def zyklus_operation(befehl, schritt=1):
        nonlocal aktueller_wert
        
        if befehl == 'hoch':
            neuer_wert = aktueller_wert + schritt
            aktueller_wert = berechne_zyklus_wert(neuer_wert)
            print(f"📈 +{schritt} → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'runter':
            neuer_wert = aktueller_wert - schritt
            aktueller_wert = berechne_zyklus_wert(neuer_wert)
            print(f"📉 -{schritt} → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'wert':
            return aktueller_wert
            
        elif befehl == 'reset':
            aktueller_wert = start_wert
            print(f"🔄 Reset → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'info':
            print(f"📋 Zyklus-Zähler Info:")
            print(f"   Aktuell: {aktueller_wert}")
            print(f"   Bereich: {minimum} bis {maximum} (zyklisch)")
            print(f"   Startwert: {start_wert}")
            return aktueller_wert
            
        else:
            print(f"❌ Unbekannter Befehl: {befehl}")
            return None
    
    return zyklus_operation
```

```python
def erstelle_statistik_zaehler(start=0):
    """
    Erstellt einen Zähler mit erweiterten Statistiken.
    
    Args:
        start (int): Startwert
        
    Returns:
        function: Statistik-Zähler-Funktion
    """
    aktueller_wert = start
    start_wert = start
    
    # Zusätzliche Statistiken
    minimum_erreicht = start
    maximum_erreicht = start
    anzahl_operationen = 0
    verlauf = [start]  # Historie der letzten 10 Werte
    
    def statistik_operation(befehl, schritt=1):
        nonlocal aktueller_wert, minimum_erreicht, maximum_erreicht
        nonlocal anzahl_operationen, verlauf
        
        if befehl == 'hoch':
            aktueller_wert += schritt
            anzahl_operationen += 1
            maximum_erreicht = max(maximum_erreicht, aktueller_wert)
            verlauf.append(aktueller_wert)
            if len(verlauf) > 10:  # Nur letzte 10 Werte behalten
                verlauf.pop(0)
            print(f"📈 +{schritt} → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'runter':
            aktueller_wert -= schritt
            anzahl_operationen += 1
            minimum_erreicht = min(minimum_erreicht, aktueller_wert)
            verlauf.append(aktueller_wert)
            if len(verlauf) > 10:
                verlauf.pop(0)
            print(f"📉 -{schritt} → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'wert':
            return aktueller_wert
            
        elif befehl == 'reset':
            aktueller_wert = start_wert
            minimum_erreicht = start_wert
            maximum_erreicht = start_wert
            anzahl_operationen = 0
            verlauf = [start_wert]
            print(f"🔄 Reset → {aktueller_wert}")
            return aktueller_wert
            
        elif befehl == 'statistiken':
            durchschnitt = sum(verlauf) / len(verlauf) if verlauf else 0
            print(f"📊 STATISTIK-ZÄHLER AUSWERTUNG")
            print(f"   Aktueller Wert: {aktueller_wert}")
            print(f"   Startwert: {start_wert}")
            print(f"   Minimum erreicht: {minimum_erreicht}")
            print(f"   Maximum erreicht: {maximum_erreicht}")
            print(f"   Anzahl Operationen: {anzahl_operationen}")
            print(f"   Durchschnitt (letzte {len(verlauf)}): {durchschnitt:.1f}")
            print(f"   Verlauf: {verlauf}")
            return {
                'aktuell': aktueller_wert,
                'minimum': minimum_erreicht,
                'maximum': maximum_erreicht,
                'operationen': anzahl_operationen,
                'durchschnitt': durchschnitt,
                'verlauf': verlauf.copy()
            }
            
        else:
            print(f"❌ Unbekannter Befehl: {befehl}")
            print("   Verfügbare Befehle: 'hoch', 'runter', 'wert', 'reset', 'statistiken'")
            return None
    
    return statistik_operation

print("\n📈 STATISTIK-ZÄHLER TEST")
print("=" * 26)
```

## 🏭 Schritt 8: Zähler-Manager (Factory Pattern)

### 8.1 Zähler-Manager implementieren

```python
def erstelle_zaehler_manager():
    """
    Erstellt einen Manager für mehrere benannte Zähler.
    
    Returns:
        function: Manager-Funktion für Zähler-Verwaltung
    """
    zaehler_sammlung = {}  # Dictionary zur Speicherung der Zähler
    
    def manager_operation(befehl, name=None, **optionen):
        nonlocal zaehler_sammlung
        
        if befehl == 'erstelle':
            if name is None:
                print("❌ Name für Zähler erforderlich")
                return None
                
            zaehler_typ = optionen.get('typ', 'basis')
            start = optionen.get('start', 0)
            
            if zaehler_typ == 'basis':
                zaehler_sammlung[name] = erstelle_zaehler(start)
            elif zaehler_typ == 'bereich':
                minimum = optionen.get('minimum', None)
                maximum = optionen.get('maximum', None)
                zaehler_sammlung[name] = erstelle_bereichs_zaehler(start, minimum, maximum)
            elif zaehler_typ == 'zyklus':
                minimum = optionen.get('minimum', 0)
                maximum = optionen.get('maximum', 9)
                zaehler_sammlung[name] = erstelle_zyklus_zaehler(start, minimum, maximum)
            elif zaehler_typ == 'statistik':
                zaehler_sammlung[name] = erstelle_statistik_zaehler(start)
            else:
                print(f"❌ Unbekannter Zähler-Typ: {zaehler_typ}")
                return None
                
            print(f"✅ Zähler '{name}' ({zaehler_typ}) erstellt")
            return zaehler_sammlung[name]
            
        elif befehl == 'hole':
            if name not in zaehler_sammlung:
                print(f"❌ Zähler '{name}' nicht gefunden")
                return None
            return zaehler_sammlung[name]
            
        elif befehl == 'liste':
            if not zaehler_sammlung:
                print("📋 Keine Zähler vorhanden")
            else:
                print(f"📋 Verfügbare Zähler ({len(zaehler_sammlung)}):")
                for zaehler_name, zaehler_func in zaehler_sammlung.items():
                    aktueller_wert = zaehler_func('wert')
                    print(f"   • {zaehler_name}: {aktueller_wert}")
            return list(zaehler_sammlung.keys())
            
        elif befehl == 'entferne':
            if name in zaehler_sammlung:
                del zaehler_sammlung[name]
                print(f"🗑️  Zähler '{name}' entfernt")
                return True
            else:
                print(f"❌ Zähler '{name}' nicht gefunden")
                return False
                
        elif befehl == 'info':
            if name is None:
                print(f"📊 Manager Info: {len(zaehler_sammlung)} Zähler verwaltet")
                return len(zaehler_sammlung)
            elif name in zaehler_sammlung:
                zaehler = zaehler_sammlung[name]
                wert = zaehler('wert')
                print(f"📊 Zähler '{name}': {wert}")
                return wert
            else:
                print(f"❌ Zähler '{name}' nicht gefunden")
                return None
        else:
            print(f"❌ Unbekannter Manager-Befehl: {befehl}")
            print("   Verfügbare Befehle: 'erstelle', 'hole', 'liste', 'entferne', 'info'")
            return None
    
    return manager_operation

print("\n🏭 ZÄHLER-MANAGER TEST")
print("=" * 23)
```

```python
def erstelle_timer():
    """Erstelle einen Timer mit Start/Stop/Reset Funktionalität."""
    import time
    
    start_zeit = None
    gesamte_zeit = 0
    laeuft = False
    
    def timer_operation(befehl):
        nonlocal start_zeit, gesamte_zeit, laeuft
        
        if befehl == 'start':
            if not laeuft:
                start_zeit = time.time()
                laeuft = True
                print("⏱️ Timer gestartet")
            else:
                print("⚠️ Timer läuft bereits")
        
        elif befehl == 'stop':
            if laeuft:
                gesamte_zeit += time.time() - start_zeit
                laeuft = False
                print(f"⏹️ Timer gestoppt. Gesamtzeit: {gesamte_zeit:.2f}s")
            else:
                print("⚠️ Timer läuft nicht")
        
        elif befehl == 'zeit':
            if laeuft:
                aktuelle_zeit = gesamte_zeit + (time.time() - start_zeit)
                return aktuelle_zeit
            return gesamte_zeit
        
        elif befehl == 'reset':
            gesamte_zeit = 0
            start_zeit = None
            laeuft = False
            print("🔄 Timer zurückgesetzt")
    
    return timer_operation

# Timer-Test
```

### Übung B: Accumulator-Pattern

```python
def erstelle_accumulator(operation='+'):
    """Erstelle einen Accumulator für verschiedene Operationen."""
    wert = 0
    historie = []
    
    def accumulate(zahl):
        nonlocal wert
        
        if operation == '+':
            wert += zahl
        elif operation == '*':
            wert *= zahl
        elif operation == 'max':
            wert = max(wert, zahl)
        elif operation == 'min':
            wert = min(wert, zahl) if historie else zahl
        
        historie.append((zahl, wert))
        print(f"{operation} {zahl} → {wert}")
        return wert
    
    # Zusätzliche Funktionen als Attribute
    accumulate.wert = lambda: wert
    accumulate.historie = lambda: historie.copy()
    accumulate.reset = lambda: historie.clear() or setattr(accumulate, 'wert', lambda: 0)
    
    return accumulate

# Accumulator-Test

print("=" * 20)

add_acc(5)
add_acc(10)
add_acc(3)
print(f"Historie: {add_acc.historie()}")
```
