```python
# Zelle 1.1: Eine einfache Notiz in Datei schreiben
def schreibe_erste_notiz():
    """Schreibt eine erste Notiz in die Datei 'notizen.txt'"""
    notiz = "Heute habe ich gelernt, wie man Dateien in Python schreibt!"
    
    # Datei mit 'with'-Statement sicher erstellen und beschreiben
    # "w" = write (schreiben), encoding für deutsche Umlaute
    with open("notizen.txt", "w", encoding="utf-8") as datei:
        datei.write(notiz)
    
    print("📝 SCHRITT 1.1: Erste Notiz schreiben")
    print("=" * 40)
    print(f"✅ Datei 'notizen.txt' erstellt")
    print(f"✅ Notiz gespeichert")
    return f"Notiz gespeichert: {notiz}"

# Test der Funktion
print(schreibe_erste_notiz())
```

```python
# Zelle 3.1: Mehrere Notizen auf einmal speichern
def sammle_tages_notizen():
    """Sammelt und speichert mehrere Notizen des Tages"""
    tages_notizen = [
        "09:00 - Morgens Python-Kurs gestartet",
        "11:00 - Erste Dateien erfolgreich erstellt",  
        "14:00 - Pause gemacht und Kaffee getrunken",
        "16:00 - Notiz-System programmiert"
    ]
    
    # Alle Notizen in Datei schreiben, jede in eigener Zeile
    with open("tages_notizen.txt", "w", encoding="utf-8") as datei:
        for notiz in tages_notizen:
            # \n fügt Zeilenendezeichen hinzu
            datei.write(notiz + "\n")
    
    print("\n📝 SCHRITT 3.1: Tages-Notizen sammeln")
    print("=" * 42)
    print(f"✅ {len(tages_notizen)} Notizen in 'tages_notizen.txt' gespeichert")
    print("✅ Jede Notiz in eigener Zeile mit \\n")
    for i, notiz in enumerate(tages_notizen, 1):
        print(f"   {i}. {notiz}")
    return f"Tages-Notizen gespeichert: {len(tages_notizen)} Einträge"

# Test der Funktion
print(sammle_tages_notizen())
```

```python
# Zelle 3.2: Neue Notiz zu bestehenden Notizen hinzufügen
def fuege_notiz_hinzu():
    """Fügt eine neue Notiz zu den bestehenden Tages-Notizen hinzu"""
    neue_notiz = "18:00 - Aufgaben erfolgreich gelöst!"
    
    # "a" = append (anhängen) - überschreibt NICHT die bestehende Datei
    with open("tages_notizen.txt", "a", encoding="utf-8") as datei:

    print("\n📝 SCHRITT 3.2: Notiz anhängen")
    print("=" * 35)
    print("✅ Modus 'a' (append) verwendet - Datei wird NICHT überschrieben")
    print(f"✅ Neue Notiz hinzugefügt: {neue_notiz}")
    return f"Notiz hinzugefügt: {neue_notiz}"
```

```python
# Zelle 4.1: Notiz sicher speichern mit Fehlerbehandlung
def sichere_notiz_speichern():
    """Speichert eine Notiz mit Fehlerbehandlung"""
    notiz = "Das ist eine wichtige Notiz, die sicher gespeichert werden muss!"
    
    # try-except Block für sichere Datei-Operation
    try:
        with open("sichere_notiz.txt", "w", encoding="utf-8") as datei:
            datei.write(notiz)
        ergebnis = f"✅ Notiz sicher gespeichert: {notiz}"
    except Exception as fehler:
        ergebnis = f"❌ Fehler beim Speichern: {fehler}"
    
    print("\n🛡️ SCHRITT 4.1: Sichere Speicherung")
    print("=" * 40)
    print("✅ try-except Block implementiert")
    print("✅ Allgemeine Exception abgefangen")
    return ergebnis
```

```python
# Zelle 4.2: Notiz sicher lesen mit Fehlerbehandlung
def sichere_notiz_lesen():
    """Liest eine Notiz mit Fehlerbehandlung"""
    
    # Spezifische Fehlerbehandlung mit FileNotFoundError
    try:
        with open("sichere_notiz.txt", "r", encoding="utf-8") as datei:
            inhalt = datei.read()
        ergebnis = f"✅ Notiz gelesen: {inhalt}"

    print("=" * 37)
    print("✅ FileNotFoundError spezifisch abgefangen")
```

```python
def zeige_datei_uebersicht():
    """Zeigt eine Übersicht aller erstellten Dateien"""

        "tages_notizen.txt",

    print("\n" + "="*50)

    print("="*50)
    
    for datei in datei_liste:
        try:
            with open(datei, "r", encoding="utf-8") as f:
                inhalt = f.read()
                zeilen_anzahl = len(inhalt.split('\n')) if inhalt else 0
            status = f"✅ Existiert ({zeilen_anzahl} Zeilen)"
            
            # Ersten Teil des Inhalts anzeigen
            vorschau = inhalt[:50] + "..." if len(inhalt) > 50 else inhalt
            print(f"📄 {datei}: {status}")
            print(f"   Vorschau: {vorschau}")
            
        except FileNotFoundError:
            print(f"❌ {datei}: Datei nicht gefunden")
        except Exception as fehler:
            print(f"⚠️ {datei}: Fehler beim Lesen - {fehler}")
    
    print(f"\n📊 STATISTIK:")
    print(f"   Erfolgreich erstellte Dateien: {len([d for d in datei_liste if self.datei_existiert(d)])}")
    print(f"   Verwendete Techniken: with-Statement, encoding, try-except")
    print(f"   Modi verwendet: 'w' (write), 'r' (read), 'a' (append)")

def datei_existiert(self, datei):
    """Hilfsfunktion um zu prüfen ob Datei existiert"""
    try:
        with open(datei, "r", encoding="utf-8") as f:
            f.read()
        return True
    except FileNotFoundError:
        return False
```
