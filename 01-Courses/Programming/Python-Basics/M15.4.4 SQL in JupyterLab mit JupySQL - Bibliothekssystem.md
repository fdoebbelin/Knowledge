## 🏗️ Datenbankschema erstellen

```sql
-- Alte Tabellen löschen (für Neustart)
DROP TABLE IF EXISTS ausleihen;
DROP TABLE IF EXISTS buecher;
DROP TABLE IF EXISTS mitglieder;
DROP TABLE IF EXISTS autoren;
```

```sql
-- Tabelle "autoren" erstellen
CREATE TABLE
  autoren (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    vorname TEXT NOT NULL,
    nachname TEXT NOT NULL,
    geburtsjahr INTEGER,
    nationalitaet TEXT DEFAULT 'Deutschland'
  );
```

```sql
-- Tabelle "buecher" erstellen
CREATE TABLE
  buecher (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    titel TEXT NOT NULL,
    isbn TEXT UNIQUE,
    erscheinungsjahr INTEGER,
    seitenzahl INTEGER,
    kategorie TEXT DEFAULT 'Roman',
    autor_id INTEGER NOT NULL,
    verfuegbar INTEGER NOT NULL DEFAULT 1,
    FOREIGN KEY (autor_id) REFERENCES autoren (id)
  );
```

```sql
-- Tabelle "mitglieder" erstellen
CREATE TABLE
  mitglieder (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    vorname TEXT NOT NULL,
    nachname TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    telefon TEXT,
    anmeldedatum TEXT NOT NULL,
    aktiv INTEGER NOT NULL DEFAULT 1
  );
```

```sql
-- Tabelle "ausleihen" erstellen
CREATE TABLE
  ausleihen (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    buch_id INTEGER NOT NULL,
    mitglied_id INTEGER NOT NULL,
    ausleihdatum TEXT NOT NULL,
    rueckgabedatum TEXT,
    zurueckgegeben INTEGER NOT NULL DEFAULT 0,
    FOREIGN KEY (buch_id) REFERENCES buecher (id),
    FOREIGN KEY (mitglied_id) REFERENCES mitglieder (id)
  );
```

## 🔍 Einfache SQL-Abfragen

```sql
-- Alle Autoren anzeigen
SELECT
  *
FROM
  autoren;
```

```sql
-- Verfügbare Bücher anzeigen
SELECT
  b.titel,
  a.nachname AS autor,
  b.kategorie
FROM
  buecher b
  JOIN autoren a ON b.autor_id = a.id
WHERE
  b.verfuegbar = 1;
```

```sql
-- Aktuelle Ausleihen anzeigen
SELECT
  m.vorname || ' ' || m.nachname AS mitglied,
  b.titel,
  a.nachname AS autor,
  au.ausleihdatum,
  CASE
    WHEN au.zurueckgegeben = 1 THEN 'Zurückgegeben am ' || au.rueckgabedatum
    ELSE 'Noch ausgeliehen'
  END AS status
FROM
  ausleihen au
  JOIN mitglieder m ON au.mitglied_id = m.id
  JOIN buecher b ON au.buch_id = b.id
  JOIN autoren a ON b.autor_id = a.id
ORDER BY
  au.ausleihdatum DESC;
```

## 📊 Statistiken und Analysen

```sql
-- Bücher pro Kategorie
SELECT
  kategorie,
  COUNT(*) AS anzahl,
  AVG(seitenzahl) AS durchschnitt_seiten
FROM
  buecher
GROUP BY
  kategorie
ORDER BY
  anzahl DESC;
```

```sql
-- Autoren mit ihren Büchern
SELECT
  a.vorname || ' ' || a.nachname AS autor_name,
  a.nationalitaet,
  COUNT(b.id) AS anzahl_buecher
FROM
  autoren a
  LEFT JOIN buecher b ON a.id = b.autor_id
GROUP BY
  a.id,
  a.vorname,
  a.nachname,
  a.nationalitaet
ORDER BY
  anzahl_buecher DESC;
```

```sql
-- Mitglieder-Aktivität
SELECT
  m.vorname || ' ' || m.nachname AS mitglied_name,
  COUNT(au.id) AS anzahl_ausleihen,
  SUM(
    CASE
      WHEN au.zurueckgegeben = 0 THEN 1
      ELSE 0
    END
  ) AS aktuell_ausgeliehen
FROM
  mitglieder m
  LEFT JOIN ausleihen au ON m.id = au.mitglied_id
GROUP BY
  m.id,
  m.vorname,
  m.nachname
ORDER BY
  anzahl_ausleihen DESC;
```

## 📈 Datenvisualisierung mit JupySQL

```sql
# SQL-Ergebnis in DataFrame speichern
result = %sql SELECT kategorie, COUNT(*) as anzahl FROM buecher GROUP BY kategorie
df_kategorien = result.DataFrame()

# Visualisierung erstellen
plt.figure(figsize=(10, 6))
plt.pie(df_kategorien['anzahl'], labels=df_kategorien['kategorie'], autopct='%1.1f%%')
plt.title('📚 Bücher pro Kategorie')
plt.show()
```

```sql
# Bücher nach Erscheinungsjahr
```

### Aufgabe 3: Komplexe Abfrage - Längste Ausleihen

```sql
-- Welche Bücher sind am längsten ausgeliehen?
SELECT
  b.titel,
  m.vorname || ' ' || m.nachname AS mitglied,
  au.ausleihdatum,
  CAST(
    julianday ('now') - julianday (au.ausleihdatum) AS INTEGER
  ) AS tage_ausgeliehen
FROM
  ausleihen au
  JOIN buecher b ON au.buch_id = b.id
  JOIN mitglieder m ON au.mitglied_id = m.id
WHERE
  au.zurueckgegeben = 0
ORDER BY
  tage_ausgeliehen DESC;
```

## 🔧 Erweiterte Features

### View erstellen

```sql
-- View für verfügbare Bücher erstellen
CREATE VIEW
  IF NOT EXISTS verfuegbare_buecher AS
SELECT
  b.id,
  b.titel,
  a.vorname || ' ' || a.nachname AS autor,
  b.kategorie,
  b.erscheinungsjahr,
  b.seitenzahl
FROM
  buecher b
  JOIN autoren a ON b.autor_id = a.id
WHERE
  b.verfuegbar = 1;
```

```sql
-- View verwenden
SELECT
  *
FROM
  verfuegbare_buecher
ORDER BY
  erscheinungsjahr DESC;
```

### Datenexport

```sql
# Alle Bücher exportieren
result = %sql SELECT b.titel, a.vorname || ' ' || a.nachname as autor, b.kategorie, b.erscheinungsjahr FROM buecher b JOIN autoren a ON b.autor_id = a.id ORDER BY a.nachname
df_export = result.DataFrame()

# Als CSV speichern
df_export.to_csv('bibliothek_buecher.csv', index=False, encoding='utf-8')
```
