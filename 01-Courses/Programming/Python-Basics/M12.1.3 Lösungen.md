    if gefunden:

## Zusammenfassung: Dictionary-Grundlagen

```python
# Die wichtigsten Dictionary-Operationen:

# 1. Erstellen
dict1 = {"key": "value"}                    # Literal
dict2 = dict(key="value")                   # Konstruktor
dict3 = {}                                  # Leer

# 2. Zugriff
wert = d["key"]                            # Direkt (kann Fehler)
wert = d.get("key", "default")             # Sicher

# 3. Prüfungen
if "key" in d:                             # Schlüssel vorhanden?
if d:                                      # Dictionary nicht leer?

# 4. Iteration
for key in d:                              # Schlüssel
for value in d.values():                   # Werte
for key, value in d.items():               # Paare

# 5. Verschachtelt
nested = {"outer": {"inner": "value"}}
wert = nested["outer"]["inner"]

# Denken Sie daran: Schlüssel müssen unveränderlich sein!
```
