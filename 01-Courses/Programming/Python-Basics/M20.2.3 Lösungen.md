def entwickle_bibliotheks_system():
    """Erstellt vollständige Bibliotheksverwaltung mit Klassen und Methoden"""
    
    bibliothek_code = '''
class Buch:
    """Repräsentiert ein einzelnes Buch in der Bibliothek"""
    
    def __init__(self, titel, autor, isbn):
        self.titel = titel
        self.autor = autor
        self.isbn = isbn
        self.gelesen = False
    
    def als_gelesen_markieren(self):
        """Markiert das Buch als gelesen"""
        self.gelesen = True
        return f"'{self.titel}' wurde als gelesen markiert"
    
    def __str__(self):
        status = "✅ Gelesen" if self.gelesen else "📖 Ungelesen"
        return f"{self.titel} von {self.autor} ({status})"


class Bibliothek:
    """Verwaltet eine Sammlung von Büchern"""
    
    def __init__(self):
        self.buecher = []
    
    def buch_hinzufuegen(self, buch):
        """Fügt ein neues Buch zur Bibliothek hinzu"""
        self.buecher.append(buch)
        return f"Buch '{buch.titel}' wurde hinzugefügt"
    
    def buch_suchen(self, suchterm):
        """Sucht Bücher nach Titel oder Autor"""
        gefundene_buecher = []
        for buch in self.buecher:
            if (suchterm.lower() in buch.titel.lower() or 
                suchterm.lower() in buch.autor.lower()):
                gefundene_buecher.append(buch)
        return gefundene_buecher
    
    def alle_buecher_anzeigen(self):
        """Zeigt alle Bücher in der Bibliothek an"""
        if not self.buecher:
            return "Die Bibliothek ist leer."
        
        ausgabe = "\\n=== Meine Bibliothek ===\\n"
        for i, buch in enumerate(self.buecher, 1):
            ausgabe += f"{i}. {buch}\\n"
        return ausgabe
    
    def statistik(self):
        """Gibt Statistiken über die Bibliothek aus"""
        gesamt = len(self.buecher)
        gelesen = sum(1 for buch in self.buecher if buch.gelesen)
        ungelesen = gesamt - gelesen
        
        return f"""
=== Bibliothek-Statistik ===
Bücher gesamt: {gesamt}
Gelesen: {gelesen}
Noch zu lesen: {ungelesen}
Lesefortschritt: {(gelesen/gesamt*100) if gesamt > 0 else 0:.1f}%
"""


# Demonstration der Bibliotheksverwaltung
def main():
    """Hauptfunktion mit Beispieldaten"""
    # Bibliothek erstellen
    meine_bibliothek = Bibliothek()
    
    # Beispiel-Bücher hinzufügen
    buecher = [
        Buch("Python Crashkurs", "Eric Matthes", "978-3-86490-543-2"),
        Buch("Clean Code", "Robert C. Martin", "978-0-13-235088-4"),
        Buch("Der Pragmatische Programmierer", "David Thomas", "978-3-446-22309-7")
    ]
    
    for buch in buecher:

    # Ein Buch als gelesen markieren

    for buch in suchergebnis:
        print(f"  - {buch}")


if __name__ == "__main__":
    main()

def erweitere_bewertungssystem():
    """Erweitert die Bibliotheksverwaltung um ein 5-Sterne-Bewertungssystem"""
    
    erweiterte_version = '''
class Buch:
    """Repräsentiert ein einzelnes Buch in der Bibliothek mit Bewertungssystem"""
    
    def __init__(self, titel, autor, isbn):
        self.titel = titel
        self.autor = autor
        self.isbn = isbn
        self.gelesen = False
        self.bewertung = None  # 1-5 Sterne oder None wenn nicht bewertet
    
    def als_gelesen_markieren(self):
        """Markiert das Buch als gelesen"""
        self.gelesen = True
        return f"'{self.titel}' wurde als gelesen markiert"
    
    def bewerten(self, sterne):
        """Bewertet das Buch mit 1-5 Sternen"""
        if not isinstance(sterne, int) or sterne < 1 or sterne > 5:
            return "Bewertung muss zwischen 1 und 5 Sternen liegen"
        
        self.bewertung = sterne
        return f"'{self.titel}' wurde mit {sterne} Sternen bewertet"
    
    def bewertung_anzeigen(self):
        """Gibt die Bewertung als Sterne-String zurück"""
        if self.bewertung is None:
            return "Noch nicht bewertet"
        return "⭐" * self.bewertung + "☆" * (5 - self.bewertung)
    
    def __str__(self):
        status = "✅ Gelesen" if self.gelesen else "📖 Ungelesen"
        bewertung = f" - {self.bewertung_anzeigen()}" if self.bewertung else ""
        return f"{self.titel} von {self.autor} ({status}){bewertung}"


class Bibliothek:
    """Verwaltet eine Sammlung von Büchern mit erweiterten Statistiken"""
    
    def __init__(self):
        self.buecher = []
    
    def buch_hinzufuegen(self, buch):
        """Fügt ein neues Buch zur Bibliothek hinzu"""
        self.buecher.append(buch)
        return f"Buch '{buch.titel}' wurde hinzugefügt"
    
    def buch_suchen(self, suchterm):
        """Sucht Bücher nach Titel oder Autor"""
        gefundene_buecher = []
        for buch in self.buecher:
            if (suchterm.lower() in buch.titel.lower() or 
                suchterm.lower() in buch.autor.lower()):
                gefundene_buecher.append(buch)
        return gefundene_buecher
    
    def alle_buecher_anzeigen(self):
        """Zeigt alle Bücher in der Bibliothek an"""
        if not self.buecher:
            return "Die Bibliothek ist leer."
        
        ausgabe = "\\n=== Meine Bibliothek ===\\n"
        for i, buch in enumerate(self.buecher, 1):
            ausgabe += f"{i}. {buch}\\n"
        return ausgabe
    
    def durchschnittsbewertung(self):
        """Berechnet die Durchschnittsbewertung aller bewerteten Bücher"""
        bewertete_buecher = [buch for buch in self.buecher if buch.bewertung is not None]
        if not bewertete_buecher:
            return 0.0
        return sum(buch.bewertung for buch in bewertete_buecher) / len(bewertete_buecher)
    
    def beste_buecher(self, min_sterne=4):
        """Gibt eine Liste der besten Bücher zurück (min. 4 Sterne)"""
        return [buch for buch in self.buecher 
                if buch.bewertung is not None and buch.bewertung >= min_sterne]
    
    def bewertungsstatistik(self):
        """Erstellt eine detaillierte Bewertungsstatistik"""
        bewertete = [buch for buch in self.buecher if buch.bewertung is not None]
        unbewertete = len(self.buecher) - len(bewertete)
        
        if not bewertete:
            return "Noch keine Bücher bewertet."
        
        # Bewertungsverteilung zählen
        verteilung = {i: 0 for i in range(1, 6)}
        for buch in bewertete:
            verteilung[buch.bewertung] += 1
        
        ausgabe = f"""
=== Bewertungsstatistik ===
Durchschnittsbewertung: {self.durchschnittsbewertung():.1f} ⭐
Bewertete Bücher: {len(bewertete)}
Unbewertete Bücher: {unbewertete}

Bewertungsverteilung:"""
        
        for sterne in range(5, 0, -1):
            anzahl = verteilung[sterne]
            balken = "█" * anzahl if anzahl > 0 else ""
            ausgabe += f"\\n{sterne} ⭐: {anzahl:2d} {balken}"
        
        # Beste Bücher auflisten
        beste = self.beste_buecher()
        if beste:
            ausgabe += f"\\n\\n🏆 Beste Bücher (4+ Sterne):"
            for buch in beste:
                ausgabe += f"\\n  • {buch.titel} ({buch.bewertung} ⭐)"
        
        return ausgabe
    
    def statistik(self):
        """Gibt erweiterte Statistiken über die Bibliothek aus"""
        gesamt = len(self.buecher)
        gelesen = sum(1 for buch in self.buecher if buch.gelesen)
        ungelesen = gesamt - gelesen
        
        basis_stats = f"""
=== Bibliothek-Statistik ===
Bücher gesamt: {gesamt}
Gelesen: {gelesen}
Noch zu lesen: {ungelesen}
Lesefortschritt: {(gelesen/gesamt*100) if gesamt > 0 else 0:.1f}%"""
        
        return basis_stats + "\\n" + self.bewertungsstatistik()

    # Erweiterte Statistiken anzeigen
    print(meine_bibliothek.statistik())
    
    # Beste Bücher anzeigen
