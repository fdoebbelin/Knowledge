def verstehe_pipeline_konzept():
    """Versteht das Konzept der Pipeline-Verarbeitung"""
    test_daten = [18.5, -999, 22.3, 25.1, 28.9, 19.2]
    
    print("🧪 SCHRITT 4.1: Pipeline-Konzept")
    print("Original-Daten:", test_daten)
    print("\nPipeline-Schritte:")
    print("  1. Gültige Werte filtern (nicht -999)")
    print("  2. Temperaturen über 20°C filtern") 
    print("  3. In Fahrenheit umrechnen")
    print("  4. Auf ganze Zahlen runden")
    
    # Pipeline manuell durchgehen
    schritt1 = [temp for temp in test_daten if temp != -999]
    schritt2 = [temp for temp in schritt1 if temp > 20]
    schritt3 = [temp * 9/5 + 32 for temp in schritt2]
    schritt4 = [round(temp) for temp in schritt3]
    
    print(f"\n  Schritt 1 (gültig): {schritt1}")
    print(f"  Schritt 2 (>20°C): {schritt2}")
    print(f"  Schritt 3 (Fahrenheit): {schritt3}")
    print(f"  Schritt 4 (gerundet): {schritt4}")
    
    return schritt4

# Zelle 5.2: Erstellungs-Performance vergleichen
def vergleiche_erstellungs_performance():
    """Vergleicht die Performance bei der Erstellung"""
    import time
    
    test_daten = list(range(100000))
    
    print("🧪 SCHRITT 5.2: Erstellungs-Performance")
    
    # List Comprehension Zeit messen
    start = time.time()
    liste_result = [x * 2 + 1 for x in test_daten if x % 2 == 0]
    list_zeit = time.time() - start
    
    # Generator Expression Zeit messen  
    start = time.time()
    generator_result = (x * 2 + 1 for x in test_daten if x % 2 == 0)
    gen_zeit = time.time() - start
    
    # Speedup berechnen
    speedup = list_zeit / gen_zeit if gen_zeit > 0 else float('inf')
    
    print(f"List Comprehension: {list_zeit:.6f} Sekunden")
    print(f"Generator Expression: {gen_zeit:.6f} Sekunden")
    print(f"Generator ist {speedup:.0f}x schneller bei der Erstellung!")
    print(f"Verarbeitete Elemente: {len(liste_result):,}")
    
    return {
        'list_zeit': list_zeit,
        'generator_zeit': gen_zeit,
        'speedup': speedup,
        'list_laenge': len(liste_result)
    }

## 💡 Lösungshinweise

### Generator Expression Syntax:
```python
# Grundsyntax (runde Klammern!)
(expression for item in iterable)

# Mit Filter
(expression for item in iterable if condition)

# Verkettete Generatoren
gen1 = (item for item in data if condition1)
gen2 = (transform(item) for item in gen1 if condition2)
```
