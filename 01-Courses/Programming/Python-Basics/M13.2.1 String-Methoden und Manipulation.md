## 2.1 Text in Groß- und Kleinbuchstaben umwandeln

```python
# Grundlegende Umwandlungen
text = "PyThOn IsT tOlL"
print("Original:", text)

# Verschiedene Umwandlungen
print("Alles groß:", text.upper())
print("Alles klein:", text.lower()) 
print("Erster Buchstabe groß:", text.capitalize())
print("Jedes Wort groß:", text.title())
```

```python
# Praktisches Beispiel: Namen normalisieren
namen = ["max MUSTERMANN", "anna schmidt", "TOM weber"]

print("Namen normalisieren:")
for name in namen:
    normalisiert = name.title()  # Jedes Wort mit großem Anfangsbuchstaben
    print(f"{name:20} → {normalisiert}")
```

```python
# Text vergleichen ohne Groß-/Kleinschreibung
passwort = "GeHeiM123"
eingabe = "geheim123"

# Direkte Vergleich (falsch)
print("Direkt verglichen:", passwort == eingabe)

# Richtig: beide in Kleinbuchstaben umwandeln
print("Klein verglichen:", passwort.lower() == eingabe.lower())

# Praktisch für Suche
text = "Python ist eine TOLLE Sprache"
suchbegriff = "tolle"

if suchbegriff.lower() in text.lower():
    print(f"'{suchbegriff}' gefunden!")
```
## 2.2 Text bereinigen

```python
# Leerzeichen entfernen
unordentlicher_text = "   Hallo Welt   "
print(f"Original: '{unordentlicher_text}'")

# Verschiedene Bereinigungen
print(f"Beide Seiten: '{unordentlicher_text.strip()}'")
print(f"Nur links: '{unordentlicher_text.lstrip()}'")
print(f"Nur rechts: '{unordentlicher_text.rstrip()}'")
```

```python
# Benutzereingaben bereinigen
def bereinige_eingabe(eingabe):
    """Bereinigt Benutzereingaben"""
    # Leerzeichen entfernen und zu Kleinbuchstaben
    sauber = eingabe.strip().lower()
    return sauber

# Test mit verschiedenen Eingaben
eingaben = ["  PYTHON  ", "java", "   C++   ", "JavaScript"]

print("Eingaben bereinigen:")
for eingabe in eingaben:
    sauber = bereinige_eingabe(eingabe)
    print(f"'{eingabe}' → '{sauber}'")
```

```python
# Unerwünschte Zeichen entfernen
telefonnummer = "+49(0123)-456-789"
print("Original:", telefonnummer)

# Bindestriche entfernen
ohne_bindestrich = telefonnummer.replace("-", "")
print("Ohne Bindestriche:", ohne_bindestrich)

# Alle Nicht-Zahlen entfernen (außer + am Anfang)
nur_zahlen = ""
for zeichen in telefonnummer:
    if zeichen.isdigit() or (zeichen == "+" and telefonnummer.startswith("+")):
        nur_zahlen += zeichen

print("Nur Zahlen:", nur_zahlen)
```
## 2.3 In Text suchen

```python
# Position eines Textes finden
email = "max.mustermann@firma.de"
print("E-Mail:", email)

# @ Position finden
at_position = email.find("@")
print("@ steht an Position:", at_position)

# Benutzername und Domain extrahieren
if at_position != -1:  # -1 bedeutet "nicht gefunden"
    benutzername = email[:at_position]
    domain = email[at_position + 1:]
    print("Benutzername:", benutzername)
    print("Domain:", domain)
```

```python
# Prüfen womit Text anfängt oder aufhört
dateien = ["bild.jpg", "dokument.pdf", "script.py", "data.csv"]

print("Dateien nach Typ sortieren:")

# Nach Dateiendung filtern
for datei in dateien:
    if datei.endswith(".jpg"):
        print(f"📸 {datei} ist ein Bild")
    elif datei.endswith(".pdf"):
        print(f"📄 {datei} ist ein Dokument")
    elif datei.endswith(".py"):
        print(f"🐍 {datei} ist Python-Code")
    else:
        print(f"❓ {datei} ist etwas anderes")
```

```python
# Text auf bestimmte Eigenschaften prüfen
test_eingaben = ["12345", "hallo", "Test123", "   ", ""]

print("Eingaben prüfen:")
for eingabe in test_eingaben:
    print(f"'{eingabe}':")
    print(f"  Nur Zahlen? {eingabe.isdigit()}")
    print(f"  Nur Buchstaben? {eingabe.isalpha()}")
    print(f"  Zahlen + Buchstaben? {eingabe.isalnum()}")
    print(f"  Nur Leerzeichen? {eingabe.isspace()}")
    print()
```
## 2.4 Text aufteilen

```python
# Text in Wörter aufteilen
satz = "Python ist eine tolle Programmiersprache"
print("Satz:", satz)

# Aufteilen in Wörter
woerter = satz.split()
print("Wörter:", woerter)
print("Anzahl Wörter:", len(woerter))

# Jedes Wort einzeln ausgeben
for i, wort in enumerate(woerter):
    print(f"Wort {i+1}: {wort}")
```

```python
# CSV-Daten verarbeiten
csv_zeile = "Anna,25,Berlin,Studentin"
print("CSV-Zeile:", csv_zeile)

# Bei Komma aufteilen
daten = csv_zeile.split(",")
print("Aufgeteilt:", daten)

# Daten zuweisen
name = daten[0]
alter = daten[1]
stadt = daten[2]
beruf = daten[3]

print(f"Name: {name}")
print(f"Alter: {alter}")
print(f"Stadt: {stadt}")
print(f"Beruf: {beruf}")
```

```python
# Dateiname und Erweiterung trennen
dateiname = "mein_dokument.pdf"
print("Dateiname:", dateiname)

# Am letzten Punkt trennen
teile = dateiname.rsplit(".", 1)  # 1 = nur einmal trennen
if len(teile) == 2:
    name = teile[0]
    erweiterung = teile[1]
    print(f"Name: {name}")
    print(f"Erweiterung: {erweiterung}")
```
## 2.5 Text zusammenfügen

```python
# Wörter zu Satz verbinden
woerter = ["Python", "macht", "Spaß"]
print("Wörter:", woerter)

# Mit Leerzeichen verbinden
satz = " ".join(woerter)
print("Satz:", satz)

# Mit anderen Zeichen verbinden
mit_bindestrich = "-".join(woerter)
print("Mit Bindestrich:", mit_bindestrich)

ohne_leerzeichen = "".join(woerter)
print("Ohne Leerzeichen:", ohne_leerzeichen)
```

```python
# CSV-Zeile erstellen
person_daten = ["Max", "30", "München", "Entwickler"]
print("Daten:", person_daten)

# Als CSV-Zeile zusammenfügen
csv_zeile = ",".join(person_daten)
print("CSV-Zeile:", csv_zeile)

# Mit verschiedenen Trennzeichen
tab_getrennt = "\t".join(person_daten)  # Tab-getrennt
print("Tab-getrennt:", tab_getrennt)
```

```python
# Pfad zusammenbauen
ordner = ["Benutzer", "Max", "Dokumente", "Python"]
print("Ordner-Teile:", ordner)

# Windows-Pfad
windows_pfad = "\\".join(ordner)
print("Windows:", windows_pfad)

# Unix/Mac-Pfad  
unix_pfad = "/".join(ordner)
print("Unix/Mac:", unix_pfad)
```
## 2.6 Text ersetzen

```python
# Einfache Ersetzung
text = "Ich mag Java. Java ist toll."
print("Original:", text)

# Java durch Python ersetzen
neuer_text = text.replace("Java", "Python")
print("Ersetzt:", neuer_text)

# Nur erste Vorkommen ersetzen
nur_erstes = text.replace("Java", "Python", 1)
print("Nur erstes:", nur_erstes)
```

```python
# Mehrere Ersetzungen
text = "Hallo Welt! Wie geht es dir, Welt?"
print("Original:", text)

# Kettung von Ersetzungen
ersetzt = text.replace("Hallo", "Hi").replace("Welt", "Python")
print("Mehrfach ersetzt:", ersetzt)

# Mit Dictionary
ersetzungen = {
    "Hallo": "Hi",
    "Welt": "Python", 
    "geht": "läuft"
}

ergebnis = text
for alt, neu in ersetzungen.items():
    ergebnis = ergebnis.replace(alt, neu)
    
print("Mit Dictionary:", ergebnis)
```
