# Zelle 3.5: Ausleihfunktionen implementieren
def entwickle_ausleihsystem():
    """Entwickelt Ausleihfunktionen im Feature-Branch"""
    
    ausleih_code = '''

def buch_ausleihen(isbn, benutzer_name):
    """Leiht ein Buch an einen Benutzer aus"""
    buch = suche_nach_isbn_intern(isbn)
    
    if not buch:
        print(f"❌ Buch mit ISBN {isbn} nicht gefunden.")
        return False
    
    if not buch["verfuegbar"]:
        print(f"❌ Buch '{buch['titel']}' ist bereits ausgeliehen.")
        return False
    
    buch["verfuegbar"] = False
    buch["ausgeliehen_an"] = benutzer_name
    print(f"✅ Buch '{buch['titel']}' an {benutzer_name} ausgeliehen.")
    return True

def buch_zurueckgeben(isbn):
    """Gibt ein ausgeliehenes Buch zurück"""
    buch = suche_nach_isbn_intern(isbn)
    
    if not buch:
        print(f"❌ Buch mit ISBN {isbn} nicht gefunden.")
        return False
    
    if buch["verfuegbar"]:
        print(f"⚠️ Buch '{buch['titel']}' ist nicht ausgeliehen.")
        return False
    
    benutzer = buch.get("ausgeliehen_an", "Unbekannt")
    buch["verfuegbar"] = True
    if "ausgeliehen_an" in buch:
        del buch["ausgeliehen_an"]
    
    print(f"✅ Buch '{buch['titel']}' von {benutzer} zurückgegeben.")
    return True

def zeige_ausgeliehene_buecher():
    """Zeigt alle ausgeliehenen Bücher an"""
    ausgeliehen = [buch for buch in buecher_liste if not buch["verfuegbar"]]
    
    if not ausgeliehen:
        print("ℹ️ Alle Bücher sind verfügbar.")
        return
    
    print("📋 Ausgeliehene Bücher:")
    for buch in ausgeliehen:
        benutzer = buch.get("ausgeliehen_an", "Unbekannt")
        print(f"   📖 {buch['titel']} - ausgeliehen an: {benutzer}")

def suche_nach_isbn_intern(isbn):
    """Interne Hilfsfunktion für ISBN-Suche"""
    for buch in buecher_liste:
        if buch["isbn"] == isbn:
            return buch
    return None

# Tests des Ausleihsystems
buch_ausleihen("978-123456789", "Anna Müller")
buch_ausleihen("978-111111111", "Tom Schmidt")
zeige_ausgeliehene_buecher()
buch_zurueckgeben("978-123456789")'''

    with open("bibliothek-system/bibliothek.py", "a", encoding="utf-8") as datei:
        datei.write(ausleih_code)
    
    print("\n📋 SCHRITT 3.5: Ausleihsystem entwickelt")
    print("=" * 42)
    print("✅ buch_ausleihen() für Buchausleihe")
