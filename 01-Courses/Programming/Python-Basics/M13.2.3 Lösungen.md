## Aufgabe 1: Benutzername-Validator

```python
# Zelle 1: Benutzername validieren
benutzernamen = [
    "max_müller",
    "Anna123",
    "123invalid", 
    "user@domain.com",
    "valid_user",
    "",
    "ab"
]

print("Benutzername-Validierung:")
print("-" * 40)

for benutzername in benutzernamen:
    # Alle Regeln prüfen
    lang_genug = len(benutzername) >= 3
    beginnt_mit_buchstabe = len(benutzername) > 0 and benutzername[0].isalpha()
    nur_erlaubte_zeichen = benutzername.replace("_", "").isalnum()
    
    # Alle Bedingungen müssen erfüllt sein
    if lang_genug and beginnt_mit_buchstabe and nur_erlaubte_zeichen:
        print(f"✅ {benutzername:15} - Gültig")
    else:
        print(f"❌ {benutzername:15} - Ungültig")
        
        # Detaillierte Fehlermeldungen
        if not lang_genug:
            print(f"   → Zu kurz (mindestens 3 Zeichen)")
        if not beginnt_mit_buchstabe:
            print(f"   → Muss mit Buchstaben beginnen")
        if not nur_erlaubte_zeichen:
            print(f"   → Nur Buchstaben, Zahlen und _ erlaubt")
```

**Erklärung:**
- `len(benutzername) >= 3` prüft die Mindestlänge
- `benutzername[0].isalpha()` prüft erstes Zeichen (nur wenn String nicht leer)
- `replace("_", "").isalnum()` entfernt alle Unterstriche und prüft dann ob nur Buchstaben/Zahlen
## Aufgabe 2: E-Mail-Cleaner

```python
# Zelle 2: E-Mail-Adressen verarbeiten
email_liste = [
    "  MAX.MUSTERMANN@GMAIL.COM  ",
    "anna.schmidt@email.de",
    "  TOM.WEBER@yahoo.com  ",
    "invalid.email",
    "  test@DOMAIN.COM  "
]

print("E-Mail-Verarbeitung:")
print("-" * 50)

gueltige_emails = []

for email in email_liste:
    # 1. E-Mail bereinigen
    saubere_email = email.strip().lower()
    
    # 2. Gültigkeit prüfen
    hat_at = "@" in saubere_email
    hat_punkt = "." in saubere_email
    
    if hat_at and hat_punkt:
        # 3. Benutzername und Domain extrahieren
        at_position = saubere_email.find("@")
        benutzername = saubere_email[:at_position]
        domain = saubere_email[at_position + 1:]
        
        print(f"✅ {saubere_email}")
        print(f"   Benutzer: {benutzername}")
        print(f"   Domain: {domain}")
        
        gueltige_emails.append(saubere_email)
    else:
        print(f"❌ {saubere_email} - Ungültig")
        if not hat_at:
            print(f"   → Fehlt: @")
        if not hat_punkt:
            print(f"   → Fehlt: .")

print(f"\nGültige E-Mails: {len(gueltige_emails)}")
for email in gueltige_emails:
    print(f"  - {email}")
```

**Erklärung:**
- `email.strip().lower()` entfernt Leerzeichen und macht alles klein
- `"@" in email` prüft ob @ enthalten ist
- `email.find("@")` findet Position von @ für Aufspaltung
- `email[:position]` und `email[position+1:]` teilen bei @ auf
## Aufgabe 3: Passwort-Stärke-Checker

```python
# Zelle 3: Passwort-Stärke prüfen
passwoerter = [
    "123456",
    "passwort", 
    "Passwort123",
    "SicheresPasswort!123",
    "abc",
    "MeinSuperSicheresPasswort2024!"
]

print("Passwort-Stärke-Analyse:")
print("-" * 60)

for passwort in passwoerter:
    punkte = 0
    details = []
    
    # Regel 1: Mindestens 8 Zeichen
    if len(passwort) >= 8:
        punkte += 1
        details.append("✓ Lang genug")
    else:
        details.append("✗ Zu kurz")
    
    # Regel 2: Großbuchstaben
    if any(c.isupper() for c in passwort):
        punkte += 1
        details.append("✓ Großbuchstaben")
    else:
        details.append("✗ Keine Großbuchstaben")
    
    # Regel 3: Kleinbuchstaben
    if any(c.islower() for c in passwort):
        punkte += 1
        details.append("✓ Kleinbuchstaben")
    else:
        details.append("✗ Keine Kleinbuchstaben")
    
    # Regel 4: Zahlen
    if any(c.isdigit() for c in passwort):
        punkte += 1
        details.append("✓ Zahlen")
    else:
        details.append("✗ Keine Zahlen")
    
    # Regel 5: Sonderzeichen
    if any(not c.isalnum() for c in passwort):
        punkte += 1
        details.append("✓ Sonderzeichen")
    else:
        details.append("✗ Keine Sonderzeichen")
    
    # Stärke bewerten
    if punkte <= 2:
        staerke = "Schwach"
        symbol = "🔴"
    elif punkte <= 4:
        staerke = "Mittel"
        symbol = "🟡"
    else:
        staerke = "Stark"
        symbol = "🟢"
    
    print(f"{symbol} '{passwort}' - {staerke} ({punkte}/5 Punkte)")
    for detail in details:
        print(f"   {detail}")
    print()
```

**Erklärung:**
- `any(c.isupper() for c in passwort)` prüft ob mindestens ein Großbuchstabe vorhanden
- `any(c.isdigit() for c in passwort)` prüft auf Zahlen
- `any(not c.isalnum() for c in passwort)` prüft auf Sonderzeichen (nicht alphanumerisch)
- Generator Expressions sind sehr effizient für solche Prüfungen
## Aufgabe 4: CSV-Daten-Processor

```python
# Zelle 4: CSV-Daten verarbeiten
csv_daten = [
    "Name,Alter,Stadt,Gehalt",  # Header
    "Max Mustermann,30,Berlin,45000",
    "Anna Schmidt,25,München,42000", 
    "Tom Weber,35,Hamburg,48000",
    "Lisa Mueller,28,Köln,46000"
]

print("CSV-Datenverarbeitung:")
print("=" * 50)

# 1. Header von Daten trennen
header = csv_daten[0].split(",")
datenzeilen = csv_daten[1:]

print(f"Header: {header}")
print(f"Anzahl Datensätze: {len(datenzeilen)}")
print()

# 2. Daten verarbeiten
personen = []
alter_liste = []
gehalt_liste = []

print("Personen-Daten:")
print("-" * 30)

for zeile in datenzeilen:
    # CSV-Zeile aufteilen
    teile = zeile.split(",")
    name = teile[0]
    alter = int(teile[1])  # Text zu Zahl umwandeln
    stadt = teile[2]
    gehalt = int(teile[3])
    
    # Person als Dictionary speichern
    person = {
        "name": name,
        "alter": alter,
        "stadt": stadt,
        "gehalt": gehalt
    }
    personen.append(person)
    
    # Für Berechnungen sammeln
    alter_liste.append(alter)
    gehalt_liste.append(gehalt)
    
    # Ausgabe
    print(f"{name:15} {alter:2d} Jahre  {stadt:8} {gehalt:6,}€")

print()

# 3. Statistiken berechnen
durchschnitts_alter = sum(alter_liste) / len(alter_liste)
durchschnitts_gehalt = sum(gehalt_liste) / len(gehalt_liste)

aelteste_person = max(personen, key=lambda p: p["alter"])
juengste_person = min(personen, key=lambda p: p["alter"])

print("📊 Statistiken:")
print("-" * 20)
print(f"Durchschnittsalter: {durchschnitts_alter:.1f} Jahre")
print(f"Durchschnittsgehalt: {durchschnitts_gehalt:,.0f}€")
print(f"Älteste Person: {aelteste_person['name']} ({aelteste_person['alter']} Jahre)")
print(f"Jüngste Person: {juengste_person['name']} ({juengste_person['alter']} Jahre)")

# 4. Zusätzliche Auswertungen
print("\n🏙️ Städte-Verteilung:")
staedte = {}
for person in personen:
    stadt = person["stadt"]
    if stadt in staedte:
        staedte[stadt] += 1
    else:
        staedte[stadt] = 1

for stadt, anzahl in staedte.items():
    print(f"  {stadt}: {anzahl} Person(en)")

# 5. Gehalt-Kategorien
print("\n💰 Gehalt-Kategorien:")
for person in personen:
    gehalt = person["gehalt"]
    if gehalt < 43000:
        kategorie = "Niedrig"
    elif gehalt < 47000:
        kategorie = "Mittel"
    else:
        kategorie = "Hoch"
    
    print(f"  {person['name']:15} {gehalt:6,}€ - {kategorie}")
```

**Erklärung:**
- `zeile.split(",")` teilt CSV-Zeile bei Kommas auf
- `int(teile[1])` wandelt Alter-Text in Zahl um
- `max(personen, key=lambda p: p["alter"])` findet Person mit höchstem Alter
- Dictionary wird verwendet um Städte zu zählen
- `{gehalt:6,}` formatiert Zahlen mit Tausender-Trennzeichen
## Bonus-Aufgabe: Text-Statistik-Generator

```python
# Bonus: Text-Statistik-Generator
text = """
Python ist eine interpretierte, hochsprachige Programmiersprache.
Python wurde von Guido van Rossum entwickelt.
Viele Entwickler verwenden Python für verschiedene Projekte.
"""

print("📝 Text-Statistik-Generator")
print("=" * 40)

# Text bereinigen
import string
sauberer_text = text.strip()

# 1. Sätze zählen (durch . getrennt)
saetze = [satz.strip() for satz in sauberer_text.split(".") if satz.strip()]
anzahl_saetze = len(saetze)

print(f"Anzahl Sätze: {anzahl_saetze}")

# 2. Längster und kürzester Satz
if saetze:
    laengster_satz = max(saetze, key=len)
    kuerzester_satz = min(saetze, key=len)
    
    print(f"Längster Satz: '{laengster_satz}' ({len(laengster_satz)} Zeichen)")
    print(f"Kürzester Satz: '{kuerzester_satz}' ({len(kuerzester_satz)} Zeichen)")

# 3. Wörter analysieren
# Interpunktion entfernen und aufteilen
text_ohne_punkte = sauberer_text.translate(str.maketrans("", "", string.punctuation))
woerter = text_ohne_punkte.lower().split()

print(f"\nAnzahl Wörter: {len(woerter)}")

# 4. Durchschnittliche Wortlänge
if woerter:
    durchschnittliche_laenge = sum(len(wort) for wort in woerter) / len(woerter)
    print(f"Durchschnittliche Wortlänge: {durchschnittliche_laenge:.1f} Zeichen")

# 5. Häufigste Wörter (ohne Füllwörter)
fuellwoerter = {"ist", "eine", "von", "der", "die", "das", "und", "für", "wurde"}

wort_haeufigkeit = {}
for wort in woerter:
    if wort not in fuellwoerter and len(wort) > 2:  # Nur längere Wörter
        if wort in wort_haeufigkeit:
            wort_haeufigkeit[wort] += 1
        else:
            wort_haeufigkeit[wort] = 1

# Nach Häufigkeit sortieren
sortierte_woerter = sorted(wort_haeufigkeit.items(), key=lambda x: x[1], reverse=True)

print(f"\n🔝 Häufigste Wörter (ohne Füllwörter):")
for wort, anzahl in sortierte_woerter[:5]:  # Top 5
    print(f"  '{wort}': {anzahl}x")

# 6. Buchstaben-Analyse
print(f"\n🔤 Buchstaben-Analyse:")
text_nur_buchstaben = ''.join(c.lower() for c in sauberer_text if c.isalpha())
print(f"Anzahl Buchstaben: {len(text_nur_buchstaben)}")

# Häufigste Buchstaben
buchstaben_count = {}
for buchstabe in text_nur_buchstaben:
    buchstaben_count[buchstabe] = buchstaben_count.get(buchstabe, 0) + 1

haeufigste_buchstaben = sorted(buchstaben_count.items(), key=lambda x: x[1], reverse=True)
print("Häufigste Buchstaben:")
for buchstabe, anzahl in haeufigste_buchstaben[:3]:
    print(f"  '{buchstabe}': {anzahl}x")
```

**Erklärung:**
- `text.split(".")` teilt bei Punkten für Sätze
- `str.maketrans("", "", string.punctuation)` entfernt alle Satzzeichen
- `max(liste, key=len)` findet längsten String
- Füllwörter werden über ein Set ausgeschlossen
- `sorted(dict.items(), key=lambda x: x[1], reverse=True)` sortiert nach Werten

- **List Comprehensions:** `[ausdruck for item in liste if bedingung]`
- **String-Formatierung:** `f"{wert:,}"` für Zahlenformatierung
