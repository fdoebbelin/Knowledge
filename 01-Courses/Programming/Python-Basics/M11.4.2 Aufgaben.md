```python
# Erstellen Sie ein einfaches Koordinaten-System
def erstelle_spielfeld():
    """Erstellt ein Spielfeld mit verschiedenen Objekten."""
    # Verwenden Sie Tupel für Koordinaten, Liste für Objekte
    objekte = [
        # Format: (x, y, objekt_typ, eigenschaften)
        ((0, 0), "start", {"farbe": "grün"}),
        ((2, 1), "münze", {"wert": 10}),
        ((3, 2), "hindernis", {"typ": "fels"}),
        ((5, 3), "münze", {"wert": 20}),
        ((7, 4), "ziel", {"farbe": "rot"})
    ]
    return objekte

def zeige_spielfeld(objekte):
    """Zeigt alle Objekte im Spielfeld an."""
    print("🎮 Spielfeld-Objekte:")
    for position, obj_typ, eigenschaften in objekte:
        x, y = position  # Tupel unpacking
        print(f"  Position ({x}, {y}): {obj_typ} - {eigenschaften}")

def finde_muenzen(objekte):
    """Findet alle Münzen und gibt ihre Positionen zurück."""
    # TODO: Implementieren Sie die Suche
    muenzen = []  # Liste für gefundene Münzen
    for position, obj_typ, eigenschaften in objekte:
        if obj_typ == "münze":
            muenzen.append(# Tupel mit Position und Wert)
    return muenzen

# Test des Spielfelds
spielfeld = erstelle_spielfeld()
zeige_spielfeld(spielfeld)

print("\n💰 Gefundene Münzen:")
muenzen = finde_muenzen(spielfeld)
for position, wert in muenzen:
    x, y = position
    print(f"  Münze bei ({x}, {y}) im Wert von {wert}")

gesamtwert = sum(wert for position, wert in muenzen)
print(f"\nGesamtwert aller Münzen: {gesamtwert}")
```

**💡 Lösungshinweise:**
- Verwenden Sie Tupel für unveränderliche Koordinaten
- Listen für veränderliche Objektsammlungen
- Tupel unpacking: `x, y = position`
- Münzen-Format: `(position, wert)`
