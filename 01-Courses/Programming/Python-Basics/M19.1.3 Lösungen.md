```python
# Zelle 5.5: Commit-Hashes für Vergleich extrahieren
def extrahiere_commit_hashes():
    """Extrahiert die ersten zwei Commit-Hashes für Vergleich"""
    import subprocess
    
    try:
        # Git log --oneline ausführen und Ausgabe parsen
        result = subprocess.run(
            ['git', 'log', '--oneline'],
            cwd='notiz-app',
            capture_output=True,
            text=True
        )
        
        commits = result.stdout.strip().split('\n')
        if len(commits) >= 2:
            erste_hash = commits[-1].split()[0]  # Erster Commit
            letzte_hash = commits[0].split()[0]   # Letzter Commit
            
            print("📈 SCHRITT 5.5: Commit-Vergleich vorbereitet")
            print("=" * 49)
            print(f"✅ Erster Commit: {erste_hash}")
            print(f"✅ Letzter Commit: {letzte_hash}")
            print(f"✅ Anzahl Commits: {len(commits)}")
            
            return f"Erster: {erste_hash}, Letzter: {letzte_hash}"
        else:
            return "Nicht genügend Commits für Vergleich"
            
    except Exception as e:
        return f"Fehler beim Extrahieren: {e}"

print(extrahiere_commit_hashes())
```

```python
def analysiere_git_entwicklung():
    """Analysiert die Git-Entwicklung des Notiz-App Projekts"""

    try:
        # Anzahl Commits ermitteln
        result = subprocess.run(
            ['git', 'rev-list', '--count', 'HEAD'],
            cwd='notiz-app',
            capture_output=True,
            text=True
        )
        commit_anzahl = int(result.stdout.strip())
        
        # Dateien im Repository
        result = subprocess.run(
            ['git', 'ls-files'],
            cwd='notiz-app',

            text=True

        # Zeilen Code zählen
```
