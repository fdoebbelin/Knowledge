# Zelle 1.1: Basis-Klassenliste mit csv.writer() erstellen
def erstelle_klassenliste():
    """Erstellt eine einfache Klassenliste in CSV-Format"""
    import csv
    
    # Schülerdaten als Liste von Listen - jede innere Liste = eine Zeile
    schueler_daten = [
        ["Name", "Alter", "Klasse"],           # Header-Zeile
        ["Anna Schmidt", "16", "10A"],         # Schüler 1
        ["Bob Müller", "15", "10A"],           # Schüler 2
        ["Charlie Wagner", "16", "10A"],       # Schüler 3
        ["Diana Fischer", "15", "10A"]         # Schüler 4
    ]
    
    # CSV-Datei mit csv.writer() erstellen
    # newline="" verhindert leere Zeilen zwischen Datenzeilen
    with open("klassenliste.csv", "w", newline="", encoding="utf-8") as datei:
        writer = csv.writer(datei, quoting=csv.QUOTE_ALL)
        for zeile in schueler_daten:
            writer.writerow(zeile)  # Jede Zeile einzeln schreiben
    
    print("📝 SCHRITT 1.1: Klassenliste erstellen")
    print("=" * 42)
    print("✅ csv.writer() verwendet")
    print("✅ newline='' für saubere CSV-Formatierung")
    print("✅ encoding='utf-8' für deutsche Umlaute")
    print(f"✅ {len(schueler_daten)} Zeilen geschrieben (inkl. Header)")
    print("✅ Datei 'klassenliste.csv' erstellt")
    return f"Klassenliste mit {len(schueler_daten)-1} Schülern erstellt"

# Test der Funktion

# Zelle 2.1: Klassenliste mit csv.reader() lesen
def lese_klassenliste():
    """Liest und zeigt die komplette Klassenliste an"""
    import csv
    
    # CSV-Datei mit csv.reader() lesen
    alle_schueler = []
    with open("klassenliste.csv", "r", encoding="utf-8") as datei:
        reader = csv.reader(datei)
        for zeile in reader:
            alle_schueler.append(zeile)  # Jede Zeile ist eine Liste
    
    print("\n📖 SCHRITT 2.1: Klassenliste lesen")
    print("=" * 40)
    print("✅ csv.reader() verwendet")
    print(f"✅ {len(alle_schueler)} Zeilen gelesen (inkl. Header)")
    print("\n📚 KOMPLETTE KLASSENLISTE:")
    
    for i, schueler in enumerate(alle_schueler):
        if i == 0:  # Header-Zeile
            print(f"  {schueler[0]:<15} {schueler[1]:<5} {schueler[2]}")
            print("  " + "-" * 30)
        else:  # Datenzeilen
            print(f"  {schueler[0]:<15} {schueler[1]:<5} {schueler[2]}")
    
    return f"Klassenliste gelesen: {len(alle_schueler)-1} Schüler"

def trenne_header_und_daten():
    """Liest CSV und trennt Header von Schülerdaten"""
    import csv
    
    with open("klassenliste.csv", "r", encoding="utf-8") as datei:
        reader = csv.reader(datei)
        
        # next() liest die erste Zeile (Header) separat
        header = next(reader)
        
        # Restliche Zeilen sind die eigentlichen Schülerdaten
        schueler_liste = []
        for zeile in reader:
            schueler_liste.append(zeile)
    
    print("\n📖 SCHRITT 2.2: Header und Daten trennen")
    print("=" * 46)
    print("✅ next(reader) für Header-Zeile verwendet")
    print("✅ Datenzeilen separat gesammelt")

    print(f"Anzahl Schüler (ohne Header): {len(schueler_liste)}")
    
    print("\nNur die Schülerdaten:")
    for i, schueler in enumerate(schueler_liste, 1):
        print(f"  Schüler {i}: {schueler}")
    
    return f"Header: {header}, Schüler: {len(schueler_liste)}"

# Zelle 3.1: Notenliste mit benannten Spalten
def erstelle_notenliste():
    """Erstellt eine Notenliste mit csv.DictWriter()"""
    import csv
    
    # Noten als Liste von Dictionaries - jedes Dict = eine Zeile
    noten_daten = [
        {"Name": "Anna Schmidt", "Fach": "Mathematik", "Note": "1.3", "Datum": "2024-01-15"},
        {"Name": "Bob Müller", "Fach": "Deutsch", "Note": "2.1", "Datum": "2024-01-15"},
        {"Name": "Charlie Wagner", "Fach": "Englisch", "Note": "1.8", "Datum": "2024-01-15"},
        {"Name": "Diana Fischer", "Fach": "Physik", "Note": "2.4", "Datum": "2024-01-15"}
    ]
    
    # CSV mit DictWriter erstellen - benannte Spalten!
    with open("noten.csv", "w", newline="", encoding="utf-8") as datei:
        # fieldnames definiert die Spaltenreihenfolge
        fieldnames = ["Name", "Fach", "Note", "Datum"]
        dict_writer = csv.DictWriter(datei, fieldnames=fieldnames)
        
        # writeheader() schreibt automatisch die Spaltenüberschriften
        dict_writer.writeheader()
        
        # Jedes Dictionary wird als Zeile geschrieben
        for note in noten_daten:
            dict_writer.writerow(note)
    
    print("\n📝 SCHRITT 3.1: Notenliste mit DictWriter")
    print("=" * 48)
    print("✅ csv.DictWriter() für benannte Spalten verwendet")
    print("✅ fieldnames definiert: ['Name', 'Fach', 'Note', 'Datum']")
    print("✅ writeheader() schreibt automatisch den Header")
    print(f"✅ {len(noten_daten)} Noten-Datensätze geschrieben")
    print("✅ Datei 'noten.csv' erstellt")
    return f"Notenliste mit {len(noten_daten)} Noten erstellt"

# Zelle 4.3: Durchschnittsnote pro Schüler berechnen
def berechne_durchschnittsnoten():
    """Berechnet die Durchschnittsnote für jeden Schüler"""
    import csv
    
    # Dictionary für Noten pro Schüler
    schueler_noten = {}  # Format: {"Name": [note1, note2, ...]}
    
    with open("noten.csv", "r", encoding="utf-8") as datei:
        dict_reader = csv.DictReader(datei)
        for note in dict_reader:
            name = note["Name"]
            note_wert = float(note["Note"])
            
            # Wenn Schüler noch nicht im Dictionary, neue Liste erstellen
            if name not in schueler_noten:
                schueler_noten[name] = []
            
            # Note zur Liste des Schülers hinzufügen
            schueler_noten[name].append(note_wert)
    
    print("\n📖 SCHRITT 4.3: Durchschnittsnoten berechnen")
    print("=" * 50)
    print("✅ Dictionary für Noten pro Schüler erstellt")
    print("✅ Mehrere Noten pro Schüler werden gesammelt")
    print("✅ sum(noten) / len(noten) für Durchschnitt")
    print(f"✅ Statistik für {len(schueler_noten)} Schüler berechnet")
    
    print("\n📈 DURCHSCHNITTSNOTEN PRO SCHÜLER:")
    for name, noten in schueler_noten.items():
        durchschnitt = sum(noten) / len(noten)
        print(f"  {name}: {durchschnitt:.2f} (aus {len(noten)} Noten)")
    
    return f"Durchschnitte für {len(schueler_noten)} Schüler berechnet"

# Test der Funktion
print(berechne_durchschnittsnoten())

## 📚 Vergleich: CSV vs. andere Formate

```python
# Zelle 9: Praktischer Formatvergleich
def vergleiche_datenformate():
    """Vergleicht CSV mit anderen Datenformaten anhand praktischer Beispiele"""
    print("\n📚 DATENFORMAT-VERGLEICH")
    print("=" * 32)
    
    # Beispieldaten in verschiedenen Formaten
    print("🧑‍🎓 STUDENT-DATEN IN VERSCHIEDENEN FORMATEN:")
    
    print("\n1️⃣ CSV-FORMAT:")
    csv_beispiel = """Name,Alter,Noten
Anna,20,"[1.3, 2.1, 1.8]"
Bob,19,"[2.4, 1.9, 2.0]" """
    print(csv_beispiel)
    print("   ✅ Einfach für Tabellen")
    print("   ❌ Probleme mit Listen in Zellen")
    
    print("\n2️⃣ JSON-FORMAT:")
    json_beispiel = """{
  "studenten": [
    {
      "name": "Anna",
      "alter": 20,
      "noten": [1.3, 2.1, 1.8],
      "adresse": {
        "stadt": "Berlin",
        "plz": "10115"
      }
    }
  ]
}"""
    print(json_beispiel)
    print("   ✅ Verschachtelte Strukturen möglich")
    print("   ✅ Listen und Dictionaries natürlich")
    print("   ❌ Größer als CSV")
    
    print("\n3️⃣ SQLITE-TABELLE:")
    sql_beispiel = """CREATE TABLE studenten (
    name TEXT,
    alter INTEGER,
    stadt TEXT
);

CREATE TABLE noten (
    student_name TEXT,
    fach TEXT,
    note REAL
);"""
    print(sql_beispiel)
    print("   ✅ Beziehungen zwischen Tabellen")
    print("   ✅ Komplexe Abfragen: WHERE, JOIN, GROUP BY")
    print("   ❌ Komplexer für einfache Daten")
    
    entscheidungsmatrix = {
        "Einfache Tabelle": "CSV",
        "Excel-Export": "CSV", 
        "Verschachtelte Daten": "JSON",
        "Konfigurationsdatei": "JSON",
        "Große Datenmengen": "SQLite",
        "Komplexe Abfragen": "SQLite",
        "Beziehungen zwischen Daten": "SQLite"
    }
    
    print("\n🎯 ENTSCHEIDUNGSHILFE:")
    for anwendung, format in entscheidungsmatrix.items():
        print(f"   {anwendung}: → {format}")
    
    return "Formatvergleich abgeschlossen"

# Vergleich zeigen
vergleiche_datenformate()
```

### 🎯 Wichtige Erkenntnisse:

- **CSV ist perfekt für Tabellendaten** - wie eine digitale Excel-Tabelle
- **newline="" und encoding="utf-8"** sind essentiell für saubere CSV-Dateien
- **csv-Modul verwenden** - niemals manuell mit Kommas arbeiten
- **DictReader/DictWriter** macht Code verständlicher durch benannte Spalten
- **Append-Modus** erweitert bestehende Dateien ohne Überschreibung
- **Datentypen beachten** - CSV-Daten sind immer Strings

### 📚 Pattern-Sammlung:
1. **CSV schreiben:** `csv.writer()` mit `newline=""` 
2. **CSV lesen:** `csv.reader()` mit Schleife
3. **DictWriter:** `writeheader()` + `writerow(dict)`
4. **DictReader:** Zugriff über `zeile['Spaltenname']`
5. **Append:** Modus `"a"` für Erweiterung
6. **Numerisch:** `float()` für Berechnungen
