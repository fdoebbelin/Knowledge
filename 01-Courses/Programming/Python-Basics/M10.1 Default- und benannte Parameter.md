## 1.1 Grundlagen der Default-Parameter

Default-Parameter ermöglichen es, Funktionen mit optionalen Parametern zu erstellen.

### Einfache Default-Parameter

```python
def begrüße_person(name, gruß="Hallo"):
    """Begrüßt eine Person mit einem anpassbaren Gruß."""
    return f"{gruß}, {name}!"

# Tests mit verschiedenen Aufrufen
print(begrüße_person("Anna"))  # Verwendet Standard-Gruß
print(begrüße_person("Bob", "Hi"))  # Eigener Gruß
print(begrüße_person("Charlie", "Guten Tag"))
```

### Mehrere Default-Parameter

```python
def erstelle_profil(name, alter=25, stadt="Berlin", beruf="Student"):
    """Erstellt ein Benutzerprofil mit Standard-Werten."""
    profil = {
        "name": name,
        "alter": alter,
        "stadt": stadt,
        "beruf": beruf
    }
    return profil

# Verschiedene Aufrufvarianten
print(erstelle_profil("Max"))
print(erstelle_profil("Lisa", 30))
print(erstelle_profil("Tom", 28, "München"))
```

### Praktisches Beispiel: Konfigurationsfunktion

```python
def konfiguriere_app(name, debug=False, port=8000, host="localhost"):
    """Konfiguriert eine Anwendung mit Standard-Einstellungen."""
    config = {
        "app_name": name,
        "debug_mode": debug,
        "server_port": port,
        "server_host": host
    }
    
    print(f"App '{name}' konfiguriert:")
    for key, value in config.items():
        print(f"  {key}: {value}")
    
    return config

# Standard-Konfiguration
konfiguriere_app("MeineApp")
```

```python
# Entwicklungs-Konfiguration
konfiguriere_app("MeineApp", debug=True, port=3000)
```

## 1.2 Benannte Parameter und Reihenfolge

Benannte Parameter ermöglichen es, Parameter in beliebiger Reihenfolge zu übergeben.

### Keyword-Arguments verwenden

```python
def berechne_rechteck(länge, breite, einheit="cm"):
    """Berechnet Fläche und Umfang eines Rechtecks."""
    fläche = länge * breite
    umfang = 2 * (länge + breite)
    
    return {
        "fläche": f"{fläche} {einheit}²",
        "umfang": f"{umfang} {einheit}",
        "abmessungen": f"{länge} x {breite} {einheit}"
    }

# Verschiedene Aufrufarten
print(berechne_rechteck(5, 3))
print(berechne_rechteck(breite=4, länge=6))  # Reihenfolge egal
print(berechne_rechteck(länge=10, breite=2, einheit="m"))
```

### Mischung von Positions- und Keyword-Arguments

```python
def formatiere_text(text, groß=False, fett=False, kursiv=False):
    """Formatiert Text mit verschiedenen Optionen."""
    ergebnis = text
    
    if groß:
        ergebnis = ergebnis.upper()
    if fett:
        ergebnis = f"**{ergebnis}**"
    if kursiv:
        ergebnis = f"*{ergebnis}*"
    
    return ergebnis

# Verschiedene Kombinationen
print(formatiere_text("Python"))
print(formatiere_text("Python", groß=True))
print(formatiere_text("Python", fett=True, kursiv=True))
print(formatiere_text("Python", kursiv=True, groß=True, fett=True))
```

### Zwingend benannte Parameter (Keyword-only)

```python
def erstelle_benutzer(name, *, email, alter, aktiv=True):
    """Erstellt einen Benutzer. Email und Alter müssen benannt werden."""
    benutzer = {
        "name": name,
        "email": email,
        "alter": alter,
        "aktiv": aktiv
    }
    return benutzer

# Korrekter Aufruf
benutzer = erstelle_benutzer("Anna", email="anna@test.de", alter=25)
print(benutzer)
```

```python
# Fehlerhafte Aufrufe (auskommentiert, würden Fehler verursachen)
# erstelle_benutzer("Bob", "bob@test.de", 30)  # Fehler: email muss benannt werden
# Stattdessen:
benutzer2 = erstelle_benutzer("Bob", email="bob@test.de", alter=30, aktiv=False)
print(benutzer2)
```

## 1.3 Häufige Fallstricke

### Problematisch: Mutable Default-Parameter

```python
# FALSCH - Nicht so machen!
def füge_element_hinzu_falsch(element, liste=[]):
    """GEFÄHRLICH: Mutable Default-Parameter!"""
    liste.append(element)
    return liste

# Problemdemonstration
print("Problematisches Verhalten:")
liste1 = füge_element_hinzu_falsch("A")
print(f"Liste 1: {liste1}")

liste2 = füge_element_hinzu_falsch("B")  # Überraschung!
print(f"Liste 2: {liste2}")  # Enthält auch "A"!

liste3 = füge_element_hinzu_falsch("C")
print(f"Liste 3: {liste3}")  # Enthält A, B und C!
```

### Korrekte Lösung mit None

```python
# RICHTIG - So sollte es gemacht werden
def füge_element_hinzu_korrekt(element, liste=None):
    """Sichere Implementierung mit None als Default."""
    if liste is None:
        liste = []  # Neue Liste für jeden Aufruf
    
    liste.append(element)
    return liste

# Korrekte Demonstration
print("\nKorrektes Verhalten:")
liste1 = füge_element_hinzu_korrekt("A")
print(f"Liste 1: {liste1}")

liste2 = füge_element_hinzu_korrekt("B")
print(f"Liste 2: {liste2}")  # Nur "B"

liste3 = füge_element_hinzu_korrekt("C")
print(f"Liste 3: {liste3}")  # Nur "C"
```

### Eigene Liste übergeben

```python
# Mit bestehender Liste arbeiten
meine_liste = ["Start"]
neue_liste = füge_element_hinzu_korrekt("Ende", meine_liste)
print(f"Erweiterte Liste: {neue_liste}")
print(f"Original verändert: {meine_liste}")  # Auch verändert!
```

### Best Practice: Dictionary-Default-Parameter

```python
def erstelle_einstellungen(name, einstellungen=None):
    """Erstellt Benutzereinstellungen mit sicheren Defaults."""
    # Standard-Einstellungen
    defaults = {
        "theme": "hell",
        "sprache": "de",
        "benachrichtigungen": True,
        "auto_save": True
    }
    
    # Benutzer-Einstellungen übernehmen
    if einstellungen is None:
        einstellungen = {}
    
    # Defaults mit Benutzer-Einstellungen kombinieren
    finale_einstellungen = defaults.copy()
    finale_einstellungen.update(einstellungen)
    finale_einstellungen["benutzername"] = name
    
    return finale_einstellungen

# Tests
print(erstelle_einstellungen("Anna"))
print(erstelle_einstellungen("Bob", {"theme": "dunkel", "sprache": "en"}))
```
