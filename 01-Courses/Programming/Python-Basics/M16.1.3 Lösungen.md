Noten-Modul für Schulverwaltung
Dieses Modul stellt Funktionen für Notenberechnungen und -bewertungen bereit.
"""

def berechne_durchschnitt(noten_liste):
    """Berechnet den Durchschnitt einer Notenliste"""
    if not noten_liste:
        return 0.0
    return round(sum(noten_liste) / len(noten_liste), 2)

def note_zu_text(note):
    """Wandelt eine Zahlennote in Textnote um"""
    if note <= 1.5:
        return "sehr gut"
    elif note <= 2.5:
        return "gut"
    elif note <= 3.5:
        return "befriedigend"
    elif note <= 4.0:
        return "ausreichend"
    else:
        return "ungenügend"

def bestanden(note):
    """Prüft ob eine Note als bestanden gilt (< 4.0)"""
    return note < 4.0

def beste_note(noten_liste):
    """Findet die beste (niedrigste) Note in einer Liste"""
    if not noten_liste:
        return None
    return min(noten_liste)

def schlechteste_note(noten_liste):
    """Findet die schlechteste (höchste) Note in einer Liste"""
    if not noten_liste:
        return None
    return max(noten_liste)

Dieses Modul verwaltet Studentendaten und stellt Funktionen zur Organisation bereit.
"""

def erstelle_student(name, alter, studiengang):
    """Erstellt ein Dictionary für einen Studenten"""
    return {
        "name": name,
        "alter": alter,
        "studiengang": studiengang
    }

def formatiere_student(student_dict):
    """Erstellt eine schöne formatierte Ausgabe eines Studenten"""
    name = student_dict["name"]
    alter = student_dict["alter"]
    studiengang = student_dict["studiengang"]
    return f"{name} ({alter} Jahre, {studiengang})"

def student_volljaehrig(student_dict):
    """Prüft ob ein Student volljährig ist (18 oder älter)"""
    return student_dict["alter"] >= 18

def studenten_nach_alter_sortieren(studenten_liste):
    """Sortiert eine Liste von Studenten nach Alter"""
    return sorted(studenten_liste, key=lambda student: student["alter"])

def finde_student_nach_name(studenten_liste, name):
    """Sucht einen Studenten in der Liste nach Namen"""
    for student in studenten_liste:
        if student["name"].lower() == name.lower():
            return student
    return None

def anzahl_studenten_pro_studiengang(studenten_liste):
    """Zählt die Anzahl Studenten pro Studiengang"""
    zaehlung = {}
    for student in studenten_liste:
        studiengang = student["studiengang"]
        zaehlung[studiengang] = zaehlung.get(studiengang, 0) + 1
    return zaehlung

# Zelle 3.1: Modul für Berichte und Statistiken erstellen
def erstelle_berichts_modul():
    """Erstellt ein Modul für Berichte und Statistiken"""
    
    berichte_modul_inhalt = '''"""
Berichte-Modul für Schulverwaltung
Dieses Modul erstellt verschiedene Berichte und Statistiken.
"""

def klassenuebersicht(studenten_liste):
    """Erstellt eine Übersicht aller Studenten in der Klasse"""
    if not studenten_liste:
        return "Keine Studenten vorhanden."
    
    uebersicht = f"Klassenstärke: {len(studenten_liste)} Studenten\\n"
    for i, student in enumerate(studenten_liste, 1):
        uebersicht += f"{i}. {student['name']} ({student['alter']} Jahre, {student['studiengang']})\\n"
    return uebersicht.strip()

def noten_statistik(alle_noten):
    """Berechnet Statistiken für alle Noten"""
    if not alle_noten:
        return {"min": 0, "max": 0, "durchschnitt": 0, "anzahl": 0}
    
    return {
        "min": min(alle_noten),
        "max": max(alle_noten),
        "durchschnitt": round(sum(alle_noten) / len(alle_noten), 2),
        "anzahl": len(alle_noten)
    }

def bestenliste(studenten_mit_noten, anzahl=3):
    """Erstellt eine Bestenliste der Top-Studenten"""
    if not studenten_mit_noten:
        return []
    
    # Nach Durchschnitt sortieren (beste zuerst)
    sortiert = sorted(studenten_mit_noten, 
                     key=lambda x: sum(x["noten"]) / len(x["noten"]))
    
    bestenliste = []
    for i in range(min(anzahl, len(sortiert))):
        student = sortiert[i]
        durchschnitt = round(sum(student["noten"]) / len(student["noten"]), 2)
        bestenliste.append({
            "platz": i + 1,
            "name": student["student"]["name"],
            "durchschnitt": durchschnitt
        })
    return bestenliste

def durchfaller_liste(studenten_mit_noten):
    """Erstellt eine Liste aller Studenten mit Note > 4.0"""
    durchfaller = []
    for eintrag in studenten_mit_noten:
        durchschnitt = sum(eintrag["noten"]) / len(eintrag["noten"])
        if durchschnitt > 4.0:
            durchfaller.append({
                "name": eintrag["student"]["name"],
                "durchschnitt": round(durchschnitt, 2)
            })
    return durchfaller

def studiengang_bericht(studenten_liste):
    """Erstellt einen Bericht über die Studiengang-Verteilung"""
    if not studenten_liste:
        return {}
    
    studiengang_stats = {}
    for student in studenten_liste:
        studiengang = student["studiengang"]
        if studiengang not in studiengang_stats:
            studiengang_stats[studiengang] = {
                "anzahl": 0,
                "studenten": []
            }
        studiengang_stats[studiengang]["anzahl"] += 1
        studiengang_stats[studiengang]["studenten"].append(student["name"])
    
    return studiengang_stats

def generiere_zeugnis(student, noten):
    """Generiert ein individuelles Zeugnis für einen Studenten"""
    if not noten:
        return "Keine Noten vorhanden"
    
    durchschnitt = round(sum(noten) / len(noten), 2)
    beste_note = min(noten)
    schlechteste_note = max(noten)

=== ZEUGNIS ===
Student: {student["name"]}
Alter: {student["alter"]} Jahre
Studiengang: {student["studiengang"]}

Noten: {noten}
Durchschnitt: {durchschnitt}
Beste Note: {beste_note}
Schlechteste Note: {schlechteste_note}
Status: {"Bestanden" if durchschnitt < 4.0 else "Nicht bestanden"}
===============
"""
    return zeugnis.strip()
