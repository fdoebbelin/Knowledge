# Zelle 5.1: List vs. Generator Memory-Verbrauch
def vergleiche_speicher_verbrauch():
    """Vergleicht den Speicherverbrauch zwischen Listen und Generatoren"""
    import sys
    
    # Größere Testdaten für deutlichen Unterschied
    große_temperaturen = list(range(-50, 50)) * 1000  # 100.000 Werte
    
    # List Comprehension (verbraucht viel Speicher)
    # Alle Werte werden sofort berechnet und gespeichert
    list_comprehension = [temp * 9/5 + 32 for temp in große_temperaturen if temp > 0]
    
    # Generator Expression (verbraucht wenig Speicher)
    # Nur die "Anleitung" wird gespeichert, keine Werte
    generator_expression = (temp * 9/5 + 32 for temp in große_temperaturen if temp > 0)
    
    # Speicherverbrauch messen
    list_speicher = sys.getsizeof(list_comprehension)
    generator_speicher = sys.getsizeof(generator_expression)
    
    # Ersparnis berechnen
    ersparnis_faktor = list_speicher / generator_speicher if generator_speicher > 0 else 0
    
    print("\n🧪 SCHRITT 5.1: Speicher-Vergleich")
    print("=" * 35)
    print(f"Testdaten: {len(große_temperaturen):,} Temperaturen")
    print(f"List Comprehension: {list_speicher:,} Bytes")
    print(f"Generator Expression: {generator_speicher:,} Bytes")
    print(f"Speicher-Ersparnis: {ersparnis_faktor:.1f}x weniger!")
    print(f"Verarbeitete Elemente: {len(list_comprehension):,}")
    
    print("✅ Generator spart enormen Speicher bei großen Datenmengen!")
    
    return {
        'list_speicher': list_speicher,
        'generator_speicher': generator_speicher,
        'ersparnis_faktor': ersparnis_faktor,
        'anzahl_elemente': len(list_comprehension)
    }

# Zelle 5.2: Erstellungs-Performance vergleichen
def vergleiche_erstellungs_performance():
    """Vergleicht die Performance bei der Erstellung"""
    import time
    
    test_daten = list(range(100000))
    
    # List Comprehension Zeit messen
    # Alle Werte werden sofort berechnet
    start = time.time()
    liste_result = [x * 2 + 1 for x in test_daten if x % 2 == 0]
    list_zeit = time.time() - start
    
    # Generator Expression Zeit messen  
    # Nur die "Berechnungsanleitung" wird erstellt
    start = time.time()
    generator_result = (x * 2 + 1 for x in test_daten if x % 2 == 0)
    gen_zeit = time.time() - start
    
    # Speedup berechnen
    speedup = list_zeit / gen_zeit if gen_zeit > 0 else float('inf')
    
    print("\n🧪 SCHRITT 5.2: Erstellungs-Performance")
    print("=" * 40)
    print(f"Testdaten: {len(test_daten):,} Elemente")
    print(f"List Comprehension: {list_zeit:.6f} Sekunden")
    print(f"Generator Expression: {gen_zeit:.6f} Sekunden")
    print(f"Generator ist {speedup:.0f}x schneller bei der Erstellung!")
    print(f"Verarbeitete Elemente: {len(liste_result):,}")
    
    # Generator nutzen (Beispiel)
    erste_5_gen = [next(generator_result) for _ in range(5)]
    print(f"Erste 5 Generator-Werte: {erste_5_gen}")
    
    print("✅ Generator ist extrem schnell bei der Erstellung!")
    
    return {
        'list_zeit': list_zeit,
        'generator_zeit': gen_zeit,
        'speedup': speedup,
        'list_laenge': len(liste_result)
    }

### 🎯 Wichtige Erkenntnisse:

- **Memory-Effizienz:** Generators sparen bis zu 90% Speicher bei großen Datenmengen
- **Lazy Evaluation:** Werte werden nur bei Bedarf berechnet, nicht alle auf einmal
- **Pipeline-Processing:** Elegant verkettbare Datenverarbeitungsschritte
- **Einmalige Nutzung:** Generators sind nach dem Durchlauf "erschöpft"
- **Built-in Integration:** Perfekte Zusammenarbeit mit sum(), max(), min(), any(), all()
- **Syntax-Einfachheit:** Nur `()` statt `[]` macht den Unterschied

### 📚 Pattern-Sammlung:
1. **Einfacher Generator:** `(expression for item in iterable)`
2. **Mit Filter:** `(expression for item in iterable if condition)`
3. **Pipeline:** `gen2 = (transform(x) for x in gen1 if condition)`
4. **Built-in Integration:** `result = sum(expression for item in iterable)`
5. **Memory-Vergleich:** `sys.getsizeof()` zum Messen verwenden
