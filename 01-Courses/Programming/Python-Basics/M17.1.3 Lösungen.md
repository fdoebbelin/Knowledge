# Zelle 3.1: Kompletten schlechten Code nach PEP 8 überarbeiten
def ueberarbeite_taschenrechner():
    """Überarbeitet schlechten Taschenrechner-Code nach PEP 8"""
    
    # ✅ Komplett neu geschrieben nach PEP 8
    def berechne_operation(wert_a, wert_b, operator):
        """
        Führt eine mathematische Operation aus.
        
        Args:
            wert_a: Erste Zahl
            wert_b: Zweite Zahl
            operator: Operation ('+', '-', '*', '/')
        
        Returns:
            Ergebnis der Berechnung oder Fehlermeldung
        """
        if operator == "+":
            return wert_a + wert_b
        elif operator == "-":
            return wert_a - wert_b
        elif operator == "*":
            return wert_a * wert_b
        elif operator == "/":
            if wert_b != 0:
                return wert_a / wert_b
            else:
                return "Fehler: Division durch Null"
        else:
            return "Fehler: Unbekannter Operator"
    
    # Taschenrechner testen
    test_rechnungen = [
        (10, 5, "+"),
        (10, 5, "-"),
        (10, 5, "*"),
        (10, 5, "/"),
        (10, 0, "/")
    ]
    
    ergebnisse = []
    for a, b, op in test_rechnungen:
        resultat = berechne_operation(a, b, op)
        ergebnisse.append(f"{a} {op} {b} = {resultat}")
    
    print("\n🧮 SCHRITT 3.1: Taschenrechner überarbeiten")
    print("=" * 50)
    print("✅ Funktionsname: snake_case (berechne_operation)")
    print("✅ Parameter: aussagekräftige Namen")
    print("✅ Einrückung: 4 Leerzeichen konsistent")
    print("✅ Leerzeichen: um alle Operatoren")
    print("✅ Docstring: vollständige Dokumentation")
    for ergebnis in ergebnisse:
        print(f"   {ergebnis}")
    
    return f"Taschenrechner überarbeitet: {len(ergebnisse)} Tests erfolgreich"

# Test der Funktion

    """Führt ein Code-Review mit PEP 8 Checkliste durch"""
    
    # Gefundene PEP 8 Probleme im ursprünglichen Code
    gefundene_probleme = [
        "MaxValue sollte MAX_VALUE sein (Konstante)",
        "CheckValue sollte check_value sein (Funktion)",
        "dataProcessor sollte DataProcessor sein (Klasse)",
        "processData sollte process_data sein (Methode)",
        "Leerzeichen fehlen: val>MaxValue sollte val > MaxValue",

        "Leerzeichen fehlen: item*2 sollte item * 2",
        "Einrückung: return-Statements nicht eingerückt"
    ]
    
    # ✅ Korrigierte Version
    MAX_VALUE = 100
    
    def check_value(val):
        """Prüft ob ein Wert im erlaubten Bereich liegt"""
        return val <= MAX_VALUE
    
    class DataProcessor:
        """Verarbeitet Daten nach bestimmten Regeln"""

            """Verarbeitet eine Datenliste"""

# Zelle 4.2: Neuen Code vollständig PEP 8-konform schreiben
def erstelle_perfekten_code():
    """Erstellt von Grund auf PEP 8-konformen Code"""
    
    # ✅ Konstanten
    MAX_NOTE = 6.0
    MIN_NOTE = 1.0
    BESTEHEN_GRENZE = 4.0
    
    # ✅ Funktionen
    def berechne_durchschnitt(noten_liste):
        """
        Berechnet den Durchschnitt einer Notenliste.
        
        Args:
            noten_liste: Liste mit Noten zwischen 1.0 und 6.0
        
        Returns:
            Durchschnittsnote gerundet auf 2 Dezimalstellen
        """
        if not noten_liste:
            return 0.0
        return round(sum(noten_liste) / len(noten_liste), 2)
    
    def bewerte_note(note):
        """
        Bewertet eine Note als Text.
        
        Args:
            note: Zahlennote zwischen 1.0 und 6.0
        
        Returns:
            Textuelle Bewertung der Note
        """
        if note <= 1.5:
            return "sehr gut"
        elif note <= 2.5:
            return "gut"
        elif note <= 3.5:
            return "befriedigend"
        elif note < BESTEHEN_GRENZE:
            return "ausreichend"
        else:
            return "ungenügend"
    
    # ✅ Klasse
    class NotenManager:
        """
        Verwaltet Noten für Studenten.
        
        Attributes:
            studenten_noten: Dictionary mit Studenten und ihren Noten
        """
        
        def __init__(self):
            """Initialisiert einen leeren NotenManager."""
            self.studenten_noten = {}
        
        def fuege_note_hinzu(self, student_name, note):
            """
            Fügt eine Note für einen Studenten hinzu.
            
            Args:
                student_name: Name des Studenten
                note: Note zwischen MIN_NOTE und MAX_NOTE
            
            Returns:
                True wenn erfolgreich, False bei ungültiger Note
            """
            if MIN_NOTE <= note <= MAX_NOTE:
                if student_name not in self.studenten_noten:
                    self.studenten_noten[student_name] = []
                self.studenten_noten[student_name].append(note)
                return True
            return False
        
        def get_durchschnitt(self, student_name):
            """
            Berechnet Durchschnitt für einen Studenten.
            
            Args:
                student_name: Name des Studenten
            
            Returns:
                Durchschnittsnote oder None wenn Student nicht existiert
            """
            if student_name in self.studenten_noten:
                return berechne_durchschnitt(
                    self.studenten_noten[student_name]
                )
            return None
    
    # ✅ System testen
    noten_system = NotenManager()
    
    # Noten hinzufügen
    noten_system.fuege_note_hinzu("Anna Schmidt", 1.3)
    noten_system.fuege_note_hinzu("Anna Schmidt", 1.7)
    noten_system.fuege_note_hinzu("Max Weber", 2.3)
    noten_system.fuege_note_hinzu("Max Weber", 2.8)
    
    # Auswertungen
    anna_schnitt = noten_system.get_durchschnitt("Anna Schmidt")
    max_schnitt = noten_system.get_durchschnitt("Max Weber")
    anna_bewertung = bewerte_note(anna_schnitt)
    max_bewertung = bewerte_note(max_schnitt)
    
    print("\n⭐ SCHRITT 4.2: Perfekten Code erstellen")
    print("=" * 48)
    print("✅ Alle PEP 8 Regeln befolgt:")
    print("   - Konstanten: UPPER_CASE")
    print("   - Funktionen: snake_case mit Docstrings")
    print("   - Klasse: PascalCase mit vollständiger Dokumentation")
    print("   - 4 Leerzeichen Einrückung überall")
    print("   - Leerzeichen um alle Operatoren")
    print("   - Zeilen unter 79 Zeichen")
    print("✅ Noten-System getestet:")
    print(f"   Anna Schmidt: ⌀ {anna_schnitt} ({anna_bewertung})")
    print(f"   Max Weber: ⌀ {max_schnitt} ({max_bewertung})")
    
    return f"Perfekter Code: {len(noten_system.studenten_noten)} Studenten verwaltet"
