## 1. Listen als Parameter - Das Problem
### Problem: Listen werden verändert!

```python
def demo_listen_problem():
    """Zeigt das Hauptproblem mit Listen als Parameter"""
    
    def verdopple_zahlen(zahlen_liste):
        """Verdoppelt alle Zahlen in der Liste"""
        for i in range(len(zahlen_liste)):
            zahlen_liste[i] = zahlen_liste[i] * 2
    
    # Test
    meine_zahlen = [1, 2, 3, 4, 5]
    print(f"Vorher: {meine_zahlen}")
    
    verdopple_zahlen(meine_zahlen)
    
    print(f"Nachher: {meine_zahlen}")  # ÜBERRASCHUNG: Verändert!
    print("😱 Die ursprüngliche Liste wurde verändert!")

demo_listen_problem()
```

**Ausgabe:**
```
Vorher: [1, 2, 3, 4, 5]
Nachher: [2, 4, 6, 8, 10]
😱 Die ursprüngliche Liste wurde verändert!
```

### Warum passiert das?

```python
def erklaere_referenzen():
    """Erklärt warum Listen verändert werden"""
    
    # Bei Zahlen passiert das NICHT
    def aendere_zahl(x):
        x = x + 100
        print(f"In der Funktion: x = {x}")
    
    # Bei Listen passiert das SCHON
    def aendere_liste(liste):
        liste.append("NEU")
        print(f"In der Funktion: {liste}")
    
    # Tests
    print("=== ZAHLEN (sicher) ===")
    zahl = 5
    print(f"Vor Funktionsaufruf: {zahl}")
    aendere_zahl(zahl)
    print(f"Nach Funktionsaufruf: {zahl}")  # Unverändert
    
    print("\n=== LISTEN (gefährlich) ===")
    liste = [1, 2, 3]
    print(f"Vor Funktionsaufruf: {liste}")
    aendere_liste(liste)
    print(f"Nach Funktionsaufruf: {liste}")  # Verändert!

erklaere_referenzen()
```

## 2. Die Lösung: Sichere Kopien
### Methode 1: copy() verwenden

```python
def sichere_funktion_v1():
    """Zeigt wie man Listen sicher kopiert"""
    
    def verdopple_zahlen_sicher(zahlen_liste):
        """Verdoppelt Zahlen OHNE das Original zu ändern"""
        # Schritt 1: Kopie erstellen
        kopie = zahlen_liste.copy()
        
        # Schritt 2: Kopie bearbeiten
        for i in range(len(kopie)):
            kopie[i] = kopie[i] * 2
        
        # Schritt 3: Kopie zurückgeben
        return kopie
    
    # Test
    original = [1, 2, 3, 4, 5]
    print(f"Original vorher: {original}")
    
    verdoppelt = verdopple_zahlen_sicher(original)
    
    print(f"Original nachher: {original}")  # Unverändert!
    print(f"Neues Ergebnis: {verdoppelt}")
    print("✅ Original ist sicher!")

sichere_funktion_v1()
```

### Methode 2: Neue Liste erstellen

```python
def sichere_funktion_v2():
    """Alternative: Neue Liste mit List Comprehension"""
    
    def verdopple_zahlen_elegant(zahlen_liste):
        """Elegante Version mit List Comprehension"""
        return [zahl * 2 for zahl in zahlen_liste]
    
    # Test
    original = [1, 2, 3, 4, 5]
    verdoppelt = verdopple_zahlen_elegant(original)
    
    print(f"Original: {original}")
    print(f"Verdoppelt: {verdoppelt}")
    print("✅ Kurz und sicher!")

sichere_funktion_v2()
```

## 3. Tupel für mehrere Rückgabewerte
### Problem: Mehrere Werte zurückgeben

```python
def demo_multiple_returns():
    """Zeigt wie man mehrere Werte elegant zurückgibt"""
    
    def berechne_kreis(radius):
        """Berechnet Durchmesser, Umfang und Fläche"""
        durchmesser = radius * 2
        umfang = 2 * 3.14159 * radius
        flaeche = 3.14159 * radius * radius
        
        # Alle drei Werte als Tupel zurückgeben
        return durchmesser, umfang, flaeche
    
    # Verwendung
    r = 5
    d, u, f = berechne_kreis(r)  # Tupel "auspacken"
    
    print(f"Kreis mit Radius {r}:")
    print(f"  Durchmesser: {d}")
    print(f"  Umfang: {u:.2f}")
    print(f"  Fläche: {f:.2f}")
    
    # Alternative: Als Tupel verwenden
    kreis_daten = berechne_kreis(3)
    print(f"\nKreis-Daten als Tupel: {kreis_daten}")

demo_multiple_returns()
```

### Praktisches Beispiel: Min und Max finden

```python
def demo_min_max():
    """Findet Minimum und Maximum mit ihren Positionen"""
    
    def finde_min_max(zahlen):
        """Findet Min/Max-Werte und ihre Positionen"""
        if not zahlen:  # Leere Liste prüfen
            return None
        
        min_wert = min(zahlen)
        max_wert = max(zahlen)
        min_position = zahlen.index(min_wert)
        max_position = zahlen.index(max_wert)
        
        return min_wert, min_position, max_wert, max_position
    
    # Test
    zahlen = [3, 7, 2, 9, 1, 8, 4]
    ergebnis = finde_min_max(zahlen)
    
    if ergebnis:
        min_w, min_pos, max_w, max_pos = ergebnis
        print(f"Zahlen: {zahlen}")
        print(f"Minimum: {min_w} an Position {min_pos}")
        print(f"Maximum: {max_w} an Position {max_pos}")

demo_min_max()
```

## 4. Erweiterte Funktionen mit *args
### Beliebig viele Listen verarbeiten

```python
def demo_args():
    """Zeigt *args mit Listen"""
    
    def kombiniere_listen(*listen):
        """Kombiniert beliebig viele Listen"""
        alle_zahlen = []
        
        for liste in listen:
            alle_zahlen.extend(liste)  # Alle Elemente hinzufügen
        
        return alle_zahlen
    
    # Tests
    liste1 = [1, 2, 3]
    liste2 = [4, 5]
    liste3 = [6, 7, 8, 9]
    
    kombiniert = kombiniere_listen(liste1, liste2, liste3)
    print(f"Liste 1: {liste1}")
    print(f"Liste 2: {liste2}")
    print(f"Liste 3: {liste3}")
    print(f"Kombiniert: {kombiniert}")
    
    # Funktioniert mit beliebig vielen Listen
    viele_listen = kombiniere_listen([1], [2, 3], [4], [5, 6, 7])
    print(f"Viele Listen: {viele_listen}")

demo_args()
```

## Häufige Fehlerquellen

```python
def haeufige_fehler():
    """Zeigt typische Anfängerfehler"""
    
    print("❌ HÄUFIGE FEHLER:")
    print()
    
    # Fehler 1: Listen ungewollt ändern
    def falscher_ansatz(liste):
        liste.sort()  # Ändert das Original!
        return liste[0]  # Erstes Element
    
    # Richtig:
    def richtiger_ansatz(liste):
        kopie = liste.copy()
        kopie.sort()
        return kopie[0]
    
    test_liste = [3, 1, 4, 2]
    print(f"Test-Liste: {test_liste}")
    
    # Fehler demonstrieren
    erstes = falscher_ansatz(test_liste)
    print(f"Nach falschem Ansatz: {test_liste}")  # Verändert!
    
    # Zurücksetzen für richtigen Test
    test_liste = [3, 1, 4, 2]
    erstes = richtiger_ansatz(test_liste)
    print(f"Nach richtigem Ansatz: {test_liste}")  # Unverändert!

haeufige_fehler()
```
