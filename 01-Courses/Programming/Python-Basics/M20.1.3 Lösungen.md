def create_base_system():
    """Erstellt das Basis-System für die Studentenverwaltung"""

studenten_liste = []

def student_hinzufuegen(name, alter, studiengang):
    """Fügt einen neuen Studenten zur Liste hinzu"""
    student = {
        "name": name,
        "alter": alter,
        "studiengang": studiengang,
        "id": len(studenten_liste) + 1

    return f"Student {name} hinzugefügt (ID: {student['id']})"

def alle_studenten_anzeigen():

    for student in studenten_liste:
        ausgabe += f"ID {student['id']}: {student['name']} ({student['alter']} Jahre, {student['studiengang']})\\n"
    return ausgabe

if __name__ == "__main__":
    print("Studentenverwaltung gestartet!")
    
    # Beispiel-Studenten hinzufügen
    print(student_hinzufuegen("Anna Schmidt", 22, "Informatik"))
    print(student_hinzufuegen("Tom Müller", 21, "BWL"))
    print(alle_studenten_anzeigen())
'''
    
    with open("studentenverwaltung.py", "w", encoding="utf-8") as f:

def add_export_functions():
    """Fügt CSV-Export/Import-Funktionalität hinzu"""
    
    export_extension = '''
def studenten_zu_csv_exportieren(dateiname="studenten_export.csv"):
    """Exportiert alle Studenten in eine CSV-Datei"""
    if not studenten_liste:
        return "Keine Studenten zum Exportieren vorhanden"
    
    csv_inhalt = "ID,Name,Alter,Studiengang\\n"
    for student in studenten_liste:
        csv_inhalt += f"{student['id']},{student['name']},{student['alter']},{student['studiengang']}\\n"
    
    try:
        with open(dateiname, "w", encoding="utf-8") as f:
            f.write(csv_inhalt)
        return f"✅ {len(studenten_liste)} Studenten erfolgreich nach {dateiname} exportiert"
    except Exception as e:
        return f"❌ Fehler beim Export: {e}"

def studenten_aus_csv_importieren(dateiname="studenten_import.csv"):
    """Importiert Studenten aus einer CSV-Datei"""
    try:
        with open(dateiname, "r", encoding="utf-8") as f:
            lines = f.readlines()
        
        if len(lines) <= 1:  # Nur Header oder leer
            return "Keine Daten zum Importieren gefunden"
        
        importiert = 0
        for line in lines[1:]:  # Header überspringen
            line = line.strip()
            if line:
                teile = line.split(",")
                if len(teile) >= 4:
                    name, alter, studiengang = teile[1], int(teile[2]), teile[3]
                    student_hinzufuegen(name, alter, studiengang)
                    importiert += 1
        
        return f"✅ {importiert} Studenten erfolgreich aus {dateiname} importiert"
    except FileNotFoundError:
        return f"❌ Datei {dateiname} nicht gefunden"
    except Exception as e:
        return f"❌ Fehler beim Import: {e}"

def backup_erstellen():
    """Erstellt ein Backup der aktuellen Studentendaten"""
    import datetime
    timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_datei = f"studenten_backup_{timestamp}.csv"
    return studenten_zu_csv_exportieren(backup_datei)
'''
    
    # Basis-Version lesen und erweitern

        'print(alle_studenten_anzeigen())',

    # Export-Funktionen demonstrieren
    print("\\n=== CSV-EXPORT/IMPORT DEMO ===")
    print(studenten_zu_csv_exportieren())
    print(backup_erstellen())'''
