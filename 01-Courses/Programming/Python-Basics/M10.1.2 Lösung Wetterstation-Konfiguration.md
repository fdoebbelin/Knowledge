```python
# Zelle 1: Grundfunktion implementieren
def zeige_wetterbericht(stadt, temperatur, einheit="Celsius", 
                       zeige_gefuehl=True, wetterdetails=None):
    """
    Zeigt einen formatierten Wetterbericht an.
    
    Parameter:
    - stadt: Name der Stadt (Pflicht)
    - temperatur: Aktuelle Temperatur (Pflicht)
    - einheit: Temperatureinheit ("Celsius" oder "Fahrenheit")
    - zeige_gefuehl: Soll gefühlte Temperatur angezeigt werden?
    - wetterdetails: Zusätzliche Wetterinformationen (Liste oder None)
    """
    # Sichere Default-Behandlung für mutable Parameter
    if wetterdetails is None:
        wetterdetails = []
    
    # Header mit Stadt
    print(f"=== WETTERBERICHT {stadt.upper()} ===")
    
    # Temperatur mit Einheit
    einheit_symbol = "°C" if einheit == "Celsius" else "°F"
    print(f"Temperatur: {temperatur}{einheit_symbol}")
    
    # Gefühlte Temperatur (vereinfacht)
    if zeige_gefuehl:
        if temperatur < 10:
            gefuehl = "kalt"
        elif temperatur < 20:
            gefuehl = "kühl"
        elif temperatur < 25:
            gefuehl = "angenehm"
        else:
            gefuehl = "warm"
        print(f"Gefühlte Temperatur: {gefuehl}")
    
    # Wetterdetails anzeigen
    if wetterdetails:
        print(f"Wetter: {', '.join(wetterdetails)}")
    
    # Konfigurationsinfo
    if einheit != "Celsius" or not zeige_gefuehl or wetterdetails:
        print("Konfiguration: Angepasst")
    else:
        print("Konfiguration: Standard")
    
    print()  # Leerzeile für bessere Trennung

# Test der Grundfunktion
print("🧪 Test 1: Grundfunktion mit minimalen Parametern")
zeige_wetterbericht("Berlin", 22.5)

print("🧪 Test 2: Mit Wetterdetails")
zeige_wetterbericht("München", 18.3, wetterdetails=["bewölkt"])

print("🧪 Test 3: Verschiedene Parameter-Kombinationen")
zeige_wetterbericht("Hamburg", 60.44, einheit="Fahrenheit", zeige_gefuehl=False)
```

**💡 Erklärung der Lösung:**
- ✅ **Sichere mutable Defaults:** `if wetterdetails is None:` verhindert das bekannte mutable-Default-Problem
- ✅ **Flexible Parameter:** Kombination aus Pflicht- und optionalen Parametern
- ✅ **String-Formatierung:** f-Strings für saubere, lesbare Ausgabe
- ✅ **Logische Defaults:** Sinnvolle Standardwerte für typische Anwendungsfälle

### Schritt 2: Gefühlte Temperatur berechnen

```python
# Zelle 2: Helper-Funktion für gefühlte Temperatur
def berechne_gefuehlte_temperatur(temperatur, einheit="Celsius"):
    """
    Berechnet eine vereinfachte 'gefühlte Temperatur'.
    
    Vereinfachte Regeln:
    - Unter 10°C: "sehr kalt" 
    - 10-15°C: "kalt"
    - 16-25°C: "angenehm"
    - 26-30°C: "warm"
    - Über 30°C: "sehr warm"
    """
    # Bei Fahrenheit zuerst zu Celsius konvertieren
    if einheit.lower() == "fahrenheit":
        celsius = (temperatur - 32) * 5/9
    else:
        celsius = temperatur
    
    # Gefühlte Temperatur kategorisieren
    if celsius < 10:
        return "sehr kalt"
    elif celsius < 16:
        return "kalt"
    elif celsius < 26:
        return "angenehm"
    elif celsius < 31:
        return "warm"
    else:
        return "sehr warm"

# Test der Helper-Funktion
print("🧪 Test der gefühlten Temperatur-Berechnung:")
test_temps = [5, 12, 20, 28, 35]
for temp in test_temps:
    gefuehl = berechne_gefuehlte_temperatur(temp)
    print(f"   {temp}°C fühlt sich {gefuehl} an")

print("\n🧪 Test mit Fahrenheit:")
fahrenheit_temps = [32, 50, 68, 80, 95]  # 0, 10, 20, 27, 35 Celsius
for temp in fahrenheit_temps:
    gefuehl = berechne_gefuehlte_temperatur(temp, "Fahrenheit")
    celsius_equiv = (temp - 32) * 5/9
    print(f"   {temp}°F ({celsius_equiv:.1f}°C) fühlt sich {gefuehl} an")
```

- ✅ **Wiederverwendbarkeit:** Separate Funktion für bessere Modularität

```python
# Zelle 3: Erweiterte Hauptfunktion
def zeige_wetterbericht(stadt, temperatur, einheit="Celsius", 
                       zeige_gefuehl=True, wetterdetails=None):
    """
    Erweiterte Version mit gefühlter Temperatur und besserer Formatierung.
    """
    # Sichere Default-Behandlung für mutable Parameter
    if wetterdetails is None:
        wetterdetails = []
    
    # Header mit Stadt
    print(f"=== WETTERBERICHT {stadt.upper()} ===")
    
    # Temperatur mit Einheit
    einheit_symbol = "°C" if einheit.lower() == "celsius" else "°F"
    print(f"Temperatur: {temperatur}{einheit_symbol}")
    
    # Gefühlte Temperatur mit verbesserter Berechnung
    if zeige_gefuehl:
        gefuehl = berechne_gefuehlte_temperatur(temperatur, einheit)
        print(f"Gefühlte Temperatur: {gefuehl}")
    
    # Wetterdetails anzeigen (nur wenn vorhanden)
    if wetterdetails:
        if len(wetterdetails) == 1:
            print(f"Wetter: {wetterdetails[0]}")
        else:
            print(f"Wetter: {', '.join(wetterdetails)}")
    
    # Erweiterte Konfigurationsinfo
    config_details = []
    if einheit.lower() != "celsius":
        config_details.append(f"Einheit: {einheit}")
    if not zeige_gefuehl:
        config_details.append("Gefühl: aus")
    if wetterdetails:
        config_details.append(f"Details: {len(wetterdetails)}")
    
    if config_details:
        print(f"Konfiguration: Angepasst ({', '.join(config_details)})")
    else:
        print("Konfiguration: Standard")
    
    print()  # Leerzeile für bessere Trennung

# Test mit verschiedenen Parameterkombinationen
```

```python
def zeige_wetterbericht_erweitert(stadt, temperatur, einheit="Celsius",

                                 farbkodierung=False, kurz_format=False,

    Erweiterte Version mit keyword-only Parametern für spezielle Optionen.
    
    Keyword-only Parameter:
    - farbkodierung: Verwendet Farb-Emojis für Temperatur
    - kurz_format: Zeigt kompakte Version
    - debug_info: Zeigt zusätzliche Debug-Informationen
    """
    # Sichere Default-Behandlung
    if wetterdetails is None:
        wetterdetails = []
    
    # Debug-Info sammeln (wenn aktiviert)
    if debug_info:
        debug_infos = {
            "parameter_count": 7,
            "mutable_defaults": "wetterdetails" if wetterdetails else "keine",
            "keyword_only_used": any([farbkodierung, kurz_format, debug_info]),
            "funktion": "zeige_wetterbericht_erweitert"
        }
    
    # Emoji-Mapping für Farbkodierung
    def get_temperatur_emoji(temp, einheit):
        celsius = temp if einheit.lower() == "celsius" else (temp - 32) * 5/9
        if celsius < 0:
            return "❄️"
        elif celsius < 10:
            return "🧊"
        elif celsius < 20:
            return "🌡️"
        elif celsius < 30:
            return "☀️"
        else:
            return "🔥"
    
    # Kurz-Format: Kompakte einzeilige Ausgabe
    if kurz_format:
        einheit_symbol = "°C" if einheit.lower() == "celsius" else "°F"
        emoji = get_temperatur_emoji(temperatur, einheit) if farbkodierung else ""
        wetter_info = f" ({', '.join(wetterdetails)})" if wetterdetails else ""
        print(f"{emoji} {stadt}: {temperatur}{einheit_symbol}{wetter_info}")
        if debug_info:
            print(f"   DEBUG: {debug_infos}")
        print()
        return
    
    # Standard-Format: Erweiterte Ausgabe
    # Header mit optionalem Emoji
    if farbkodierung:
        emoji = get_temperatur_emoji(temperatur, einheit)
        print(f"{emoji} === WETTERBERICHT {stadt.upper()} === {emoji}")
    else:
        print(f"=== WETTERBERICHT {stadt.upper()} ===")
    
    # Temperatur mit optionalem Emoji
    einheit_symbol = "°C" if einheit.lower() == "celsius" else "°F"
    temp_emoji = get_temperatur_emoji(temperatur, einheit) if farbkodierung else ""
    print(f"Temperatur: {temp_emoji}{temperatur}{einheit_symbol}")
    
    # Gefühlte Temperatur
    if zeige_gefuehl:
        gefuehl = berechne_gefuehlte_temperatur(temperatur, einheit)
        gefuehl_emoji = "😊" if "angenehm" in gefuehl else "🥶" if "kalt" in gefuehl else "🥵" if "warm" in gefuehl else "🌡️"
        if farbkodierung:
            print(f"Gefühlte Temperatur: {gefuehl_emoji} {gefuehl}")
        else:
            print(f"Gefühlte Temperatur: {gefuehl}")
    
    # Wetterdetails mit optionalen Emojis
    if wetterdetails:
        wetter_emojis = {
            "sonnig": "☀️", "bewölkt": "☁️", "regnerisch": "🌧️", 
            "windig": "💨", "schnee": "❄️", "gewitter": "⛈️"
        }
        if farbkodierung:
            details_mit_emoji = []
            for detail in wetterdetails:
                emoji = wetter_emojis.get(detail.lower(), "🌤️")
                details_mit_emoji.append(f"{emoji} {detail}")
            print(f"Wetter: {', '.join(details_mit_emoji)}")
        else:
            print(f"Wetter: {', '.join(wetterdetails)}")
    
    # Konfigurationsinformationen
    config_details = []
    if einheit.lower() != "celsius":
        config_details.append(f"Einheit: {einheit}")
    if not zeige_gefuehl:
        config_details.append("Gefühl: aus")
    if wetterdetails:
        config_details.append(f"Details: {len(wetterdetails)}")
    if farbkodierung:
        config_details.append("Farbe: an")
    if kurz_format:
        config_details.append("Format: kurz")
    
    if config_details:
        print(f"Konfiguration: Angepasst ({', '.join(config_details)})")
    else:
        print("Konfiguration: Standard")
    
    # Debug-Informationen
    if debug_info:
        print(f"\n🔧 DEBUG-INFO:")
        for key, value in debug_infos.items():
            print(f"   {key}: {value}")
        print(f"   aufgerufen_mit: {locals()}")
    
    print()  # Leerzeile

# Test der erweiterten Funktion
```

```python
# Zelle 5: Konfigurationsfunktion mit Presets
def erstelle_wetter_konfiguration(preset="standard", **custom_optionen):
    """
    Erstellt Konfigurationsdictionaries für verschiedene Anwendungsfälle.
    
    Verfügbare Presets:
    - "standard": Ausgewogene Standardkonfiguration
    - "kompakt": Minimale, platzsparende Ausgabe
    - "ausfuehrlich": Detaillierte Ausgabe mit allen Optionen
    - "dashboard": Optimiert für Dashboard-Anzeige
    """
    
    presets = {
        "standard": {
            "einheit": "Celsius",
            "zeige_gefuehl": True,
            "farbkodierung": False,
            "kurz_format": False,
            "debug_info": False
        },
        
        "kompakt": {
            "einheit": "Celsius",
            "zeige_gefuehl": False,
            "farbkodierung": False,
            "kurz_format": True,
            "debug_info": False
        },
        
        "ausfuehrlich": {
            "einheit": "Celsius", 
            "zeige_gefuehl": True,
            "farbkodierung": True,
            "kurz_format": False,
            "debug_info": True
        },
        
        "dashboard": {
            "einheit": "Celsius",
            "zeige_gefuehl": True,
            "farbkodierung": True,
            "kurz_format": False,
            "debug_info": False
        }
    }
    
    # Basis-Konfiguration aus Preset laden
    if preset not in presets:
        print(f"⚠️ Unbekanntes Preset '{preset}', verwende 'standard'")
        preset = "standard"
    
    basis_config = presets[preset].copy()  # Kopie für sichere Modifikation
    
    # Custom-Optionen überschreiben Preset-Werte
    basis_config.update(custom_optionen)
    
    return basis_config

def wetter_mit_konfiguration(stadt, temperatur, wetterdetails=None, 
                           konfiguration=None):
    """
    Wetterberichtfunktion die eine Konfiguration akzeptiert.
    """
    if konfiguration is None:
        konfiguration = erstelle_wetter_konfiguration()
    
    if wetterdetails is None:
        wetterdetails = []
    
    # Aufruf der erweiterten Funktion mit entpackter Konfiguration
    return zeige_wetterbericht_erweitert(
        stadt, temperatur, wetterdetails=wetterdetails, **konfiguration
    )
```

```python
def wetter_mit_validierung(stadt, temperatur, einheit="Celsius", **kwargs):
    """
    Bonus: Parameter-Validierung hinzufügen.

    if not isinstance(stadt, str) or not stadt.strip():

    elif einheit.lower() == "fahrenheit":
```

```python
# Zelle 9: Interaktive Experimente zum Verstehen
def experimentiere_mit_defaults():
    """
    Interaktive Experimente zum Verständnis von Default-Parametern.
    """
    print("🔬 === INTERAKTIVE EXPERIMENTE ===")
    print()
    
    # Experiment 1: Mutable Default Problem demonstrieren
    print("🧪 Experiment 1: Das mutable Default Problem")
    
    def falsche_implementierung(stadt, details=[]):  # FALSCH!
        details.append(f"Besucht: {stadt}")
        return details
    
    def richtige_implementierung(stadt, details=None):  # RICHTIG!
        if details is None:
            details = []
        details.append(f"Besucht: {stadt}")
        return details
    
    print("   Falsche Implementierung (mutable Default):")
    result1 = falsche_implementierung("Berlin")
    print(f"   Erster Aufruf: {result1}")
    result2 = falsche_implementierung("München")  
    print(f"   Zweiter Aufruf: {result2}")  # Achtung: enthält auch Berlin!
    print(f"   ❌ Problem: Alter Zustand bleibt erhalten!")
    
    print("\n   Richtige Implementierung (None Default):")
    result3 = richtige_implementierung("Hamburg")
    print(f"   Erster Aufruf: {result3}")
    result4 = richtige_implementierung("Köln")
    print(f"   Zweiter Aufruf: {result4}")  # Nur Köln enthalten
    print(f"   ✅ Lösung: Jeder Aufruf ist unabhängig!")
    
    print("\n" + "="*50)
    
    # Experiment 2: Keyword-only Parameter
    print("🧪 Experiment 2: Keyword-only Parameter erzwingen")
    
    def normale_funktion(a, b, c=3):
        return f"a={a}, b={b}, c={c}"
    
    def keyword_only_funktion(a, b, *, c=3):
        return f"a={a}, b={b}, c={c}"
    
    print("   Normale Funktion - beide Aufrufe funktionieren:")
    print(f"   normal(1, 2, 3): {normale_funktion(1, 2, 3)}")
    print(f"   normal(1, 2, c=3): {normale_funktion(1, 2, c=3)}")
    
    print("\n   Keyword-only Funktion:")
    print(f"   keyword_only(1, 2, c=3): {keyword_only_funktion(1, 2, c=3)}")
    print("   keyword_only(1, 2, 3): ❌ TypeError - c muss benannt werden!")
    
    try:
        keyword_only_funktion(1, 2, 3)
    except TypeError as e:
        print(f"   Fehler wie erwartet: {e}")
    
    print("\n" + "="*50)
    
    # Experiment 3: Parameter-Entpackung
    print("🧪 Experiment 3: Parameter-Entpackung verstehen")
    
    def flexible_funktion(name, **options):
        print(f"   Name: {name}")
        print(f"   Optionen: {options}")
        return options
    
    print("   Funktion mit **kwargs:")
    config1 = {"alter": 25, "stadt": "Berlin", "beruf": "Entwickler"}
    print("   Aufruf: flexible_funktion('Max', **config)")
    result = flexible_funktion("Max", **config1)
    
    print("\n   Direkte Keyword-Argumente:")
    print("   Aufruf: flexible_funktion('Anna', alter=30, hobby='Lesen')")
    result2 = flexible_funktion("Anna", alter=30, hobby="Lesen")
    
    print("\n✅ Alle Experimente erfolgreich!")
```
