```python
# Zelle 1: Zeitmessung-Dekorator implementieren
import time

def zeitmessung(func):
    """
    Dekorator der die Ausführungszeit einer Funktion misst.
    
    Parameter:
    - func: Die zu dekorierende Funktion
    
    Rückgabe:
    - Wrapper-Funktion die Zeit misst und anzeigt
    """
    def wrapper(*args, **kwargs):
        # Startzeit erfassen
        start_zeit = time.time()
        
        # Funktion ausführen
        result = func(*args, **kwargs)
        
        # Endzeit erfassen und Differenz berechnen
        end_zeit = time.time()
        dauer = end_zeit - start_zeit
        
        # Zeit anzeigen
        print(f"⏱️ {func.__name__} dauerte {dauer:.4f} Sekunden")
        
        # Ergebnis zurückgeben
        return result
    
    return wrapper

# Test-Funktion mit Zeitmessung dekorieren
@zeitmessung
def langsame_berechnung(x):
    """Simuliert eine langsame Berechnung."""
    time.sleep(0.2)  # Simuliert Berechnungszeit
    return x * x

# Tests der Zeitmessung
print("🧪 TEST DER ZEITMESSUNG")

print("\n🧪 Test 1: Einfache Berechnung")

print(f"Ergebnis: {ergebnis1}")

print("\n🧪 Test 2: Verschiedene Parameter")
ergebnis2 = langsame_berechnung(3)
print(f"Ergebnis: {ergebnis2}")

print("\n🧪 Test 3: Fibonacci-Test (ohne Cache)")
@zeitmessung
def fibonacci_mit_zeit(n):
    if n <= 1:

print("Fibonacci(8) - wird sehr langsam:")
```

```python
# Zelle 2: Cache-Dekorator implementieren
def einfacher_cache(func):
    """
    Dekorator der Funktionsergebnisse zwischenspeichert.
    
    Parameter:
    - func: Die zu dekorierende Funktion
    
    Rückgabe:
    - Wrapper-Funktion mit Cache-Funktionalität
    """
    # Cache-Dictionary als Closure-Variable erstellen
    cache = {}
    
    def wrapper(*args, **kwargs):
        # Cache-Key aus Argumenten erstellen
        # Einfache Variante: String-Representation
        key = str(args) + str(sorted(kwargs.items()))
        
        # Prüfen ob Ergebnis bereits im Cache
        if key in cache:
            print(f"💾 Cache-Hit für {func.__name__} mit Argumenten {args}")
            return cache[key]
        
        # Falls nicht im Cache: Funktion ausführen
        print(f"🔄 Cache-Miss für {func.__name__} - Berechnung läuft...")
        result = func(*args, **kwargs)
        
        # Ergebnis im Cache speichern
        cache[key] = result
        print(f"💾 Ergebnis für {func.__name__} wurde gecacht")
        
        return result
    
    return wrapper

# Test der Cache-Funktionalität
@einfacher_cache
def fibonacci_mit_cache(n):
    """Fibonacci mit Cache-Dekorator."""
    if n <= 1:
        return n
    return fibonacci_mit_cache(n-1) + fibonacci_mit_cache(n-2)

@einfacher_cache
def teure_berechnung(x, y):
    """Simuliert eine teure Berechnung."""
    time.sleep(0.2)
    return x ** y
```

```python
# Zelle 3: Kombinierter Performance-Dekorator
import time

def performance_monitor(func):
    """
    Kombinierter Dekorator für Zeitmessung und Caching.
    
    Parameter:
    - func: Die zu dekorierende Funktion
    
    Rückgabe:
    - Wrapper mit Zeit- und Cache-Funktionalität
    """
    # Cache-Dictionary initialisieren
    cache = {}
    cache_stats = {"hits": 0, "misses": 0}
    
    def wrapper(*args, **kwargs):
        # Cache-Key erstellen (robustere Version)
        try:
            key = (args, tuple(sorted(kwargs.items())))
        except TypeError:
            # Falls Argumente nicht hashable sind
            key = str(args) + str(sorted(kwargs.items()))
        
        # Startzeit erfassen
        start_zeit = time.time()
        
        # Cache prüfen
        if key in cache:
            # Cache-Hit: Gespeichertes Ergebnis verwenden
            result = cache[key]
            dauer = time.time() - start_zeit
            cache_stats["hits"] += 1
            print(f"💾 {func.__name__}: Cache-Hit in {dauer:.6f}s (Hit #{cache_stats['hits']})")
        else:
            # Cache-Miss: Funktion ausführen
            result = func(*args, **kwargs)
            cache[key] = result
            dauer = time.time() - start_zeit
            cache_stats["misses"] += 1
            print(f"🔄 {func.__name__}: Berechnung in {dauer:.4f}s (Miss #{cache_stats['misses']})")
        
        return result
    
    # Cache-Statistiken als Funktion hinzufügen
    def cache_info():
        total = cache_stats["hits"] + cache_stats["misses"]
        hit_rate = cache_stats["hits"] / total * 100 if total > 0 else 0
        return {
            "hits": cache_stats["hits"],
            "misses": cache_stats["misses"],
            "hit_rate": f"{hit_rate:.1f}%",
            "cache_size": len(cache)
        }
    
    wrapper.cache_info = cache_info
    wrapper.cache_clear = lambda: cache.clear() or cache_stats.update({"hits": 0, "misses": 0})
    
    return wrapper

# Test des kombinierten Dekorators

def datenbankabfrage(user_id):
```

### Schritt 4: Metadaten erhalten mit functools.wraps

```python
# Zelle 4: Verbesserte Dekoratoren mit functools.wraps
import functools
import time

def verbesserter_performance_monitor(func):
    """
    Verbesserter Performance-Dekorator mit erhaltenen Metadaten.
    """
    cache = {}
    stats = {"hits": 0, "misses": 0, "total_time": 0}
    
    @functools.wraps(func)  # Erhält ursprüngliche Metadaten!
    def wrapper(*args, **kwargs):
        # Cache-Key erstellen
        try:
            key = (args, tuple(sorted(kwargs.items())))
        except TypeError:
            key = str(args) + str(sorted(kwargs.items()))
        
        start_zeit = time.time()
        
        if key in cache:
            # Cache-Hit
            result = cache[key]
            dauer = time.time() - start_zeit
            stats["hits"] += 1
            stats["total_time"] += dauer
            print(f"💾 {func.__name__}: Cache-Hit in {dauer:.6f}s")
        else:
            # Cache-Miss
            result = func(*args, **kwargs)
            cache[key] = result
            dauer = time.time() - start_zeit
            stats["misses"] += 1
            stats["total_time"] += dauer
            print(f"🔄 {func.__name__}: Berechnung in {dauer:.4f}s")
        
        return result
    
    # Zusätzliche Utility-Funktionen
    def get_stats():
        total_calls = stats["hits"] + stats["misses"]
        hit_rate = stats["hits"] / total_calls * 100 if total_calls > 0 else 0
        avg_time = stats["total_time"] / total_calls if total_calls > 0 else 0
        
        return {
            "function": func.__name__,
            "total_calls": total_calls,
            "cache_hits": stats["hits"],
            "cache_misses": stats["misses"],
            "hit_rate": f"{hit_rate:.1f}%",
            "cache_size": len(cache),
            "total_time": f"{stats['total_time']:.4f}s",
            "average_time": f"{avg_time:.4f}s"
        }
    
    def clear_cache():
        cache.clear()
        stats.update({"hits": 0, "misses": 0, "total_time": 0})
        print(f"🧹 Cache für {func.__name__} wurde geleert")
    
    # Utility-Funktionen an wrapper anhängen
    wrapper.get_stats = get_stats
    wrapper.clear_cache = clear_cache
    
    return wrapper

# Test der verbesserten Dekoratoren

    """Führt eine komplexe mathematische Berechnung durch."""
    time.sleep(0.1)  # Simuliert aufwändige Berechnung
    result = x ** y + x * y
    return result

@verbesserter_performance_monitor
def string_verarbeitung(text, upper=False):
    """Verarbeitet einen String mit optionaler Groß-/Kleinschreibung."""
    time.sleep(0.05)  # Simuliert Textverarbeitung
    result = text.upper() if upper else text.lower()
    return f"Verarbeitet: {result}"

# Tests der verbesserten Implementierung
print("🧪 TEST DER VERBESSERTEN DEKORATOREN")
print("=" * 45)

print("\n🧪 Test 1: Metadaten-Erhaltung prüfen")
print(f"Funktionsname: {komplexe_berechnung.__name__}")
print(f"Dokumentation: {komplexe_berechnung.__doc__}")
print(f"Modul: {komplexe_berechnung.__module__}")

print("\n🧪 Test 2: Funktionalität mit verschiedenen Parametern")
result1 = komplexe_berechnung(2, 3)
print(f"komplexe_berechnung(2, 3) = {result1}")

result2 = komplexe_berechnung(2, 3)  # Gleiche Parameter - Cache-Hit
print(f"komplexe_berechnung(2, 3) = {result2}")

result3 = komplexe_berechnung(3, 2)  # Andere Parameter - Cache-Miss
print(f"komplexe_berechnung(3, 2) = {result3}")

print("\n🧪 Test 3: Keyword-Arguments")

print(f"Text 1: {text1}")

text2 = string_verarbeitung("HELLO World", upper=True)
print(f"Text 2: {text2}")

text3 = string_verarbeitung("HELLO World")  # Cache-Hit
print(f"Text 3: {text3}")

print("\n📊 Detaillierte Statistiken:")
print("\nkomplexe_berechnung:")

for key, value in stats1.items():
    print(f"   {key}: {value}")

for key, value in stats2.items():

print("\n🧪 Test 4: Cache-Reset")

print("Nach Cache-Reset:")
print(f"komplexe_berechnung Cache-Größe: {komplexe_berechnung.get_stats()['cache_size']}")
print(f"string_verarbeitung Cache-Größe: {string_verarbeitung.get_stats()['cache_size']}")

print("\n✅ Verbesserte Dekoratoren funktionieren perfekt!")
```

## 🎓 Zusammenfassung der Lösung

### Wichtige Konzepte implementiert:

1. **Dekorator-Grundlagen:**
    
    ```python
    def dekorator(func):
        def wrapper(*args, **kwargs):
            # Code vor Aufruf
            result = func(*args, **kwargs)
            # Code nach Aufruf
            return result
        return wrapper
    ```
    
2. **Closures für Zustand:**
    
    ```python
    def cache_dekorator(func):
        cache = {}  # Closure-Variable
        def wrapper(*args, **kwargs):
            # cache bleibt zwischen Aufrufen erhalten
    ```
    
3. **functools.wraps:**
    
    ```python
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        # Erhält __name__, __doc__, etc.
    ```
    
4. **Utility-Funktionen:**
    
    ```python
    wrapper.get_stats = lambda: {...}
    wrapper.clear_cache = lambda: cache.clear()
    ```
    

### Praktische Vorteile:

- **Performance-Monitoring:** Automatische Zeitmessung ohne Code-Änderung
- **Cache-System:** Transparente Optimierung für teure Operationen
- **Debugging:** Cache-Statistiken für Performance-Analyse
- **Wiederverwendbarkeit:** Ein Dekorator für viele Funktionen

### Reale Anwendungen:

- **Web-APIs:** Caching von Datenbankabfragen
- **Machine Learning:** Memoization für teure Berechnungen
- **Monitoring:** Performance-Tracking in Produktionssystemen
- **Debugging:** Automatisches Logging ohne Code-Verschmutzung

### Design-Patterns gelernt:

- **Decorator Pattern:** Funktionalität transparent erweitern
- **Closure Pattern:** Zustand zwischen Aufrufen erhalten
- **Observer Pattern:** Performance-Metriken sammeln
- **Cache Pattern:** Ergebnisse zwischenspeichern
