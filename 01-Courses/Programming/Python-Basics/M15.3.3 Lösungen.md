# Zelle 3.3: Katalog-Statistiken erstellen
def berechne_katalog_statistiken():
    """Berechnet verschiedene Statistiken über den Buchkatalog"""
    import json
    
    # Katalog laden
    with open("katalog.json", "r", encoding="utf-8") as datei:
        bücher = json.load(datei)
    
    # Verschiedene Statistiken berechnen
    gesamt_anzahl = len(bücher)
    
    # Verfügbare Bücher zählen
    verfügbare_anzahl = 0
    for buch in bücher:
        if buch["verfügbar"]:
            verfügbare_anzahl += 1
    
    # Durchschnittspreis berechnen
    preis_summe = 0
    for buch in bücher:
        preis_summe += buch["preis"]
    durchschnittspreis = round(preis_summe / gesamt_anzahl, 2)
    
    # Statistiken als Dictionary zusammenfassen
    statistiken = {
        "gesamt": gesamt_anzahl,
        "verfügbar": verfügbare_anzahl,
        "durchschnittspreis": durchschnittspreis
    }
    
    print(f"\n📊 SCHRITT 3.3: Katalog-Statistiken")
    print("=" * 40)
    print(f"✅ Gesamtanzahl Bücher: {gesamt_anzahl}")
    print(f"✅ Verfügbare Bücher: {verfügbare_anzahl}")
    print(f"✅ Durchschnittspreis: {durchschnittspreis}€")
    print(f"✅ Verfügbarkeitsrate: {round(verfügbare_anzahl/gesamt_anzahl*100, 1)}%")
    return f"Katalog-Statistiken: {statistiken}"

# Test der Funktion
print(berechne_katalog_statistiken())

def sichere_json_leseoperation():

def zeige_json_dateien_uebersicht():
    """Zeigt eine Übersicht aller erstellten JSON-Dateien"""

    import os
    
    json_dateien = [
        "buch.json",

        "buch_erweitert.json",
        "sicheres_buch.json"
    ]

    print("="*60)

        try:
            # Prüfen ob Datei existiert und JSON-Format korrekt ist
            with open(datei, "r", encoding="utf-8") as f:
                daten = json.load(f)
            
            # Dateigröße ermitteln
            groesse = os.path.getsize(datei)
            
            # Datentyp bestimmen
            if isinstance(daten, list):
                typ = f"Liste mit {len(daten)} Einträgen"

                typ = f"Dictionary mit {len(daten)} Keys"

        except FileNotFoundError:
            print(f"❌ {datei}: Datei nicht gefunden")
        except json.JSONDecodeError:
            print(f"⚠️ {datei}: Ungültiges JSON-Format")
        except Exception as fehler:
            print(f"⚠️ {datei}: Fehler - {fehler}")
    
    print(f"\n📊 STATISTIK:")
    print(f"   ✅ Erfolgreich: {erfolgreiche_dateien}/{len(json_dateien)} JSON-Dateien")
