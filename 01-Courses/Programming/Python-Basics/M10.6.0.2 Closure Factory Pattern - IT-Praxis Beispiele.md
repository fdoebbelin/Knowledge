## Was ist das Closure Factory Pattern?

Das **Closure Factory Pattern** ist ein Entwurfsmuster, bei dem eine "Fabrik-Funktion" spezialisierte Funktionen erstellt, die sich an bestimmte Konfigurationen "erinnern". Diese erstellten Funktionen sind **Closures** - sie behalten Zugriff auf Variablen aus ihrem ursprünglichen Erstellungskontext.
## Grundkonzept

```python
def factory_funktion(konfiguration):
    def worker_funktion(daten):
        # Kann auf 'konfiguration' zugreifen (Closure!)
        return verarbeite(daten, konfiguration)
    return worker_funktion
```

```python
def erstelle_logger(level="INFO", prefix=""):
    """
    Factory-Funktion: Erstellt einen spezialisierten Logger.
    
    Args:
        level: Log-Level (DEBUG, INFO, WARNING, ERROR)
        prefix: Präfix für alle Log-Nachrichten
    
    Returns:
        Logger-Funktion, die Nachrichten mit fester Konfiguration loggt
    """
    import datetime
    
    def log(message):
        timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        formatted_message = f"[{timestamp}] [{level}]"
        
        if prefix:
            formatted_message += f" [{prefix}]"
            
        formatted_message += f" {message}"
        print(formatted_message)
    
    # Metadaten an Logger anhängen
    log.level = level
    log.prefix = prefix
    
    return log

# Verschiedene Logger erstellen
```

```python
def erstelle_api_client(base_url, api_key=None, timeout=30):
    """
    Erstellt einen API-Client mit fester Konfiguration.
    
    Args:
        base_url: Basis-URL der API
        api_key: Optional - API-Schlüssel für Authentifizierung
        timeout: Timeout in Sekunden
    
    Returns:
        API-Client-Funktion mit verschiedenen HTTP-Methoden
    """
    import time
    
    def make_request(endpoint, method="GET", data=None):
        # Vollständige URL zusammenbauen
        full_url = f"{base_url.rstrip('/')}/{endpoint.lstrip('/')}"
        
        # Headers vorbereiten
        headers = {"User-Agent": "Python-API-Client/1.0"}
        if api_key:
            headers["Authorization"] = f"Bearer {api_key}"
        
        # Simulation einer HTTP-Anfrage (normalerweise requests.get/post/etc.)
        print(f"🌐 {method} {full_url}")
        print(f"   Headers: {headers}")
        if data:
            print(f"   Data: {data}")
        
        # Timeout simulieren
        time.sleep(0.1)  # Kurze Pause für Realismus
        
        # Mockup-Response
        return {
            "status": 200,
            "url": full_url,
            "method": method,
            "response_time": f"{timeout}ms",
            "data": f"Response from {endpoint}"
        }
    
    # Client-Methoden als Dictionary
    def get(endpoint):
        return make_request(endpoint, "GET")
    
    def post(endpoint, data):
        return make_request(endpoint, "POST", data)
    
    def put(endpoint, data):
        return make_request(endpoint, "PUT", data)
    
    def delete(endpoint):
        return make_request(endpoint, "DELETE")
    
    # Client-Interface mit Metadaten
    client = {
        "get": get,
        "post": post,
        "put": put,
        "delete": delete
    }
    
    # Konfiguration speichern
    client["config"] = {
        "base_url": base_url,
        "api_key": api_key[:10] + "..." if api_key else None,
        "timeout": timeout
    }
    
    return client

# Verschiedene API-Clients erstellen
```

**Zelle 5 - Validator-Factory:**

```python
def erstelle_validator(rules):
    """
    Factory für Datenvalidatoren mit konfigurierbaren Regeln.
    
    Args:
        rules: Dictionary mit Validierungsregeln
        
    Returns:
        Validator-Funktion die Daten gegen die Regeln prüft
    """
    
    def validate(data):
        errors = []
        warnings = []
        
        for field, field_rules in rules.items():
            value = data.get(field)
            
            # Required Check
            if field_rules.get("required", False) and not value:
                errors.append(f"Feld '{field}' ist erforderlich")
                continue
            
            if value is None:
                continue
                
            # Type Check
            expected_type = field_rules.get("type")
            if expected_type and not isinstance(value, expected_type):
                errors.append(f"Feld '{field}' muss vom Typ {expected_type.__name__} sein")
                continue
                
            # Length Check (für Strings)
            if isinstance(value, str):
                min_len = field_rules.get("min_length")
                max_len = field_rules.get("max_length")
                
                if min_len and len(value) < min_len:
                    errors.append(f"Feld '{field}' muss mindestens {min_len} Zeichen haben")
                    
                if max_len and len(value) > max_len:
                    errors.append(f"Feld '{field}' darf maximal {max_len} Zeichen haben")
            
            # Range Check (für Zahlen)
            if isinstance(value, (int, float)):
                min_val = field_rules.get("min_value")
                max_val = field_rules.get("max_value")
                
                if min_val is not None and value < min_val:
                    errors.append(f"Feld '{field}' muss mindestens {min_val} sein")
                    
                if max_val is not None and value > max_val:
                    errors.append(f"Feld '{field}' darf maximal {max_val} sein")
            
            # Pattern Check
            pattern = field_rules.get("pattern")
            if pattern and isinstance(value, str):
                import re
                if not re.match(pattern, value):
                    errors.append(f"Feld '{field}' entspricht nicht dem Format")
        
        return {
            "valid": len(errors) == 0,
            "errors": errors,
            "warnings": warnings,
            "data": data
        }
    
    # Validator-Metadaten
    validate.rules = rules
    return validate

# Verschiedene Validatoren erstellen
```

## Schritt 4: File-Processor Factory (Integration)

```python
def erstelle_file_processor(base_path="./", allowed_extensions=None, max_size_mb=10):
    """
    Factory für File-Processing mit konfigurierbaren Sicherheitsregeln.
    
    Args:
        base_path: Basis-Pfad für Datei-Operationen
        allowed_extensions: Liste erlaubter Dateierweiterungen
        max_size_mb: Maximale Dateigröße in MB
        
    Returns:
        File-Processor mit verschiedenen Operationen
    """
    import os
    
    # Default-Extensions wenn keine angegeben
    if allowed_extensions is None:
        allowed_extensions = ['.txt', '.json', '.csv', '.log']
    
    def validate_file(filename, size_mb=0):
        """Validiert Datei gegen die Konfiguration."""
        errors = []
        
        # Extension Check
        file_ext = os.path.splitext(filename)[1].lower()
        if file_ext not in allowed_extensions:
            errors.append(f"Dateierweiterung '{file_ext}' nicht erlaubt")
        
        # Size Check  
        if size_mb > max_size_mb:
            errors.append(f"Datei zu groß: {size_mb}MB (max: {max_size_mb}MB)")
        
        # Path Traversal Check
        if '..' in filename or filename.startswith('/'):
            errors.append("Unsichere Pfad-Zeichen erkannt")
            
        return errors
    
    def read_file(filename):
        """Liest eine Datei (simuliert)."""
        errors = validate_file(filename)
        if errors:
            return {"success": False, "errors": errors}
        
        filepath = os.path.join(base_path, filename)
        
        # Simulation des Datei-Lesens
        return {
            "success": True,
            "filepath": filepath,
            "content": f"[Inhalt von {filename}]",
            "size": "1.2MB"
        }
    
    def write_file(filename, content, size_mb=1):
        """Schreibt eine Datei (simuliert)."""
        errors = validate_file(filename, size_mb)
        if errors:
            return {"success": False, "errors": errors}
        
        filepath = os.path.join(base_path, filename)
        
        # Simulation des Datei-Schreibens
        return {
            "success": True,
            "filepath": filepath,
            "message": f"Datei {filename} erfolgreich geschrieben",
            "size": f"{size_mb}MB"
        }
    
    def list_allowed_types():
        """Zeigt erlaubte Dateitypen."""
        return {
            "allowed_extensions": allowed_extensions,
            "max_size_mb": max_size_mb,
            "base_path": base_path
        }
    
    # Processor-Interface
    return {
        "read": read_file,
        "write": write_file,
        "info": list_allowed_types,
        "validate": validate_file
    }

# Verschiedene File-Processors erstellen
print("=== File-Processor Factory ===")
```

## Schritt 9: Vollständige IT-Factory (Integration)

**Zelle 9 - Komplette IT-Services Factory:**

```python
def erstelle_it_services():
    """
    Vollständige Factory für IT-Services.
    Kombiniert alle vorherigen Konzepte in einer einzigen Factory.
    
    Returns:
        Dictionary mit verschiedenen Service-Factories
    """
    
    # Alle Factory-Funktionen wiederverwenden
    def erstelle_logger_service(level="INFO", prefix=""):
        import datetime
        def log(message):
            timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            formatted = f"[{timestamp}] [{level}]"
            if prefix:
                formatted += f" [{prefix}]"
            formatted += f" {message}"
            print(formatted)
        log.level = level
        log.prefix = prefix
        return log
    
    def erstelle_api_service(base_url, api_key=None, timeout=30):
        def make_request(endpoint, method="GET", data=None):
            full_url = f"{base_url.rstrip('/')}/{endpoint.lstrip('/')}"
            headers = {"User-Agent": "Python-API-Client/1.0"}
            if api_key:
                headers["Authorization"] = f"Bearer {api_key}"
            
            return {
                "status": 200,
                "url": full_url,
                "method": method,
                "headers": headers,
                "data": data
            }
        
        return {
            "get": lambda ep: make_request(ep, "GET"),
            "post": lambda ep, data: make_request(ep, "POST", data),
            "config": {"base_url": base_url, "timeout": timeout}
        }
    
    def erstelle_validator_service(rules):
        def validate(data):
            errors = []
            for field, field_rules in rules.items():
                value = data.get(field)
                if field_rules.get("required", False) and not value:
                    errors.append(f"Feld '{field}' ist erforderlich")
            return {"valid": len(errors) == 0, "errors": errors}
        validate.rules = rules
        return validate
    
    # Service-Factory-Interface
    return {
        "logger": erstelle_logger_service,
        "api_client": erstelle_api_service,
        "validator": erstelle_validator_service
    }

# IT-Services Factory erstellen
it_factory = erstelle_it_services()

print("=== IT-Services Factory ===")
print("Verfügbare Services:")
for service_name in it_factory.keys():
    print(f"  - {service_name}")
```

## Zusammenfassung

Das **Closure Factory Pattern** in der IT-Praxis ermöglicht:

1. **Konfigurierbare Services**: Logger, API-Clients, Validators mit festen Einstellungen
2. **Environment-spezifische Konfigurationen**: Dev/Staging/Production
3. **Wiederverwendbare Komponenten**: Einmal definiert, mehrfach konfiguriert
4. **Testbarkeit**: Mock-Services mit eigenen Konfigurationen
5. **Kapselung**: Keine globalen Variablen, saubere Trennung

Die Closures "erinnern" sich an ihre Konfiguration und bieten so eine elegante Lösung für typische IT-Herausforderungen wie Service-Management, API-Integration und Datenvalidierung.
