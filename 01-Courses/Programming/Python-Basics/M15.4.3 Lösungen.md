# Zelle 4.3: Noten zusammen mit Studentennamen ausgeben
def zeige_noten_mit_namen():
    """Zeigt alle Noten zusammen mit den Studentennamen"""
    import sqlite3
    
    # Verbindung zur Datenbank herstellen
    verbindung = sqlite3.connect('studenten.db')
    cursor = verbindung.cursor()
    
    # JOIN-Abfrage um Studenten- und Noten-Tabelle zu verknüpfen
    sql_befehl = """
    SELECT studenten.name, noten.fach, noten.note, noten.datum
    FROM studenten
    JOIN noten ON studenten.id = noten.student_id
    ORDER BY studenten.name, noten.fach
    """
    cursor.execute(sql_befehl)
    noten_mit_namen = cursor.fetchall()
    
    # Verbindung schließen

    studenten_noten = {}
    for name, fach, note, datum in noten_mit_namen:
        if name not in studenten_noten:

# Zelle 5.2: Durchschnittsnote für jeden Studenten berechnen
def berechne_durchschnittsnoten():
    """Berechnet die Durchschnittsnote für jeden Studenten"""
    import sqlite3
    
    # Verbindung zur Datenbank herstellen
    verbindung = sqlite3.connect('studenten.db')
    cursor = verbindung.cursor()
    
    # SELECT mit GROUP BY und AVG() Aggregatfunktion
    sql_befehl = """
    SELECT studenten.name, ROUND(AVG(noten.note), 2) as durchschnitt
    FROM studenten
    JOIN noten ON studenten.id = noten.student_id
    GROUP BY studenten.id, studenten.name
    ORDER BY durchschnitt
    """
    cursor.execute(sql_befehl)
    durchschnittsnoten = cursor.fetchall()
    
    # Verbindung schließen
    cursor.close()
    verbindung.close()
    
    print("\n📊 SCHRITT 5.2: Durchschnittsnoten berechnen")
    print("=" * 48)
    print("✅ JOIN zwischen studenten und noten Tabelle")
    print("✅ GROUP BY für Gruppierung nach Student")

    print("✅ ROUND() für Rundung auf 2 Dezimalstellen")

    try:

        verbindung = sqlite3.connect('studenten.db')

# Zelle 7.1: 'with' Statement für automatische Datenbankverbindung-Verwaltung
def verwende_context_manager():
    """Verwendet 'with' für automatische Datenbankverbindung-Verwaltung"""
    import sqlite3
    
    # 'with' Statement für automatisches Schließen der Verbindung
    with sqlite3.connect('studenten.db') as verbindung:
        cursor = verbindung.cursor()
        
        # Neue Tabelle 'kurse' erstellen
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS kurse (
                id INTEGER PRIMARY KEY,
                name TEXT NOT NULL,
                credits INTEGER
            )
        """)
        
        # Beispielkurs einfügen
        cursor.execute(
            "INSERT OR IGNORE INTO kurse (name, credits) VALUES (?, ?)",
            ('Einführung in SQLite', 3)
        )
        
        # Weitere Kurse hinzufügen
        weitere_kurse = [
            ('Python Programmierung', 5),
            ('Datenbank Design', 4),
            ('Web Development', 6)
        ]
        cursor.executemany(
            "INSERT OR IGNORE INTO kurse (name, credits) VALUES (?, ?)",
            weitere_kurse
        )
        
        # Alle Kurse abfragen
        cursor.execute("SELECT id, name, credits FROM kurse")
        alle_kurse = cursor.fetchall()
        
        # Verbindung wird automatisch geschlossen wenn 'with' Block verlassen wird!
    
    print("\n🔄 SCHRITT 7.1: Context Manager verwenden")
    print("=" * 46)
    print("✅ 'with sqlite3.connect()' für automatisches Schließen")
    print("✅ Keine expliziten cursor.close() und verbindung.close() nötig")
    print("✅ Tabelle 'kurse' erstellt mit 3 Spalten")
    print("✅ INSERT OR IGNORE verhindert Duplikate")
    print("✅ executemany() für mehrere Kurse verwendet")
    print(f"✅ {len(alle_kurse)} Kurse in Datenbank:")
    for kurs_id, name, credits in alle_kurse:
        print(f"   - ID {kurs_id}: {name} ({credits} Credits)")
    print("✅ Datenbankverbindung automatisch geschlossen!")
    return f"✅ Context Manager verwendet - {len(alle_kurse)} Kurse in Datenbank"

# Test der Funktion

                anzahl = cursor.fetchone()[0]
