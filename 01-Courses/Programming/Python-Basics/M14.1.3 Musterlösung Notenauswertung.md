```python
# Zelle 6: List Comprehension Patterns verstehen
def erklaere_patterns():
    """Erklärt die verschiedenen List Comprehension Patterns"""
    noten = [2, 1, 4, 3, 5, 2, 1, 3, 4, 2, 6, 1, 3, 2, 4]
    
    print("\n🧠 LIST COMPREHENSION PATTERNS")
    print("=" * 40)
    
    # Pattern 1: Nur Filtern (keine Transformation)
    print("1️⃣ NUR FILTERN:")
    print("   Syntax: [element for element in liste if bedingung]")
    bestanden = [note for note in noten if note <= 4]
    print(f"   Beispiel: [note for note in noten if note <= 4]")
    print(f"   Ergebnis: {bestanden}")
    
    # Pattern 2: Nur Transformieren (ohne Filter)
    print("\n2️⃣ NUR TRANSFORMIEREN:")
    print("   Syntax: [transform(element) for element in liste]")
    quadrate = [note ** 2 for note in [1, 2, 3]]
    print(f"   Beispiel: [note ** 2 for note in [1, 2, 3]]")
    print(f"   Ergebnis: {quadrate}")
    
    # Pattern 3: Filtern UND Transformieren
    print("\n3️⃣ FILTERN + TRANSFORMIEREN:")
    print("   Syntax: [transform(element) for element in liste if bedingung]")
    lobkaertchen = ["Sehr gut!" for note in noten if note <= 2]
    print(f"   Beispiel: ['Sehr gut!' for note in noten if note <= 2]")
    print(f"   Ergebnis: {lobkaertchen}")
    
    # Pattern 4: Dictionary-Lookup
    print("\n4️⃣ DICTIONARY-LOOKUP:")
    print("   Syntax: [dictionary[element] for element in liste]")
    punkte_dict = {1: 15, 2: 12, 3: 9}
    beispiel_noten = [1, 2, 3]
    punkte = [punkte_dict[note] for note in beispiel_noten]
    print(f"   Beispiel: [punkte_dict[note] for note in [1, 2, 3]]")
    print(f"   Ergebnis: {punkte}")
    
    return bestanden, quadrate, lobkaertchen, punkte

# Patterns erklären
erklaere_patterns()
```

### 🎯 Wichtige Erkenntnisse:

- **List Comprehensions** sind kompakter und oft schneller als normale Schleifen
- **Schrittweise Entwicklung** hilft beim Verständnis: Iteration → Filter → Transformation
- **Pattern-Erkennung** erleichtert die Anwendung in neuen Situationen
- **Lesbarkeit** ist wichtiger als Kompaktheit - bei komplexer Logik normale Schleifen verwenden

### 📚 Pattern-Sammlung:
1. **Nur Filtern:** `[x for x in liste if bedingung]`
2. **Nur Transformieren:** `[transform(x) for x in liste]`
3. **Filtern + Transformieren:** `[transform(x) for x in liste if bedingung]`
4. **Dictionary-Lookup:** `[dictionary[x] for x in liste]`
