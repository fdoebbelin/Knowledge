## 1. Was sind Tupel?
Tupel sind wie Listen, aber **unveränderlich** (können nicht geändert werden).

```python
# Liste (veränderlich)
meine_liste = [1, 2, 3]
print(f"Liste: {meine_liste}")

# Tupel (unveränderlich)
mein_tupel = (1, 2, 3)
print(f"Tupel: {mein_tupel}")

print(f"Liste-Typ: {type(meine_liste)}")
print(f"Tupel-Typ: {type(mein_tupel)}")
```
## 2. Tupel erstellen

```python
# Leeres Tupel
leer = ()
print(f"Leer: {leer}")

# Tupel mit mehreren Elementen
koordinaten = (10, 20, 30)
print(f"Koordinaten: {koordinaten}")

# Verschiedene Datentypen
person = ("Anna", 25, "Berlin")
print(f"Person: {person}")
```

```python
# WICHTIG: Ein-Element-Tupel braucht ein Komma!
kein_tupel = (42)      # Das ist nur eine Zahl!
ein_tupel = (42,)      # Das ist ein Tupel!

print(f"(42): {kein_tupel}, Typ: {type(kein_tupel)}")
print(f"(42,): {ein_tupel}, Typ: {type(ein_tupel)}")
```

```python
# Tupel ohne Klammern
punkt = 100, 200  # Auch ein Tupel!
print(f"Punkt: {punkt}, Typ: {type(punkt)}")

# Mehrere Variablen gleichzeitig
name, alter = "Bob", 30
print(f"Name: {name}, Alter: {alter}")
```
## 3. Warum sind Tupel unveränderlich?

```python
# Listen können geändert werden
liste = [1, 2, 3]
print(f"Liste vorher: {liste}")
liste[0] = 99  # Funktioniert!
print(f"Liste nachher: {liste}")

# Tupel können NICHT geändert werden
tupel = (1, 2, 3)
print(f"Tupel: {tupel}")

# Das gibt einen Fehler:
try:
    tupel[0] = 99  # Fehler!
except TypeError as e:
    print(f"Fehler: {e}")
```

```python
# Tupel haben auch keine Methoden zum Ändern
liste = [1, 2, 3]
tupel = (1, 2, 3)

# Liste hat viele Methoden
print("Listen-Methoden:", [m for m in dir(liste) if not m.startswith('_')])

# Tupel hat nur wenige Methoden
print("Tupel-Methoden:", [m for m in dir(tupel) if not m.startswith('_')])
```
## 4. Auf Tupel zugreifen

```python
# Wie bei Listen: Index und Slicing
wochentage = ("Montag", "Dienstag", "Mittwoch", "Donnerstag", "Freitag")
print(f"Wochentage: {wochentage}")

# Einzelne Elemente
print(f"Erster Tag: {wochentage[0]}")
print(f"Letzter Tag: {wochentage[-1]}")

# Slicing
print(f"Erste drei: {wochentage[:3]}")
print(f"Letzte zwei: {wochentage[-2:]}")
```

```python
# Die wenigen Tupel-Methoden
buchstaben = ('a', 'b', 'c', 'b', 'd', 'b')
print(f"Buchstaben: {buchstaben}")

# count() - wie oft kommt ein Element vor?
anzahl_b = buchstaben.count('b')
print(f"Anzahl 'b': {anzahl_b}")

# index() - wo steht ein Element?
position_b = buchstaben.index('b')
print(f"Erste Position von 'b': {position_b}")
```
## 5. Tuple Unpacking - Das Beste an Tupeln!
### Grundlagen

```python
# Tupel in einzelne Variablen "entpacken"
punkt = (100, 200)
print(f"Punkt: {punkt}")

x, y = punkt  # Unpacking!
print(f"x = {x}")
print(f"y = {y}")
```

```python
# Mit mehr Elementen
person = ("Anna", 25, "Berlin")
name, alter, stadt = person

print(f"Name: {name}")
print(f"Alter: {alter}")
print(f"Stadt: {stadt}")
```

### Variablen tauschen - Elegant!

```python
# Der alte Weg (mit Hilfsvariable)
a = 10
b = 20
print(f"Vorher: a={a}, b={b}")

# temp = a
# a = b  
# b = temp

# Der Python-Weg (mit Tupel):
a, b = b, a  # Viel eleganter!
print(f"Nachher: a={a}, b={b}")
```

### Funktionen mit mehreren Rückgabewerten

```python
def berechne_kreis(radius):
    """Berechnet Umfang und Fläche"""
    umfang = 2 * 3.14159 * radius
    flaeche = 3.14159 * radius * radius
    return umfang, flaeche  # Gibt Tupel zurück!

# Unpacking beim Empfangen
u, f = berechne_kreis(5)
print(f"Umfang: {u:.2f}")
print(f"Fläche: {f:.2f}")

# Oder als ganzes Tupel
ergebnis = berechne_kreis(3)
print(f"Ergebnis: {ergebnis}")
```
## 6. Erweiterte Unpacking-Tricks

```python
# Mit * den "Rest" sammeln
zahlen = (1, 2, 3, 4, 5, 6)
print(f"Zahlen: {zahlen}")

erste, *mittlere, letzte = zahlen
print(f"Erste: {erste}")
print(f"Mittlere: {mittlere}")  # Das wird eine Liste!
print(f"Letzte: {letzte}")
```

```python
# Verschiedene Muster
alphabet = ('a', 'b', 'c', 'd', 'e')

# Erste zwei, dann Rest
erste, zweite, *rest = alphabet
print(f"Erste zwei: {erste}, {zweite}")
print(f"Rest: {rest}")

# Alles außer letzten zwei
*anfang, vorletzte, letzte = alphabet
print(f"Anfang: {anfang}")
print(f"Letzte zwei: {vorletzte}, {letzte}")
```
## 7. Praktische Beispiele

### Koordinaten

```python
# 2D-Punkte als Tupel
punkt_a = (0, 0)
punkt_b = (3, 4)

print(f"Punkt A: {punkt_a}")
print(f"Punkt B: {punkt_b}")

# Entfernung berechnen
def entfernung(p1, p2):
    x1, y1 = p1  # Unpacking!
    x2, y2 = p2  # Unpacking!
    
    dx = x2 - x1
    dy = y2 - y1
    return (dx*dx + dy*dy) ** 0.5

dist = entfernung(punkt_a, punkt_b)
print(f"Entfernung: {dist}")
```

### Datenbank-Zeilen

```python
# Studenten als Tupel (unveränderliche Datensätze)
studenten = [
    ("Anna", 20, "Informatik"),
    ("Bob", 22, "Mathematik"),
    ("Charlie", 19, "Physik")
]

print("Studentenliste:")
for student in studenten:
    name, alter, fach = student  # Unpacking in der Schleife!
    print(f"  {name} ({alter}) studiert {fach}")
```

### RGB-Farben

```python
# Farben als RGB-Tupel
rot = (255, 0, 0)
gruen = (0, 255, 0)
blau = (0, 0, 255)

print("Farben:")
farben = [("Rot", rot), ("Grün", gruen), ("Blau", blau)]

for name, (r, g, b) in farben:  # Verschachteltes Unpacking!
    print(f"  {name}: RGB({r}, {g}, {b})")
```
## 8. Listen vs Tupel - Wann was verwenden?

```python
# Listen für veränderliche Daten
einkaufsliste = ["Brot", "Milch"]  # Kann sich ändern
einkaufsliste.append("Käse")       # Neue Artikel hinzufügen
print(f"Einkaufsliste: {einkaufsliste}")

# Tupel für feste Daten
person = ("Anna", 25, "Berlin")    # Feste Struktur
# person[0] = "Bob"                # Geht nicht!
print(f"Person: {person}")
```

```python
# Tupel als Dictionary-Keys (Listen gehen nicht!)
positionen = {}
positionen[(0, 0)] = "Start"
positionen[(1, 1)] = "Spieler"
positionen[(5, 5)] = "Ziel"

print("Spielfeld:")
for (x, y), inhalt in positionen.items():
    print(f"  Position ({x}, {y}): {inhalt}")

# Das würde nicht funktionieren:
# positionen[[0, 0]] = "Start"  # Fehler: Listen sind nicht hashbar!
```
## 9. Konvertierung zwischen Listen und Tupeln

```python
# Liste zu Tupel
meine_liste = [1, 2, 3, 4, 5]
als_tupel = tuple(meine_liste)
print(f"Liste: {meine_liste}")
print(f"Als Tupel: {als_tupel}")

# Tupel zu Liste
mein_tupel = (10, 20, 30)
als_liste = list(mein_tupel)
print(f"Tupel: {mein_tupel}")
print(f"Als Liste: {als_liste}")
```

```python
# Wann ist das nützlich?
# Beispiel: Sammeln als Liste, dann als Tupel "einfrieren"
messungen = []
messungen.append(23.5)
messungen.append(24.1)
messungen.append(22.8)

# Als Tupel für unveränderliche Speicherung
final_messungen = tuple(messungen)
print(f"Finale Messungen: {final_messungen}")
```
## 10. Häufige Fehler vermeiden

```python
# FEHLER 1: Ein-Element-Tupel ohne Komma
falsch = (42)     # Das ist eine Zahl!
richtig = (42,)   # Das ist ein Tupel!

print(f"Falsch: {falsch}, Typ: {type(falsch)}")
print(f"Richtig: {richtig}, Typ: {type(richtig)}")

# FEHLER 2: Tupel ändern wollen
tupel = (1, 2, 3)
# tupel[0] = 99   # TypeError!

# RICHTIG: Neues Tupel erstellen
neues_tupel = (99,) + tupel[1:]
print(f"Neues Tupel: {neues_tupel}")
```
## Zusammenfassung

**Tupel erstellen:**
- Mehrere Elemente: `(1, 2, 3)`
- Ein Element: `(42,)` - Komma wichtig!
- Ohne Klammern: `1, 2, 3`
- Aus Liste: `tuple([1, 2, 3])`

**Wichtige Eigenschaften:**
- **Unveränderlich** - können nicht geändert werden
- **Geordnet** - Elemente haben feste Reihenfolge
- **Indexierbar** - `tupel[0]`, `tupel[-1]`
- **Slicing** - `tupel[1:3]`

**Tupel-Methoden:**
- `tupel.count(element)` - Anzahl Vorkommen
- `tupel.index(element)` - Position finden

**Tuple Unpacking:**
- `x, y = punkt`
- `erste, *rest, letzte = zahlen`
- `for name, alter in personen:`

**Wann verwenden:**
- ✅ **Koordinaten:** `(x, y, z)`
- ✅ **Datenbankzeilen:** Feste Struktur
- ✅ **Funktions-Rückgaben:** Mehrere Werte
- ✅ **Dictionary-Keys:** Zusammengesetzte Schlüssel
- ✅ **Konfiguration:** Unveränderliche Werte

**Listen vs Tupel:**
- **Listen:** Wenn sich Daten ändern können
- **Tupel:** Wenn Daten fest sind
