```python
# Zelle 1.1: Separate Funktionen für jede Rechenoperation
def addiere_zahlen(erste_zahl, zweite_zahl):
    """Addiert zwei Zahlen und gibt das Ergebnis zurück"""
    ergebnis = erste_zahl + zweite_zahl
    return ergebnis

def subtrahiere_zahlen(minuend, subtrahend):
    """Subtrahiert zweite Zahl von erster Zahl"""
    ergebnis = minuend - subtrahend
    return ergebnis

def multipliziere_zahlen(faktor1, faktor2):
    """Multipliziert zwei Zahlen miteinander"""
    ergebnis = faktor1 * faktor2
    return ergebnis

def dividiere_zahlen(dividend, divisor):
    """Dividiert erste Zahl durch zweite Zahl mit Fehlerbehandlung"""
    if divisor == 0:
        return "Fehler: Division durch Null nicht möglich"
    
    ergebnis = dividend / divisor
    return ergebnis

# Tests der einzelnen Funktionen
```

```python
# Zelle 2.2: Rabattlogik in eigener Funktion
def berechne_mengenrabatt_satz(bestellmenge):
    """Berechnet den Rabattsatz basierend auf der Bestellmenge"""
    
    # Konstanten (in echtem Code global definiert)
    GROSSMENGEN_SCHWELLE = 100
    MENGENRABATT_SCHWELLE = 50
    KLEINRABATT_SCHWELLE = 20
    GROSSMENGEN_RABATT = 0.15
    MENGENRABATT_SATZ = 0.10
    KLEINRABATT_SATZ = 0.05
    
    if bestellmenge >= GROSSMENGEN_SCHWELLE:
        rabattsatz = GROSSMENGEN_RABATT
        kategorie = "Großmengenrabatt"
    elif bestellmenge >= MENGENRABATT_SCHWELLE:
        rabattsatz = MENGENRABATT_SATZ
        kategorie = "Mengenrabatt"
    elif bestellmenge >= KLEINRABATT_SCHWELLE:
        rabattsatz = KLEINRABATT_SATZ
        kategorie = "Kleinrabatt"
    else:
        rabattsatz = 0.0
        kategorie = "Kein Rabatt"
    
    rabatt_info = {
        "satz": rabattsatz,
        "prozent": rabattsatz * 100,
        "kategorie": kategorie
    }
    
    print(f"🎯 RABATTBERECHNUNG:")
    print(f"   Menge: {bestellmenge} Stück")
    print(f"   Kategorie: {rabatt_info['kategorie']}")
    print(f"   Rabattsatz: {rabatt_info['prozent']}%")
    
    return rabattsatz

def berechne_endpreis(artikelpreis, bestellmenge):
    """Berechnet den Endpreis inklusive Mengenrabatt"""
    
    # Rabattsatz ermitteln
    rabattsatz = berechne_mengenrabatt_satz(bestellmenge)
    
    # Berechnungen durchführen
    grundpreis = artikelpreis * bestellmenge
    rabattbetrag = grundpreis * rabattsatz
    endpreis = grundpreis - rabattbetrag
    
    preis_details = {
        "artikelpreis": artikelpreis,
        "menge": bestellmenge,
        "grundpreis": grundpreis,
        "rabatt_prozent": rabattsatz * 100,
        "rabattbetrag": rabattbetrag,
        "endpreis": endpreis
    }
    
    print(f"\n💰 PREISBERECHNUNG:")
    print(f"   Artikelpreis: {preis_details['artikelpreis']:.2f}€")
    print(f"   Menge: {preis_details['menge']} Stück")
    print(f"   Grundpreis: {preis_details['grundpreis']:.2f}€")
    print(f"   Rabatt: {preis_details['rabatt_prozent']}% ({preis_details['rabattbetrag']:.2f}€)")
    print(f"   Endpreis: {preis_details['endpreis']:.2f}€")
    
    return endpreis

# Test der Preisberechnung
endpreis = berechne_endpreis(15.99, 75)
print(f"\nBestellsumme: {endpreis:.2f}€")
```

```python
# Zelle 3.1: DRY-Prinzip für Lagerprüfung
def pruefe_einzellager(lager_name, aktueller_bestand):
    """Prüft den Status eines einzelnen Lagers"""
    
    # Konstante für Mindestbestand
    MINDESTBESTAND = 100
    
    if aktueller_bestand < MINDESTBESTAND:
        status = "Nachbestellen!"
        status_symbol = "❌"
        aktion_erforderlich = True
    else:
        status = "Ausreichend"
        status_symbol = "✅"
        aktion_erforderlich = False
    
    lager_status = {
        "name": lager_name,
        "bestand": aktueller_bestand,
        "mindestbestand": MINDESTBESTAND,
        "status": status,
        "symbol": status_symbol,
        "nachbestellen": aktion_erforderlich
    }
    
    print(f"📦 LAGER EINZELPRÜFUNG:")
    print(f"   {lager_status['symbol']} {lager_status['name']}: {lager_status['bestand']} Stück")
    print(f"   Status: {lager_status['status']}")
    print(f"   Mindestbestand: {lager_status['mindestbestand']}")
    
    return lager_status

# Test der Einzellager-Prüfung
lager_a_status = pruefe_einzellager("Lager A", 150)
lager_b_status = pruefe_einzellager("Lager B", 89)
```

### Schritt 3.2: Gesamtlagerbericht mit DRY-Prinzip
```python
# Zelle 3.2: Vollständiger Lagerbericht ohne Code-Duplikation
def erstelle_gesamtlagerbericht():
    """Erstellt einen Bericht über alle Lager ohne Code-Duplikation"""
    
    # Lagerdaten (normalerweise aus Datenbank)
    alle_lager = {
        "Lager A": 150,
        "Lager B": 89,
        "Lager C": 234,
        "Lager D": 67,
        "Lager E": 201
    }
    
    # Alle Lager prüfen - KEINE Duplikation!
    lager_ergebnisse = []
    nachbestellung_erforderlich = 0
    
    for lager_name, bestand in alle_lager.items():
        # Wiederverwendung der Einzelprüfungs-Funktion
        status = pruefe_einzellager(lager_name, bestand)
        lager_ergebnisse.append(status)
        
        if status["nachbestellen"]:
            nachbestellung_erforderlich += 1
    
    # Gesamtbericht erstellen
    gesamtbericht = {
        "anzahl_lager": len(alle_lager),
        "nachbestellen": nachbestellung_erforderlich,
        "ausreichend": len(alle_lager) - nachbestellung_erforderlich,
        "details": lager_ergebnisse
    }
    
    print(f"\n📋 GESAMTLAGERBERICHT:")
    print("=" * 30)
    for status in gesamtbericht["details"]:
        print(f"{status['symbol']} {status['name']} ({status['bestand']}): {status['status']}")
    
    print(f"\n📊 ZUSAMMENFASSUNG:")
    print(f"   Gesamtlager: {gesamtbericht['anzahl_lager']}")
    print(f"   Ausreichend: {gesamtbericht['ausreichend']}")
    print(f"   Nachbestellen: {gesamtbericht['nachbestellen']}")
    
    if gesamtbericht["nachbestellen"] > 0:
        print(f"   ⚠️  Aktion erforderlich: {gesamtbericht['nachbestellen']} Lager nachbestellen!")
    else:
        print(f"   ✅ Alle Lager ausreichend gefüllt")
    
    return gesamtbericht

# Test des Gesamtberichts
lagerbericht = erstelle_gesamtlagerbericht()
print(f"\nLagerverwaltung: {lagerbericht['nachbestellen']} von {lagerbericht['anzahl_lager']} Lagern benötigen Nachbestellung")
```
