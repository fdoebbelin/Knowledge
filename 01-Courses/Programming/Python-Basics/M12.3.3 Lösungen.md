## Aufgabe 6: Berechtigungssystem

```python
# 1. Berechtigungssets definieren
admin_rechte = {"lesen", "schreiben", "löschen", "admin"}
editor_rechte = {"lesen", "schreiben", "bearbeiten"}
viewer_rechte = {"lesen"}

print("=== Berechtigungs-System ===")
print(f"Admin-Rechte: {admin_rechte}")
print(f"Editor-Rechte: {editor_rechte}")
print(f"Viewer-Rechte: {viewer_rechte}")

# 2. Benutzer und ihre Rollen
benutzer_rollen = {
    "alice": admin_rechte,
    "bob": editor_rechte,
    "charlie": viewer_rechte,
    "diana": admin_rechte,
    "eva": viewer_rechte
}

print(f"\nBenutzer und Rollen: {list(benutzer_rollen.keys())}")

# 3. Funktionen implementieren
def hat_berechtigung(benutzer, berechtigung):
    """
    Prüft ob ein Benutzer eine bestimmte Berechtigung hat.
    
    Args:
        benutzer (str): Benutzername
        berechtigung (str): Gewünschte Berechtigung
        
    Returns:
        bool: True wenn Berechtigung vorhanden, False wenn nicht
    """
    benutzer_rechte = benutzer_rollen.get(benutzer, set())
    return berechtigung in benutzer_rechte

def zeige_berechtigungen(benutzer):
    """Zeigt alle Berechtigungen eines Benutzers an"""
    rechte = benutzer_rollen.get(benutzer, set())
    if rechte:
        print(f"{benutzer} hat folgende Berechtigungen: {', '.join(sorted(rechte))}")
    else:
        print(f"{benutzer} ist nicht im System oder hat keine Berechtigungen")
    return rechte

def kann_bearbeiten(benutzer):
    """Prüft ob Benutzer bearbeiten kann (schreiben ODER bearbeiten)"""
    return (hat_berechtigung(benutzer, "schreiben") or 
            hat_berechtigung(benutzer, "bearbeiten"))

# Funktionen testen
print(f"\n=== Berechtigungs-Tests ===")

# Einzelne Berechtigungen testen
test_faelle = [
    ("alice", "löschen"),
    ("bob", "löschen"), 
    ("charlie", "lesen"),
    ("alice", "admin"),
    ("eva", "schreiben"),
    ("unbekannt", "lesen")
]

for benutzer, berechtigung in test_faelle:
    ergebnis = hat_berechtigung(benutzer, berechtigung)
    status = "✅ JA" if ergebnis else "❌ NEIN"
    print(f"{benutzer} kann '{berechtigung}': {status}")

# Alle Berechtigungen anzeigen
print(f"\n=== Alle Benutzer-Berechtigungen ===")
for benutzer in benutzer_rollen:
    zeige_berechtigungen(benutzer)

# Bearbeitungs-Tests
print(f"\n=== Bearbeitungs-Tests ===")
for benutzer in benutzer_rollen:
    kann_edit = kann_bearbeiten(benutzer)
    status = "✅ JA" if kann_edit else "❌ NEIN"
    print(f"{benutzer} kann bearbeiten: {status}")

# Zusätzliche Analyse: Wer hat welche Rechte?
print(f"\n=== Rechte-Analyse ===")
alle_rechte = admin_rechte | editor_rechte | viewer_rechte  # Vereinigung aller Rechte

for recht in sorted(alle_rechte):
    benutzer_mit_recht = [user for user in benutzer_rollen 
                         if hat_berechtigung(user, recht)]
    print(f"'{recht}': {', '.join(benutzer_mit_recht)}")
```

```python
for person in benutzerdaten:

    if name in name_vorkommen:
        name_vorkommen[name] += 1
    else:
        name_vorkommen[name] = 1

print(f"\n=== Duplikat-Analyse ===")
duplikate_gefunden = False
for name, anzahl in name_vorkommen.items():
    if anzahl > 1:
        print(f"'{name}' kommt {anzahl}x vor (Duplikat!)")
        duplikate_gefunden = True
    else:
        print(f"'{name}' kommt 1x vor")

    print("Keine Duplikate gefunden")

# 4. Bereinigte Liste ohne Duplikate erstellen
print(f"\n=== Bereinigung ===")

bereinigte_daten_v1 = []
gesehene_namen = set()

for person in benutzerdaten:
    name = person["name"]
    if name not in gesehene_namen:
        bereinigte_daten_v1.append(person)
        gesehene_namen.add(name)
```

```python
def erstelle_testdaten(groesse):
    """Erstellt Testdaten in verschiedenen Formaten"""
    daten = list(range(groesse))
    random.shuffle(daten)  # Zufällige Reihenfolge
    
    test_liste = daten.copy()
    test_set = set(daten)
    
    return test_liste, test_set

# Performance-Test-Funktion
def messe_zeit(func, *args):
    """Misst die Ausführungszeit einer Funktion"""
    start = time.time()
    ergebnis = func(*args)
    ende = time.time()
    return ende - start, ergebnis

# Test 1: Membership-Tests
def teste_membership(datenstruktur, element):
    """Testet ob Element in Datenstruktur vorhanden ist"""
    return element in datenstruktur

    speedup = liste_zeit / set_zeit if set_zeit > 0 else 0
```
