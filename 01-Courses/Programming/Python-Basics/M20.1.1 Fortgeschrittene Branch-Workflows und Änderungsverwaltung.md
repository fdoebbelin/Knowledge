## 1. Was sind fortgeschrittene Branch-Workflows?

### Problem mit linearer Entwicklung
Ohne systematische Branch-Workflows entstehen schnell Probleme:
- Mehrere Entwickler arbeiten gleichzeitig am selben Code
- Features werden halbfertig in den Hauptbranch integriert
- Fehlerhafte Änderungen blockieren die gesamte Entwicklung
- Code-Reviews sind schwierig ohne klare Feature-Abgrenzung
- Experimentelle Features gefährden die Stabilität

### Lösung mit Branch-Strategien
Professionelle Branch-Workflows lösen diese Probleme:
- **Feature-Isolation**: Jedes Feature entwickelt sich unabhängig
- **Parallele Entwicklung**: Teams arbeiten ohne gegenseitige Störung
- **Code-Qualität**: Reviews vor Integration ins Hauptprojekt
- **Stabilität**: main-Branch bleibt immer funktionsfähig
- **Flexibilität**: Einfaches Zurücknehmen problematischer Changes

### Git-Flow für Anfänger verstehen
Ein vereinfachter Git-Workflow besteht aus:
- **main-Branch**: Produktive, stabile Version
- **feature-Branches**: Neue Funktionen entwickeln
- **bugfix-Branches**: Fehler isoliert beheben
- **Merge-Strategie**: Kontrollierte Integration fertiger Features

def erklaere_feature_branching():
    """Erklärt das Konzept von Feature-Branches für parallele Entwicklung"""
    
    branch_strategien = {
        "feature/division-funktion": "Neue Division-Funktionalität isoliert entwickeln",
        "feature/multiplikation-erweitert": "Erweiterte Multiplikation mit Fehlerbehandlung",
        "bugfix/subtraction-edge-case": "Problem bei negativen Zahlen beheben",
        "hotfix/critical-security-patch": "Sicherheitslücke schnell schließen"
    }
    
    print("🌳 FEATURE-BRANCH-STRATEGIEN:")
    for branch, zweck in branch_strategien.items():
        branch_typ = branch.split('/')[0]
        emoji = "🚀" if branch_typ == "feature" else "🐛" if branch_typ == "bugfix" else "🚨"
        print(f"   {emoji} {branch}")
        print(f"      → {zweck}")
    
    return "Feature-Branches ermöglichen isolierte, parallele Entwicklung"

print(erklaere_feature_branching())

def demonstriere_merge_konflikt():
    """Simuliert einen Merge-Konflikt durch gleichzeitige Änderungen"""
    
    # Versuch, zweiten Branch zu mergen (wird Konflikt erzeugen)
    import subprocess
    
    merge_result = subprocess.run(
        ["git", "merge", "feature/multiplikation-erweitert"], 
        cwd="git-advanced-demo", 
        capture_output=True, 
        text=True
    )
    
    # Status nach Merge-Versuch prüfen
    status_result = subprocess.run(
        ["git", "status"], 
        cwd="git-advanced-demo", 
        capture_output=True, 
        text=True
    )
    
    konflikt_info = {
        "merge_erfolg": merge_result.returncode == 0,
        "merge_output": merge_result.stderr.strip(),
        "status_nach_merge": status_result.stdout.strip()
    }
    
    return konflikt_info

konflikt_result = demonstriere_merge_konflikt()
if not konflikt_result["merge_erfolg"]:
    print("⚠️ MERGE-KONFLIKT ERKANNT:")
    print("Grund:", konflikt_result["merge_output"])
    print("\n📋 Aktueller Status:")
    print(konflikt_result["status_nach_merge"])
else:
    print("✅ Merge erfolgreich ohne Konflikte!")

def analysiere_konflikt_datei():
    """Analysiert die Konflikt-Marker in der betroffenen Datei"""
    
    try:
        with open("git-advanced-demo/taschenrechner.py", "r", encoding="utf-8") as f:
            datei_inhalt = f.read()
        
        if "<<<<<<< HEAD" in datei_inhalt:
            konflikt_bereiche = []
            lines = datei_inhalt.split('\n')
            
            in_konflikt = False
            for i, line in enumerate(lines):
                if "<<<<<<< HEAD" in line:
                    konflikt_bereiche.append(f"Zeile {i+1}: Konflikt beginnt")
                    in_konflikt = True
                elif "=======" in line and in_konflikt:
                    konflikt_bereiche.append(f"Zeile {i+1}: Trennlinie zwischen Versionen")
                elif ">>>>>>>" in line and in_konflikt:
                    konflikt_bereiche.append(f"Zeile {i+1}: Konflikt endet")
                    in_konflikt = False
            
            return f"🔍 Konflikt-Analyse: {len(konflikt_bereiche)//3} Konflikte gefunden"
        else:
            return "ℹ️ Keine Konflikt-Marker gefunden"
            
    except FileNotFoundError:
        return "❌ Konflikt-Datei nicht gefunden"

print(analysiere_konflikt_datei())
