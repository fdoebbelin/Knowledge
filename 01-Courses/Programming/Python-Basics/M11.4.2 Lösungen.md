def bewerte_datenstruktur(beschreibung, aendert_sich, feste_struktur, als_key, performance_kritisch):
    """
    Hilfsfunktion zur systematischen Bewertung von Datenstrukturen.
    
    Parameter:
    - beschreibung: Beschreibung des Anwendungsfalls
    - aendert_sich: Ändern sich die Daten? (True/False)
    - feste_struktur: Ist die Struktur fest? (True/False)
    - als_key: Wird als Dictionary-Key verwendet? (True/False)
    - performance_kritisch: Ist Performance wichtig? (True/False)
    
    Returns:
    - "TUPEL" oder "LISTE" mit Begründung
    """
    tupel_punkte = 0
    listen_punkte = 0
    begruendung = []
    
    # Kriterium 1: Veränderbarkeit
    if not aendert_sich:
        tupel_punkte += 1
        begruendung.append("Daten ändern sich nicht → Tupel-Punkt")
    else:
        listen_punkte += 1
        begruendung.append("Daten ändern sich → Listen-Punkt")
    
    # Kriterium 2: Struktur
    if feste_struktur:
        tupel_punkte += 1
        begruendung.append("Feste Struktur → Tupel-Punkt")
    else:
        listen_punkte += 1
        begruendung.append("Variable Struktur → Listen-Punkt")
    
    # Kriterium 3: Dictionary-Key Verwendung
    if als_key:
        tupel_punkte += 1
        begruendung.append("Als Dictionary-Key → Tupel-Punkt")
    
    # Kriterium 4: Performance
    if performance_kritisch:
        tupel_punkte += 1
        begruendung.append("Performance wichtig → Tupel-Punkt")
    
    # Entscheidung treffen
    if tupel_punkte > listen_punkte:
        empfehlung = "TUPEL"
    elif listen_punkte > tupel_punkte:
        empfehlung = "LISTE"
    else:
        empfehlung = "BEIDE MÖGLICH (gleiche Punktzahl)"
    
    print(f"\n📊 Bewertung: {beschreibung}")
    for grund in begruendung:
        print(f"   {grund}")
    print(f"   Tupel-Punkte: {tupel_punkte}, Listen-Punkte: {listen_punkte}")
    print(f"   Empfehlung: {empfehlung}")
    
    return empfehlung

# Beispiel-Bewertungen anwenden
bewerte_datenstruktur("RGB-Farbwerte", False, True, True, True)
bewerte_datenstruktur("Chat-Teilnehmer", True, False, False, False)
bewerte_datenstruktur("3D-Koordinaten", False, True, True, True)

def messe_zugriffs_performance(datenstruktur, anzahl_zugriffe=100000):
    """
    Misst die Geschwindigkeit von Index-Zugriffen.
    
    Parameter:
    - datenstruktur: Liste oder Tupel zum Testen
    - anzahl_zugriffe: Anzahl der zufälligen Zugriffe
    
    Returns:
    - Durchschnittliche Zugriffszeit pro Operation
    """
    import random
    
    # Zufällige Indizes generieren (innerhalb der Grenzen)
    max_index = len(datenstruktur) - 1
    zufalls_indizes = [random.randint(0, max_index) for _ in range(anzahl_zugriffe)]
    
    # Zugriffs-Zeit messen
    start_zeit = time.time()
    for index in zufalls_indizes:
        # Einfacher Zugriff ohne weitere Verarbeitung
        _ = datenstruktur[index]
    gesamt_zeit = time.time() - start_zeit
    
    return gesamt_zeit

def erstelle_spielfeld():
    """
    Erstellt ein Dictionary-basiertes Spielfeld.
    Tupel-Koordinaten als Keys ermöglichen schnelle Position-Lookups.
    """
    spielfeld = {}
    
    # Verschiedene Spielfeld-Positionen definieren
    # Format: (x, y) -> {"typ": str, "inhalt": str, "eigenschaften": dict}
    positionen = [
        ((0, 0), {"typ": "start", "inhalt": "Eingang", "farbe": "grün"}),
        ((1, 1), {"typ": "münze", "inhalt": "10 Gold", "wert": 10}), 
        ((2, 1), {"typ": "hindernis", "inhalt": "Felsen", "durchgang": False}),
        ((3, 2), {"typ": "power_up", "inhalt": "Schild", "schutz": 5}),
        ((5, 5), {"typ": "ziel", "inhalt": "Ausgang", "farbe": "rot"})
    ]
    
    # Dictionary aufbauen - Tupel-Keys ermöglichen das
    for position, eigenschaften in positionen:
        spielfeld[position] = eigenschaften
    
    return spielfeld

def zeige_spielfeld(spielfeld):
    """Zeigt das Spielfeld strukturiert an."""
    print("\n🎮 Spielfeld-Layout:")
    print("   (x, y) -> Typ: Inhalt")
    print("   " + "="*30)
    
    for position, eigenschaften in spielfeld.items():
        x, y = position  # Tupel unpacking
        typ = eigenschaften["typ"]
        inhalt = eigenschaften["inhalt"]
        print(f"   ({x}, {y}) -> {typ}: {inhalt}")

def pruefe_position(spielfeld, position):
    """
    Prüft was sich an einer Position befindet.
    Demonstriert schnellen Dictionary-Lookup mit Tupel-Key.
    """
    if position in spielfeld:  # O(1) Lookup dank Hash-Tabelle
        return spielfeld[position]
    else:
        return {"typ": "leer", "inhalt": "Nichts hier", "begehbar": True}

def finde_typ(spielfeld, gesuchter_typ):
    """Findet alle Positionen eines bestimmten Typs."""
    gefundene_positionen = []
    
    for position, eigenschaften in spielfeld.items():
        if eigenschaften["typ"] == gesuchter_typ:
            gefundene_positionen.append(position)
    
    return gefundene_positionen

def berechne_entfernung(pos1, pos2):
    """Berechnet Manhattan-Distanz zwischen zwei Positionen."""
    x1, y1 = pos1  # Tupel unpacking
    x2, y2 = pos2
    return abs(x2 - x1) + abs(y2 - y1)  # Manhattan-Distanz

def berechne_rechteck(laenge, breite):
    """
    Berechnet Umfang, Fläche und Diagonale eines Rechtecks.
    
    Parameter:
    - laenge: Länge des Rechtecks
    - breite: Breite des Rechtecks
    
    Returns:
        tuple: (umfang, flaeche, diagonale)
        - umfang: Umfang des Rechtecks
        - flaeche: Fläche des Rechtecks  
        - diagonale: Länge der Diagonale
    """
    # Umfang berechnen: 2 * (Länge + Breite)
    umfang = 2 * (laenge + breite)
    
    # Fläche berechnen: Länge * Breite
    flaeche = laenge * breite
    
    # Diagonale berechnen: √(Länge² + Breite²) - Pythagoras
    diagonale = (laenge**2 + breite**2)**0.5
    
    # Tupel zurückgeben - Reihenfolge ist wichtig!
    return umfang, flaeche, diagonale

def berechne_kreis(radius):
    """
    Berechnet Umfang und Fläche eines Kreises.
    
    Parameter:
    - radius: Radius des Kreises
    
    Returns:
        tuple: (umfang, flaeche)
        - umfang: Umfang des Kreises (2πr)
        - flaeche: Fläche des Kreises (πr²)
    """
    import math
    
    # Umfang berechnen: 2 * π * radius
    umfang = 2 * math.pi * radius
    
    # Fläche berechnen: π * radius²
    flaeche = math.pi * radius**2
    
    # Tupel mit zwei Werten zurückgeben
    return umfang, flaeche

    - noten_liste: Liste von Noten (1.0 = beste, 6.0 = schlechteste)
    
    Returns:
        tuple: (durchschnitt, beste_note, schlechteste_note, anzahl)
        - durchschnitt: Arithmetisches Mittel der Noten
        - beste_note: Beste Note (niedrigster Wert)
        - schlechteste_note: Schlechteste Note (höchster Wert)
        - anzahl: Anzahl der Noten
    """
    # Edge Case: Leere Liste behandeln
    if not noten_liste:
        return None, None, None, 0
    
    # Durchschnitt berechnen
    durchschnitt = sum(noten_liste) / len(noten_liste)
    
    # In Deutschland: 1 = beste Note, 6 = schlechteste Note
    beste_note = min(noten_liste)      # Kleinster Wert = beste Note
    schlechteste_note = max(noten_liste)  # Größter Wert = schlechteste Note
    
    # Anzahl der Noten
    anzahl = len(noten_liste)
    
    # Strukturierte Rückgabe als Tupel
    return durchschnitt, beste_note, schlechteste_note, anzahl

def finde_extreme_werte(zahlen):
    """
    Findet Minimum, Maximum und Spannweite einer Zahlenreihe.
    
    Parameter:
    - zahlen: Liste oder Tupel von Zahlen
    
    Returns:
        tuple: (minimum, maximum, spannweite)
        - minimum: Kleinster Wert
        - maximum: Größter Wert
        - spannweite: Differenz zwischen Maximum und Minimum
    """
    # Edge Case: Leere Sequenz behandeln
    if not zahlen:
        return None, None, None
    
    # Extreme Werte finden
    minimum = min(zahlen)
    maximum = max(zahlen)
    
    # Spannweite berechnen
    spannweite = maximum - minimum
    
    # Alle drei Werte als Tupel zurückgeben
    return minimum, maximum, spannweite

def berechne_quartile(zahlen):
    """
    Berechnet Quartile einer Zahlenreihe.
    
    Parameter:
    - zahlen: Liste von Zahlen
    
    Returns:
        tuple: (q1, median, q3, iqr)
        - q1: Erstes Quartil (25. Perzentil)
        - median: Median (50. Perzentil)
        - q3: Drittes Quartil (75. Perzentil)
        - iqr: Interquartilsabstand (Q3 - Q1)
    """
    if not zahlen:
        return None, None, None, None
    
    # Zahlen sortieren für Quartil-Berechnung
    sortierte_zahlen = sorted(zahlen)
    n = len(sortierte_zahlen)
    
    # Median berechnen (mittlerer Wert)
    if n % 2 == 0:
        # Gerade Anzahl: Mittelwert der beiden mittleren Werte
        median = (sortierte_zahlen[n//2 - 1] + sortierte_zahlen[n//2]) / 2
    else:
        # Ungerade Anzahl: Mittlerer Wert
        median = sortierte_zahlen[n//2]
    
    # Q1 und Q3 berechnen (vereinfachte Methode)
    q1_index = n // 4
    q3_index = 3 * n // 4
    
    q1 = sortierte_zahlen[q1_index]
    q3 = sortierte_zahlen[q3_index]
    
    # Interquartilsabstand
    iqr = q3 - q1
    
    return q1, median, q3, iqr

def analysiere_text(text):
    """
    Analysiert einen Text und gibt verschiedene Statistiken zurück.
    
    Parameter:
    - text: Zu analysierender String
    
    Returns:
        tuple: (woerter, zeichen, saetze, durchschnitt_wort_laenge)
    """
    # Grundlegende Zählungen
    zeichen = len(text)
    
    # Wörter zählen (einfache Methode)
    woerter_liste = text.split()
    woerter = len(woerter_liste)
    
    # Sätze zählen (vereinfacht: nach . ! ? zählen)
    satzzeichen = text.count('.') + text.count('!') + text.count('?')
    saetze = max(1, satzzeichen)  # Mindestens 1 Satz
    
    # Durchschnittliche Wortlänge
    if woerter > 0:
        durchschnitt_wort_laenge = sum(len(wort.strip('.,!?')) for wort in woerter_liste) / woerter
    else:
        durchschnitt_wort_laenge = 0
    
    return woerter, zeichen, saetze, durchschnitt_wort_laenge

def berechne_koordinaten_statistik(koordinaten_liste):
    """
    Berechnet Statistiken für eine Liste von 2D-Koordinaten.
    
    Parameter:
    - koordinaten_liste: Liste von (x, y) Tupeln
    
    Returns:
        tuple: (zentroid, bounding_box, max_distanz)
        - zentroid: (center_x, center_y) - Mittelpunkt aller Koordinaten
        - bounding_box: ((min_x, min_y), (max_x, max_y)) - Umschließendes Rechteck
        - max_distanz: Größte Distanz zwischen zwei Punkten
    """
    if not koordinaten_liste:
        return None, None, None
    
    # Alle x und y Koordinaten extrahieren
    x_coords = [x for x, y in koordinaten_liste]
    y_coords = [y for x, y in koordinaten_liste]
    
    # Zentroid berechnen (Durchschnitt aller Punkte)
    zentroid_x = sum(x_coords) / len(x_coords)
    zentroid_y = sum(y_coords) / len(y_coords)
    zentroid = (zentroid_x, zentroid_y)  # Tupel als Rückgabe
    
    # Bounding Box berechnen
    min_x, max_x = min(x_coords), max(x_coords)
    min_y, max_y = min(y_coords), max(y_coords)
    bounding_box = ((min_x, min_y), (max_x, max_y))  # Verschachtelte Tupel
    
    # Maximale Distanz zwischen allen Punktpaaren finden
    max_distanz = 0
    for i, punkt1 in enumerate(koordinaten_liste):
        for punkt2 in koordinaten_liste[i+1:]:
            x1, y1 = punkt1
            x2, y2 = punkt2
            distanz = ((x2-x1)**2 + (y2-y1)**2)**0.5  # Euklidische Distanz
            max_distanz = max(max_distanz, distanz)
    
    return zentroid, bounding_box, max_distanz

def berechne_entfernung(pos1, pos2):
    """
    Berechnet die Entfernung zwischen zwei Positionen.
    
    Parameter:
    - pos1, pos2: Tupel mit (x, y) Koordinaten
    
    Returns:
        float: Euklidische Entfernung zwischen den Punkten
    """
    x1, y1 = pos1  # Tupel unpacking
    x2, y2 = pos2
    
    # Euklidische Distanz: √((x2-x1)² + (y2-y1)²)
    entfernung = ((x2 - x1) ** 2 + (y2 - y1) ** 2) ** 0.5
    return entfernung
